70 template <typename Rep, typename Period>
71 explicit TimeDelta(std::chrono::duration<Rep, Period> dur) {
72 // Convert to microseconds
73 auto microseconds = std::chrono::duration_cast<std::chrono::microseconds>(dur);
74 int64_t total = microseconds.count();
75 if (total < m_min_micro_seconds || total > m_max_micro_seconds) {
76 throw std::out_of_range("TimeDelta value out of range");
77 }
78 m_duration = bt::time_duration(0, 0, 0, total);
79 }
