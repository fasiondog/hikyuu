287 typedef FuncWrapper event_type;
288 ThreadSafeQueue<event_type> m_event_queue; // Message queue
289
292 template <typename ResultType>
293 using event_handle = std::future<ResultType>;
294
295 // Event-loop wait hooks (see setEventWaitHooks); empty in non-Python runtimes
296 std::function<void()> m_pre_wait_hook;
297 std::function<void()> m_post_wait_hook;
298
299 // Incremented on each stop(); a queued event task captures it and self-drops when stale,
300 // so a one-shot callback suppressed by stop() cannot run once after a restart
301 std::atomic<uint64_t> m_event_generation{0};
302
303 // Whether the event loop is actually running; stop() only wakes the loop when it is, so a
304 // stop() before start() (or during a backtest) leaves no stale wake-up task behind
305 std::atomic_bool m_event_loop_running{false};
306
307#ifdef _MSC_VER
308#pragma warning(push)
309#pragma warning(disable : 4996)
310#endif
311
313 template <typename FunctionType>
314 auto event(FunctionType f) {
315 typedef typename std::invoke_result<FunctionType>::type result_type;
316 uint64_t gen = m_event_generation.load(std::memory_order_acquire);
317 auto guarded = [this, gen, f = std::move(f)]() -> result_type {
318 if (gen != m_event_generation.load(std::memory_order_acquire)) {
319 return result_type();
320 }
321 return f();
322 };
323 std::packaged_task<result_type()> task(std::move(guarded));
324 event_handle<result_type> res(task.get_future());
325 m_event_queue.push(std::move(task));
326 return res;
327 }
