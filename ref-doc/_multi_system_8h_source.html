1/*
2 * Copyright (c) 2024 hikyuu.org
3 *
4 * Created on: 2024-09-13
5 * Author: fasiondog
6 *
7 * The aggregate trading system (portfolio backtesting).
8 *
9 * It holds multiple sub-systems (single-security systems or nested aggregates) and drives them
10 * on the open/close stages of a fully aligned time axis, then places the orders on the single
11 * real account of the top layer. It supports arbitrary nesting (with circular reference
12 * detection) and a rebalancing cycle, and carries the portfolio-level fund allocation (AF,
13 * see AllocateFundsBase for the L1/L2/L3 layers).
14 *
15 * Running modes (the mode is held by the aggregate system itself, set via setMode; the preset
16 * factories PF_Simple / PF_WithoutAF / PF_SignalAggregate configure it):
17 *
18 * - **Mode A "Signal Aggregation" (the default)**: every sub-system gets a shadow account
19 * funded with `m_sub_init_cash` as a pure signal source. The shadows run continuously and are
20 * not reset between the bars (run(reset=true) still resets them with the whole system), so
21 * every sub-system keeps submitting its position intent; on the sampling day
22 * the parent reads the selected ones' current net exposure snapshot and synthesizes one
23 * per-instrument signed net target by the L2 conversion (weight x net exposure ratio x the
24 * parent net assets), then orders uniformly on its own account. The shadow bookkeeping never
25 * touches the real funds.
26 *
27 * - **Mode B "Fund Allocation" (quota allocation, FOF/MOM style)**: every selected sub-system is
28 * calibrated to its quota on the rebalancing day BEFORE it is driven (zero-quota liquidation,
29 * then per sub the unselected liquidation or the selected cash recycle, then reduce the
30 * over-quota part, inject the gap), so the sub-system trades with the exact allocated quota; the
31 * parent mirrors the real instructions of the sub-systems (L2 pass-through) on its own account. The
32 * shadow accounts start from zero and follow the cost function of the parent account, keeping in
33 * step with the parent trades.
34 *
35 * - **Mode C "Shared Account Compatibility" (the legacy Portfolio behavior)**: no shadow account
36 * is created, every sub-system trades DIRECTLY on the single real account of the parent
37 * (`shared_tm`) and is sized by its own MM. The parent only gates the entry with the SE and
38 * keeps the running set driven day by day, so that the sell signal of a sub-system is realized on
39 * the very day it occurs; no L2 conversion is done and the parent does not place the order again.
40 * For master compatibility, buy_delay/sell_delay are forced to !trade_on_close. It exists to
41 * reproduce the results of the Portfolio before the refactoring.
42 *
43 * A suspended instrument comes in two data shapes, and the tradability is decided by the same
44 * guards as the System entries:
45 *
46 * - no bar on the suspended day (the daily source drops the day): the leaf cannot run, so the mode
47 * A sampling values it at the last quoted close and folds its queued requests from the request
48 * lists; the parent legs wait for the next tradable open;
49 * - a padded bar on the suspended day (open == high == low == close == the previous close, zero
50 * volume): the leaf runs, but the entry guards refuse the trade and queue a delayed request; the
51 * parent applies the same guards to its own legs, so it never fills at the padded price either.
52 *
53 * Minute data carries no padded suspension day at all (only the zero-volume minutes), so
54 * transAmount/transCount == 0 is the only suspension test there, and can_trade_when_high_eq_low
55 * decides whether a one-price bar is tradable (on daily data the same parameter separates a
56 * one-price board from a padded suspension bar). The mode B pass-through legs are not re-judged:
57 * the sub-system already gated them with its own parameters.
58 *
59 * Serialization restores the shadow-account registry and the scheduling counters, so a loaded
60 * system can continue with run(reset=false); the mode C running pools are rebuilt on the next
61 * rebalancing day.
62 *
63 * See the doc comments of AllocateFundsBase for the precise L1/L2/L3 semantics and PF_Simple /
64 * PF_WithoutAF for the preset configurations of the modes.
65 */

847 friend class boost::serialization::access;
848 template <class Archive>
849 void serialize(Archive& ar, const unsigned int) {
850 // The current schema only; no compatibility with the older archives
851 ar& BOOST_SERIALIZATION_BASE_OBJECT_NVP(System);
852 ar& BOOST_SERIALIZATION_NVP(m_sys_list);
853 // The shadow-account registry is a bare pointer set: save it by the child slot index and
854 // rebuild it after loading, otherwise a continued run would discard the restored child
855 // books and liquidate the parent holdings as orphans in mode A
856 std::vector<int> shadow;
857 if (Archive::is_saving::value) {
858 shadow.resize(m_sys_list.size(), 0);
859 for (size_t i = 0; i < m_sys_list.size(); ++i) {
860 shadow[i] = m_shadow_sys.count(m_sys_list[i].get()) ? 1 : 0;
861 }
862 }
863 ar& BOOST_SERIALIZATION_NVP(shadow);
864 if (Archive::is_loading::value) {
865 m_shadow_sys.clear();
866 for (size_t i = 0; i < shadow.size() && i < m_sys_list.size(); ++i) {
867 if (shadow[i]) {
868 m_shadow_sys.insert(m_sys_list[i].get());
869 }
870 }
871 }
872 ar& BOOST_SERIALIZATION_NVP(m_sub_init_cash);
873 ar& BOOST_SERIALIZATION_NVP(m_mode);
874 ar& BOOST_SERIALIZATION_NVP(m_af);
875 ar& BOOST_SERIALIZATION_NVP(m_se);
876 ar& BOOST_SERIALIZATION_NVP(m_path);
877
878 // The scheduling phase must survive, otherwise a loaded system restarts the rebalancing
879 // cycle at zero and the rebalancing days shift
880 uint64_t close_index =
881 static_cast<uint64_t>(Archive::is_saving::value ? m_close_day_index : 0);
882 ar& boost::serialization::make_nvp("close_day_index", close_index);
883 if (Archive::is_loading::value) {
884 m_close_day_index = static_cast<size_t>(close_index);
885 }
886 uint64_t last_close = m_last_close_date.number();
887 ar& boost::serialization::make_nvp("last_close_date", last_close);
888 if (Archive::is_loading::value) {
889 m_last_close_date = Datetime(last_close);
890 }
891
892 // Configuration lists (calendar axis / external rebalancing dates): they drive the
893 // trading days and the rebalancing days, so they must be restored with the system
894 std::vector<uint64_t> axis;
895 std::vector<uint64_t> adjust_days;
896 if (Archive::is_saving::value) {
897 axis.reserve(m_date_axis.size());
898 for (const auto& d : m_date_axis) {
899 axis.push_back(d.number());
900 }
901 adjust_days.reserve(m_adjust_dates.size());
902 for (const auto& d : m_adjust_dates) {
903 adjust_days.push_back(d.number());
904 }
905 }
906 ar& boost::serialization::make_nvp("date_axis", axis);
907 ar& boost::serialization::make_nvp("adjust_dates", adjust_days);
908 if (Archive::is_loading::value) {
909 m_date_axis.clear();
910 for (auto n : axis) {
911 m_date_axis.push_back(Datetime(n));
912 }
913 m_adjust_dates.clear();
914 for (auto n : adjust_days) {
915 m_adjust_dates.insert(Datetime(n));
916 }
917 }
918
919 // The deferred calibration legs: kind / child slot / instrument / number / quota
920 std::vector<int> cal_kind;
921 std::vector<int> cal_sys;
922 std::vector<Stock> cal_stock;
923 std::vector<double> cal_number;
924 std::vector<double> cal_quota;
925 if (Archive::is_saving::value) {
926 for (const auto& a : m_pending_calibration) {
927 int slot = -1;
928 for (size_t i = 0; i < m_sys_list.size(); ++i) {
929 if (m_sys_list[i] == a.sys) {
930 slot = static_cast<int>(i);
931 break;
932 }
933 }
934 cal_kind.push_back(static_cast<int>(a.kind));
935 cal_sys.push_back(slot);
936 cal_stock.push_back(a.stock);
937 cal_number.push_back(a.number);
938 cal_quota.push_back(a.quota);
939 }
940 }
941 ar& boost::serialization::make_nvp("cal_kind", cal_kind);
942 ar& boost::serialization::make_nvp("cal_sys", cal_sys);
943 ar& boost::serialization::make_nvp("cal_stock", cal_stock);
944 ar& boost::serialization::make_nvp("cal_number", cal_number);
945 ar& boost::serialization::make_nvp("cal_quota", cal_quota);
946 if (Archive::is_loading::value) {
947 m_pending_calibration.clear();
948 for (size_t i = 0; i < cal_kind.size() && i < cal_sys.size() && i < cal_stock.size() &&
949 i < cal_number.size() && i < cal_quota.size();
950 ++i) {
951 if (cal_sys[i] < 0 || cal_sys[i] >= static_cast<int>(m_sys_list.size())) {
952 continue;
953 }
954 m_pending_calibration.push_back({static_cast<CalibrationKind>(cal_kind[i]),
955 m_sys_list[cal_sys[i]], cal_stock[i],
956 cal_number[i], cal_quota[i]});
957 }
958 }
959 }
960#endif /* HKU_SUPPORT_SERIALIZATION */
961};
