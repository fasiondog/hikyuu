43typedef std::pair<size_t, size_t> range_t;
44
45inline std::vector<range_t> parallelIndexRange(size_t start, size_t end, size_t cpu_num = 0) {
46 std::vector<std::pair<size_t, size_t>> ret;
47 if (start >= end) {
48 return ret;
49 }
50
51 size_t total = end - start;
52 if (cpu_num == 0) {
53 cpu_num = std::thread::hardware_concurrency();
54 }
55 if (cpu_num <= 1) {
56 ret.emplace_back(start, end);
57 return ret;
58 }
59
60 size_t per_num = total / cpu_num;
61 if (per_num > 0) {
62 for (size_t i = 0; i < cpu_num; i++) {
63 size_t first = i * per_num + start;
64 ret.emplace_back(first, first + per_num);
65 }
66 }
67
68 for (size_t i = per_num * cpu_num + start; i < end; i++) {
69 ret.emplace_back(i, i + 1);
70 }
71
72 return ret;
73}

75template <typename FunctionType, class TaskGroup = MQThreadPool>
76void parallel_for_index_void(size_t start, size_t end, FunctionType f, int cpu_num = 0) {
77 auto ranges = parallelIndexRange(start, end, cpu_num);
78 if (ranges.empty()) {
79 return;
80 }
81
82 TaskGroup tg(cpu_num == 0 ? std::thread::hardware_concurrency() : cpu_num);
83 for (size_t i = 0, total = ranges.size(); i < total; i++) {
84 tg.submit([=, range = ranges[i]]() {
85 for (size_t ix = range.first; ix < range.second; ix++) {
86 f(ix);
87 }
88 });
89 }
90 tg.join();
91 return;
92}

94template <typename FunctionType, class TaskGroup = MQThreadPool>
95auto parallel_for_index(size_t start, size_t end, FunctionType f, size_t cpu_num = 0) {
96 std::vector<typename std::invoke_result<FunctionType, size_t>::type> ret;
97 auto ranges = parallelIndexRange(start, end, cpu_num);
98 if (ranges.empty()) {
99 return ret;
100 }
101
102 TaskGroup tg(cpu_num == 0 ? std::thread::hardware_concurrency() : cpu_num);
103 std::vector<std::future<std::vector<typename std::invoke_result<FunctionType, size_t>::type>>>
104 tasks;
105 for (size_t i = 0, total = ranges.size(); i < total; i++) {
106 tasks.emplace_back(tg.submit([func = f, range = ranges[i]]() {
107 std::vector<typename std::invoke_result<FunctionType, size_t>::type> one_ret;
108 for (size_t ix = range.first; ix < range.second; ix++) {
109 one_ret.emplace_back(func(ix));
110 }
111 return one_ret;
112 }));
113 }
114
115 for (auto& task : tasks) {
116 auto one = task.get();
117 for (auto&& value : one) {
118 ret.emplace_back(std::move(value));
119 }
120 }
121
122 return ret;

204size_t HKU_UTILS_API remaining_stack_size(size_t* total_size = nullptr);

207inline bool has_stack_for_foreign_task() {
208 size_t total_size = 0;
209 size_t remaining = remaining_stack_size(&total_size);
210 // Keep the original behavior when the stack range is unknown
211 HKU_IF_RETURN(remaining == 0 || total_size == 0, true);
212
213 // Keeps a quarter of the stack, at least 256KB, free for the task to be executed
214 constexpr size_t kMinReserve = 256 * 1024;
215 size_t reserve = std::max(total_size / 4, kMinReserve);
216 return remaining > reserve;
217}

236template <typename FutureContainer>
237void wait_for_all_non_blocking(GlobalStealThreadPool& pool, FutureContainer& futures) {
238 // If the current thread is a worker thread, its child tasks are added to the front of its own
239 // queue and the other threads cannot get the child tasks, so a wake-up is needed
240 // When it is not a worker thread, its child tasks are added to the master queue and no active
241 // wake-up is needed
242 bool is_work_thread = GlobalStealThreadPool::is_work_thread();
243 if (is_work_thread) {
244 pool.wake_up();
245 }
246
247 bool all_ready = false;
248 auto init_delay = std::chrono::microseconds(1);
249 auto delay = init_delay;
250 const auto max_delay = std::chrono::microseconds(1000);
251 int spin_count = 0;
252
253 while (!all_ready && !pool.done()) {
254 all_ready = true;
255 for (auto& future : futures) {
256 if (future.wait_for(std::chrono::nanoseconds(0)) != std::future_status::ready) {
257 all_ready = false;
258 break;
259 }
260 }
261
262 // If not all the tasks are finished, try to execute the accumulated tasks
263 if (!all_ready) {
264 bool helped = false;
265 if (is_work_thread) {
266 helped = pool.run_available_task_once(!detail::has_stack_for_foreign_task());
267 } else if (detail::help_reentry_depth() == 0 && detail::has_stack_for_foreign_task()) {
268 detail::HelpReentryGuard guard;
269 helped = pool.run_available_task_once();
270 }
271 if (helped) {
272 delay = init_delay;
273 spin_count = 0;
274 } else if (pool.done()) {
275 return;
276 } else {
277 // When there is no task to steal at present, the unfinished tasks may be running in
278 // the other worker threads. First yield the time slice several times, then enter
279 // the microsecond level exponential backoff (with an upper limit of 1ms), ensuring
280 // that there is no leftover task when the subsequent get of the caller throws an
281 // exception, while avoiding a long busy wait.
282 if (spin_count < 8) {
283 ++spin_count;
284 std::this_thread::yield();
285 } else {
286 std::this_thread::sleep_for(delay);
287 if (delay < max_delay) {
288 delay = std::min(delay * 2, max_delay);
289 }
290 }
291 }
292 }
293 }

313template <typename FutureType>
314void global_wait_task(FutureType& future) {
315 auto* tg = get_global_task_group();
316 bool ready = false;
317 auto init_delay = std::chrono::microseconds(1);
318 auto delay = init_delay;
319 const auto max_delay = std::chrono::microseconds(50000);
320
321 while (!ready && !tg->done()) {
322 ready = true;
323 if (future.wait_for(std::chrono::nanoseconds(0)) != std::future_status::ready) {
324 ready = false;
325 }
326
327 // If the task is not finished, try to execute the local tasks
328 if (!ready) {
329 bool helped = false;
330 if (GlobalStealThreadPool::is_work_thread()) {
331 helped = tg->run_available_task_once(!detail::has_stack_for_foreign_task());
332 } else if (detail::help_reentry_depth() == 0 && detail::has_stack_for_foreign_task()) {
333 detail::HelpReentryGuard guard;
334 helped = tg->run_available_task_once();
335 }
336 if (helped) {
337 delay = init_delay;
338 } else if (tg->done()) {
339 break;
340 } else {
341 std::this_thread::sleep_for(delay);
342 if (delay < max_delay) {
343 delay = std::min(delay * 2, max_delay);
344 }
345 }
346 }
347 }

350template <typename FunctionType>
351auto global_parallel_for_index_void(size_t start, size_t end, FunctionType&& f,
352 size_t threshold = 2, bool enable_nested = true) {
353 HKU_IF_RETURN(start >= end, void());
354
355 // If the number of the tasks is less than the threshold, or the current thread is a worker
356 // thread and the nesting is forbidden, it is executed directly
357 if ((end - start) < threshold || (!enable_nested && GlobalStealThreadPool::is_work_thread())) {
358 for (size_t i = start; i < end; i++) {
359 f(i);
360 }
361 return;
362 }
363
364 auto* tg = get_global_task_group();
365 HKU_ASSERT(tg);
366
367 auto ranges = parallelIndexRange(start, end, tg->worker_num());
368 if (ranges.empty()) {
369 return;
370 }
371
372 std::vector<std::future<void>> tasks;
373 tasks.reserve(ranges.size());
374 for (size_t i = 0, total = ranges.size(); i < total; i++) {
375 tasks.emplace_back(tg->submit([func = f, range = ranges[i]]() {
376 for (size_t ix = range.first; ix < range.second; ix++) {
377 func(ix);
378 }
379 }));
380 }
381
382 wait_for_all_non_blocking(*tg, tasks);
383
384 for (auto& task : tasks) {
385 task.get();
386 }
387
388 return;
389}

587template <typename T>
588auto await_future(std::future<T> fut) -> asio::awaitable<T> {
589 auto exec = co_await asio::this_coro::executor;
590 asio::steady_timer timer(exec);
591
592 // Use the timer for an efficient polling, avoiding the busy wait
593 while (fut.wait_for(std::chrono::seconds(0)) != std::future_status::ready) {
594 timer.expires_after(std::chrono::microseconds(100));
595 co_await timer.async_wait(asio::use_awaitable);
596 }
597
598 co_return fut.get(); // It may throw an exception
599}

776template <typename Executor, typename Func>
777inline auto submit_future(Executor exec, Func&& func)
778 -> std::shared_ptr<std::future<typename std::invoke_result_t<Func>>> {
779 using result_type = typename std::invoke_result_t<Func>;
780
781 std::future<result_type> future;
782 try {
783 auto task = std::make_shared<std::packaged_task<result_type()>>(std::forward<Func>(func));
784 future = task->get_future();
785 exec.execute([task = std::move(task)]() mutable { (*task)(); });
786 } catch (...) {
787 std::promise<result_type> promise;
788 promise.set_exception(std::current_exception());
789 future = promise.get_future();
790 }
791 return std::make_shared<std::future<result_type>>(std::move(future));
792}

804template <typename T>
805inline asio::awaitable<T> await_future_ec(std::shared_ptr<std::future<T>> future) {
806 auto exec = co_await asio::this_coro::executor;
807 asio::steady_timer timer(exec);
808 try {
809 while (future->wait_for(std::chrono::seconds(0)) != std::future_status::ready) {
810 timer.expires_after(std::chrono::microseconds(100));
811 co_await timer.async_wait(asio::use_awaitable);
812 }
813 if constexpr (std::is_void_v<T>) {
814 future->get();
815 co_return;
816 } else {
817 co_return future->get();
818 }
819 } catch (const boost::system::system_error&) {
820 // Timer cancellation/failure (e.g. operation_aborted): keep the original error code
821 throw;
822 } catch (const std::exception&) {
823 throw boost::system::system_error(
824 boost::system::errc::make_error_code(boost::system::errc::io_error));
825 } catch (...) {
826 throw boost::system::system_error(
827 boost::system::errc::make_error_code(boost::system::errc::invalid_argument));
828 }
829}

881template <typename Executor, typename Func>
882auto co_run(Executor exec, Func func) -> asio::awaitable<typename std::invoke_result_t<Func>> {
883 // Take func by value: as a coroutine parameter it must outlive the caller's temporary, since
884 // the coroutine may be kept suspended long after the call expression ends
885 // Never move the awaitable handler onto exec: poll the future from the home executor so a
886 // destroyed home io_context can never be posted back to
887 co_return co_await await_future(detail::submit_future(exec, std::forward<Func>(func)));
888}
