19// Calculate the industry neutralized factor: subtract the group mean within every industry group,
20// residual = y_i - mean(y_{group(i)}).
21//
22// Mathematical equivalence: when the industry attribution is mutually exclusive (a stock belongs to
23// one industry only) and only this one categorical variable is neutralized,
24// "taking the residual of the one-hot multiple regression without the global intercept" is strictly
25// equivalent to "subtracting the group mean":
26// min_{β} Σ_i (y_i - Σ_k X_{i,k} β_k)^2; since X is a mutually exclusive one-hot, it can be split
27// into the independent sums of every industry,
28// the analytical solution is β_k = mean(y_{group(k)}) and the residual is
29// e_i = y_i - mean(y_{group(i)}).
30// Therefore there is no need to build the design matrix and perform the QR decomposition; two
31// passes of scanning O(n) are enough, and it is naturally compatible with the daily dynamic NaN
32// mask (the invalid values are skipped when calculating the group mean), avoiding the model
33// misspecification of the original integer encoding + univariate regression (the result depends on
34// the arrangement order of the block list) and the contention of the Eigen matrix heap allocation
35// under multiple threads.
36//
37// labels[i] = the industry index of the stock i (0..blk_count-1), and it is blk_count if there is
38// no attribution (skipped and the residual is set to NaN).
39inline PriceList calculate_industry_residuals(const PriceList& y, const PriceList& labels,
40 size_t blk_count) {
41 HKU_ASSERT(y.size() == labels.size());
42
43 const size_t n = y.size();
44 PriceList residuals(n, Null<price_t>());
45
46 // When the whole market is one industry (k=1) or there is no industry classification,
47 // subtracting the mean degrades to subtracting the global mean; all NaN when there is no valid
48 // data.
49 if (blk_count == 0) {
50 return residuals;
51 }
52
53 std::vector<double> sums(blk_count, 0.0);
54 std::vector<size_t> counts(blk_count, 0);
55
56 // Pass 1: accumulate the sum and the count of the valid y of every group (the NaN/Inf and the
57 // stocks without attribution are skipped)
58 for (size_t i = 0; i < n; ++i) {
59 if (std::isnan(y[i]) || std::isinf(y[i])) {
60 continue;
61 }
62 double label = labels[i];
63 // Guard against NaN/Inf/negative/huge values: static_cast<size_t>(a negative value) is UB,
64 // it may overflow to SIZE_MAX and cause a segfault of sums[g] out of range, so it must be
65 // intercepted with a double comparison before the cast.
66 if (std::isnan(label) || std::isinf(label) || label < 0.0 ||
67 label >= static_cast<double>(blk_count)) {
68 continue; // No attribution or an illegal label does not participate in the group mean
69 }
70 size_t g = static_cast<size_t>(label);
71 sums[g] += y[i];
72 counts[g]++;
73 }
74
75 // Pass 2: residual = y_i - mean(y_{group(i)})
76 for (size_t i = 0; i < n; ++i) {
77 if (std::isnan(y[i]) || std::isinf(y[i])) {
78 continue; // Keep NaN (consistent with the semantics of the original
79 // calculate_residuals)
80 }
81 double label = labels[i];
82 if (std::isnan(label) || std::isinf(label) || label < 0.0 ||
83 label >= static_cast<double>(blk_count)) {
84 continue; // No attribution or an illegal label: the residual is set to NaN
85 }
86 size_t g = static_cast<size_t>(label);
87 if (counts[g] == 0) {
88 continue; // The group has no valid sample: the residual is set to NaN
89 }
90 residuals[i] = y[i] - sums[g] / static_cast<double>(counts[g]);
91 }
92
93 return residuals;
94}
