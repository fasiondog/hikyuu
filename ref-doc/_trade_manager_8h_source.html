460 // Transient incremental short-circuit cache for updateWithWeight: the exclusive upper bound
461 // of the ex-rights date already checked for that stock. Only point lookups/assigns (never

463 // serialized and reset on _reset; the m_tmp_ prefix marks non-persistent derived state.
464 // Assumes the ex-rights data is final before the account advances: within a process lifetime it

503 ar& BOOST_SERIALIZATION_NVP(m_init_datetime);
504 ar& BOOST_SERIALIZATION_NVP(m_init_cash);

512 namespace bs = boost::serialization;
513 BorrowRecordList borrow;
514 ar& bs::make_nvp<BorrowRecordList>("m_borrow_stock", borrow);
515 BorrowRecordList::const_iterator bor_iter = borrow.begin();
516 for (; bor_iter != borrow.end(); ++bor_iter) {
517 m_borrow_stock[bor_iter->stock.id()] = *bor_iter;
518 }
519 PositionRecordList position;
520 ar& bs::make_nvp<PositionRecordList>("m_position", position);
521 PositionRecordList::const_iterator iter = position.begin();
522 for (; iter != position.end(); ++iter) {
523 m_position[iter->stock.id()] = *iter;
524 }
525 ar& BOOST_SERIALIZATION_NVP(m_position_history);
526
527 position.clear();
528 ar& bs::make_nvp<PositionRecordList>("m_short_position", position);
529 iter = position.begin();
530 for (; iter != position.end(); ++iter) {
531 m_short_position[iter->stock.id()] = *iter;
532 }
533 ar& BOOST_SERIALIZATION_NVP(m_short_position_history);
534 ar& BOOST_SERIALIZATION_NVP(m_trade_list);
535 ar& BOOST_SERIALIZATION_NVP(m_actions);
