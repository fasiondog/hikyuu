32struct HKU_API MomentResult {
33 Datetime datetime; // The corresponding moment
34 FundsRecord funds_before_open; // The [before open trade] snapshot -> the denominator of the tradesOnOpen ratio
35 FundsRecord funds_before_close; // The [before close trade] snapshot (= after the open execution) -> the denominator of the tradesOnClose ratio
36 FundsRecord funds; // The final balance / total assets / market value at this moment (including the long/short and the margin trading)
37 std::vector<PositionRecord> positions; // The current positions (multiple instruments for the aggregate form; lazily filled)
38 TradeRecordList tradesOnOpen; // The trades executed at the [open] stage
39 TradeRecordList tradesOnClose; // The trades executed at the [close] stage
40 std::vector<TradeRequest> delayOnNextOpen; // The requests delayed to the open of the next moment
41 TradeSuggestionList suggestions; // The suggestion instructions produced at this moment (the complete semantics)
42 TradeSuggestionList pendingOnNextOpen; // The aggregate target orders to execute at the next
43 // open (trade_on_close=false); the multi-instrument
44 // counterpart of delayOnNextOpen
45
46 json ext; // The extensible area (lazily constructed, an empty object by default)
47
48 TradeRecordList allTrades() const {
49 TradeRecordList r = tradesOnOpen;
50 r.insert(r.end(), tradesOnClose.begin(), tradesOnClose.end());
51 return r;
52 }
53
54 bool empty() const {
55 return tradesOnOpen.empty() && tradesOnClose.empty() && suggestions.empty();
56 }
57
58 json toJson() const {
59 json j;
60 j["datetime"] = datetime.str();
61 auto append_trade = [](json& arr, const TradeRecord& tr) {
62 json r;
63 r["datetime"] = tr.datetime.str();
64 r["number"] = tr.number;
65 r["realPrice"] = tr.realPrice;
66 r["from"] = static_cast<int>(tr.from);
67 arr.push_back(r);
68 };
69 json open = json::array();
70 for (const auto& tr : tradesOnOpen) {
71 append_trade(open, tr);
72 }
73 j["tradesOnOpen"] = open;
74 json close = json::array();
75 for (const auto& tr : tradesOnClose) {
76 append_trade(close, tr);
77 }
78 j["tradesOnClose"] = close;
79 j["suggestions"] = suggestions.size();
80 j["ext"] = ext;
81 return j;
82 }

84 template <typename T>
85 T get(const string& key, const T& def) const {
86 if (ext.is_object()) {
87 auto it = ext.find(key);
88 if (it != ext.end()) {
89 return it->get<T>();
90 }
91 }
92 return def;
93 }
94
95 template <typename T>
96 void set(const string& key, const T& value) {
97 ext[key] = value;
98 }
99};
