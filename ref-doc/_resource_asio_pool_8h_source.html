100 explicit ResourceAsioPool(const Parameter &param, size_t max_count = 128,
101 size_t max_waiters = 1000)
102 : m_maxCount(max_count),
103 m_param(param),
104 m_resourceList(max_count == 0 ? 128 : max_count),
105 m_maxWaiters(max_waiters > 0 ? max_waiters : 1000),
106 m_waiters(std::make_shared<WaiterRegistry>()) {
107 // The waiting queue is reserved up front, so joining it never allocates and never throws
108 m_waiters->queue.reserve(m_maxWaiters);
109 if (max_waiters == 0) {
110 HKU_WARN("ResourceAsioPool: max_waiters is 0, using default 1000");
111 } else if (m_maxWaiters > 10000) {
112 double estimated_mb = m_maxWaiters * 150.0 / 1024 / 1024;
113 HKU_WARN(
114 "ResourceAsioPool: large max_waiters={}, "
115 "estimated memory usage at full capacity: ~{:.1f}MB",
116 m_maxWaiters, estimated_mb);
117 }

123 virtual ~ResourceAsioPool() {
124 // Mark that it is being destructed, preventing new resource acquisitions and returns
125 m_is_destroying.store(true, std::memory_order_release);
126
127 // Take all the waiters away at once. Every node is co-owned by the waiting queue and by the
128 // waiting coroutine, so whoever claims the reserved resource first owns it, while the node
129 // itself stays alive as long as the other side still holds a reference.
130 std::vector<std::weak_ptr<WaiterNode>> waiters;
131 {
132 std::lock_guard<std::mutex> lock(m_waiters->mutex);
133 waiters.swap(m_waiters->queue);
134 }
135 m_registeredWaiters.fetch_sub(waiters.size(), std::memory_order_relaxed);
136
137 for (const auto &waiter_ref : waiters) {
138 auto waiter = waiter_ref.lock();
139 if (!waiter) {
140 // The waiting coroutine is gone, its node dies together with it
141 continue;
142 }
143 ResourceType *p =
144 waiter->handed.exchange(WaiterNode::gaveUp(), std::memory_order_acq_rel);
145 // Only the resource nobody will ever claim may be released here. The waiting coroutines
146 // are not timed out explicitly: whenever one of them resumes it finds gaveUp() and
147 // takes the time out branch, which reads nothing from the pool any more.
148 if (p != nullptr && p != WaiterNode::gaveUp()) {
149 delete p;
150 releaseResourceCount();
151 }
152 }
153
154 // Wait for all the active resources to be returned
155 // When m_count == m_idleCount all the resources have been returned to the idle queue
156 std::unique_lock<MutexType> lock(m_destroy_mutex);
157 m_destroy_waiting.store(true, std::memory_order_seq_cst);
158 m_destroy_cv.wait(lock, [this]() {
159 return m_count.load(std::memory_order_seq_cst) ==
160 m_idleCount.load(std::memory_order_seq_cst);
161 });
162 m_destroy_waiting.store(false, std::memory_order_relaxed);
163
164 // At this time all the resources are in the idle queue, release them
165 ResourceType *p = nullptr;
166 while (m_resourceList.pop(p)) {
167 if (p) {
168 delete p;
169 }
170 }
171 }

204 stdx::expected<ResourcePtr, std::string> get() {
205 // 1. Try to acquire a resource from the idle queue
206 ResourceType *p = takeIdleResource();
207 if (p) {
208 return stdx::expected<ResourcePtr, std::string>(ResourcePtr(p, ResourceCloser(this)));
209 }
210
211 // 2. There is no idle resource but the upper limit has not been reached -> create a new
212 // resource. The slot is reserved with a CAS first, so the upper limit stays a hard one
213 // even when several threads create resources at the same time
214 if (reserveResourceSlot()) {
215 try {
216 p = new ResourceType(m_param);
217 } catch (const std::exception &e) {
218 releaseResourceCount();
219 return stdx::unexpected(
220 std::string(fmt::format("Failed create a new Resource! {}", e.what())));
221 } catch (...) {
222 releaseResourceCount();
223 return stdx::unexpected(
224 std::string("Failed create a new Resource! Unknown error!"));
225 }
226 return stdx::expected<ResourcePtr, std::string>(ResourcePtr(p, ResourceCloser(this)));
227 }
228
229 // 3. The upper limit has been reached and there is no idle resource -> return a failure
230 // directly
231 return stdx::unexpected(fmt::format("No available resource, max_count={}, current_count={}",
232 m_maxCount, m_count.load(std::memory_order_relaxed)));
233 }

242 awaitable<stdx::expected<ResourcePtr, std::string>> asyncGet(
243 std::chrono::steady_clock::duration timeout = std::chrono::seconds(5)) {
244 // Try to acquire a resource from the idle queue
245 ResourceType *p = takeIdleResource();
246 if (p) {
247 co_return stdx::expected<ResourcePtr, std::string>(
248 ResourcePtr(p, ResourceCloser(this)));
249 }
250
251 // There is no idle resource but the upper limit has not been reached -> create a new
252 // resource. The slot is reserved with a CAS first, so the upper limit stays a hard one even
253 // when several coroutines create resources at the same time
254 if (reserveResourceSlot()) {
255 try {
256 p = new ResourceType(m_param);
257 } catch (const std::exception &e) {
258 releaseResourceCount();
259 co_return stdx::unexpected(
260 std::string(fmt::format("Failed create a new Resource! {}", e.what())));
261 } catch (...) {
262 releaseResourceCount();
263 co_return stdx::unexpected(
264 std::string("Failed create a new Resource! Unknown error!"));
265 }
266 co_return stdx::expected<ResourcePtr, std::string>(
267 ResourcePtr(p, ResourceCloser(this)));
268 }
269
270 // The upper limit has been reached -> enter the waiting queue
271 auto executor = co_await this_coro::executor;
272 auto waiter = std::make_shared<WaiterNode>();
273 waiter->timer = std::make_shared<net::steady_timer>(executor);
274 waiter->timer->expires_after(timeout);
275
276 // Both messages are built before the wait is started: nothing may be read from the pool on
277 // the failure branches, because the pool can already be gone when this coroutine resumes,
278 // and nothing between the wait being started and its co_await may throw (an awaitable with
279 // a pending operation must never be destroyed without being awaited)
280 auto timeout_msg =
281 fmt::format("ResourceAsioPool get timeout, max_count={}, current_count={}", m_maxCount,
282 m_count.load(std::memory_order_relaxed));
283 auto queue_full_msg = fmt::format("Waiter queue is full (max={})", m_maxWaiters);
284
285 // Start the wait before joining the queue, so that every wake up done by a resource
286 // returning thread hits an outstanding operation
287 net::error_code ec;
288 auto wait_result = waiter->timer->async_wait(net::redirect_error(net::use_awaitable, ec));
289
290 // Keep our own reference to the waiting queue, so that unregistering stays valid even when
291 // the pool is destroyed while we are suspended
292 auto waiters = m_waiters;
293
294 if (!pushWaiter(waiters, waiter)) {
295 waiter->timer->expires_after(std::chrono::milliseconds(0));
296 co_await std::move(wait_result);
297 co_return stdx::unexpected(queue_full_msg);
298 }
299
300 // A resource may have been handed over between the checks above and joining the queue, take
301 // it instead of sleeping for a whole timeout
302 ResourceType *idle = takeIdleResource();
303 if (idle) {
304 waiter->timer->expires_after(std::chrono::milliseconds(0));
305 co_await std::move(wait_result);
306 ResourceType *reserved =
307 waiter->handed.exchange(WaiterNode::gaveUp(), std::memory_order_acq_rel);
308 removeWaiter(waiters, waiter);
309 if (reserved != nullptr && reserved != WaiterNode::gaveUp()) {
310 // Somebody handed a second resource over to us, give ours back
311 returnResource(idle, nullptr);
312 idle = reserved;
313 }
314 co_return stdx::expected<ResourcePtr, std::string>(
315 ResourcePtr(idle, ResourceCloser(this)));
316 }
317
318 // Note the error code is not looked at: it only tells how we were woken up, the result is
319 // decided by who wins the atomic exchange below
320 co_await std::move(wait_result);
321
322 ResourceType *reserved =
323 waiter->handed.exchange(WaiterNode::gaveUp(), std::memory_order_acq_rel);
324 removeWaiter(waiters, waiter);
325
326 if (reserved != nullptr && reserved != WaiterNode::gaveUp()) {
327 co_return stdx::expected<ResourcePtr, std::string>(
328 ResourcePtr(reserved, ResourceCloser(this)));
329 }
330
331 co_return stdx::unexpected(timeout_msg);
332 }

349 void releaseIdleResource() {
350 ResourceType *p = nullptr;
351 while (m_resourceList.pop(p)) {
352 if (p) {
353 m_idleCount.fetch_sub(1); // Decrease the idle count
354 delete p;
355 releaseResourceCount(); // Decrease the count and notify the destructor
356 }
357 }

360private:
362 ResourceType *takeIdleResource() {
363 ResourceType *p = nullptr;
364 if (m_resourceList.pop(p)) {
365 m_idleCount.fetch_sub(1, std::memory_order_relaxed);
366 return p;
367 }
368 return nullptr;
369 }

373 void releaseResourceCount() {
374 m_count.fetch_sub(1, std::memory_order_seq_cst);
375 notifyCountChanged();
376 }

384 void notifyCountChanged() {
385 if (m_destroy_waiting.load(std::memory_order_seq_cst)) {
386 std::unique_lock<MutexType> lock(m_destroy_mutex);
387 m_destroy_cv.notify_one();
388 }

392 bool pushWaiter(const std::shared_ptr<WaiterRegistry> &waiters,
393 const std::shared_ptr<WaiterNode> &waiter) {
394 std::lock_guard<std::mutex> lock(waiters->mutex);
395 if (waiters->queue.size() >= m_maxWaiters) {
396 return false;
397 }
398 waiters->queue.emplace_back(waiter);
399 m_registeredWaiters.fetch_add(1, std::memory_order_relaxed);
400 return true;
401 }
402
404 void removeWaiter(const std::shared_ptr<WaiterRegistry> &waiters,
405 const std::shared_ptr<WaiterNode> &waiter) {
406 std::lock_guard<std::mutex> lock(waiters->mutex);
407 for (auto iter = waiters->queue.begin(); iter != waiters->queue.end(); ++iter) {
408 if (iter->lock() == waiter) {
409 waiters->queue.erase(iter);
410 m_registeredWaiters.fetch_sub(1, std::memory_order_relaxed);
411 return;
412 }
413 }
414 }

420 bool handToWaiter(ResourceType *p) {
421 // The fast path of the resource returning: without any registered waiter there is nothing
422 // to scan and no lock to take. A waiter registered just after this check finds the resource
423 // by itself through the idle queue re-check below
424 if (m_registeredWaiters.load(std::memory_order_acquire) == 0) {
425 return false;
426 }
427
428 while (true) {
429 std::shared_ptr<WaiterNode> waiter;
430 {
431 std::lock_guard<std::mutex> lock(m_waiters->mutex);
432 for (auto iter = m_waiters->queue.begin(); iter != m_waiters->queue.end();) {
433 auto candidate = iter->lock();
434 if (!candidate) {
435 // The waiting coroutine is gone, clean its node up on the way
436 iter = m_waiters->queue.erase(iter);
437 m_registeredWaiters.fetch_sub(1, std::memory_order_relaxed);
438 continue;
439 }
440 ResourceType *current = candidate->handed.load(std::memory_order_relaxed);
441 if (current == WaiterNode::gaveUp()) {
442 // It has given up, clean its node up on the way
443 iter = m_waiters->queue.erase(iter);
444 m_registeredWaiters.fetch_sub(1, std::memory_order_relaxed);
445 continue;
446 }
447 if (current == nullptr) {
448 waiter = std::move(candidate);
449 break;
450 }
451 // A resource has been handed over to it but has not been taken yet
452 ++iter;
453 }
454 if (!waiter) {
455 return false;
456 }
457 }
458
459 ResourceType *expected = nullptr;
460 if (waiter->handed.compare_exchange_strong(expected, p, std::memory_order_acq_rel)) {
461 // Wake it up with cancel(): it never touches the expiry, which the waiting
462 // coroutine may rearm at the very same moment
463 waiter->timer->cancel();
464 return true;
465 }
466
467 // It has just given up, look for the next one
468 }
469 }

474 bool reserveResourceSlot() {
475 size_t current = m_count.load(std::memory_order_relaxed);
476 while (m_maxCount == 0 || current < m_maxCount) {
477 if (m_count.compare_exchange_weak(current, current + 1, std::memory_order_acq_rel)) {
478 return true;
479 }
480 }
481 return false;
482 }

484 class ResourceCloser {
485 public:
486 explicit ResourceCloser(ResourceAsioPool *pool) : m_pool(pool) {}
487
488 void operator()(ResourceType *conn) {
489 if (conn) {
490 // If the pool is bound, the resource is returned; otherwise it is deleted
491 if (m_pool) {
492 m_pool->returnResource(conn, this);
493 } else {
494 delete conn;
495 }
496 }
497 }
498
499 private:
500 ResourceAsioPool *m_pool;
501 };
502
504 void returnResource(ResourceType *p, ResourceCloser *closer) {
505 if (!p) [[unlikely]] {
506 HKU_WARN("ResourceAsioPool::returnResource: nullptr");
507 return;
508 }
509
510 // If it is being destructed, delete the resource directly
511 if (m_is_destroying.load(std::memory_order_acquire)) {
512 delete p;
513 releaseResourceCount();
514 return;
515 }
516
517 // Try to hand it over to a waiter that has not given up yet
518 if (handToWaiter(p)) {
519 return;
520 }
521
522 // There is no waiter to hand it over to, put it back into the idle queue
523 if (!m_resourceList.push(p)) {
524 // The queue is full, delete it directly
525 delete p;
526 releaseResourceCount();
527 return;
528 }
529
530 m_idleCount.fetch_add(1, std::memory_order_seq_cst);
531 notifyCountChanged();
532 }

534 std::atomic<size_t> m_count{0}; // The number of the currently active resources
535 std::atomic<size_t> m_idleCount{0}; // The current number of the idle resources
536 size_t m_maxCount; // The maximum resource upper limit
537 Parameter m_param;
538 boost::lockfree::queue<ResourceType *> m_resourceList;
539
540 size_t m_maxWaiters; // The runtime logical upper limit: the maximum number of the waiters
541 std::shared_ptr<WaiterRegistry> m_waiters; // The waiting queue, shared with the waiting
542 // coroutines
543 MutexType m_destroy_mutex; // The mutex protecting the destructor waiting
544 // condition variable
545 std::condition_variable_any m_destroy_cv; // Used to notify the destructor that a
546 // resource has been returned
547 std::atomic<bool> m_is_destroying{false}; // Marks whether it is being destructed
548 std::atomic<size_t> m_registeredWaiters{
549 0}; // The number of the registered waiters, it may be a little too large
550 std::atomic<bool> m_destroy_waiting{
551 false}; // Marks whether the destructor is waiting on the condition variable
552};

642 virtual ~ResourceAsioVersionPool() {
643 // Mark that it is being destructed, preventing new resource acquisitions and returns
644 m_is_destroying.store(true, std::memory_order_release);
645
646 // Take all the waiters away at once. Every node is co-owned by the waiting queue and by the
647 // waiting coroutine, so whoever claims the reserved resource first owns it, while the node
648 // itself stays alive as long as the other side still holds a reference.
649 std::vector<std::weak_ptr<WaiterNode>> waiters;
650 {
651 std::lock_guard<std::mutex> lock(m_waiters->mutex);
652 waiters.swap(m_waiters->queue);
653 }
654 m_registeredWaiters.fetch_sub(waiters.size(), std::memory_order_relaxed);
655
656 for (const auto &waiter_ref : waiters) {
657 auto waiter = waiter_ref.lock();
658 if (!waiter) {
659 // The waiting coroutine is gone, its node dies together with it
660 continue;
661 }
662 ResourceType *p =
663 waiter->handed.exchange(WaiterNode::gaveUp(), std::memory_order_acq_rel);
664 // Only the resource nobody will ever claim may be released here. The waiting coroutines
665 // are not timed out explicitly: whenever one of them resumes it finds gaveUp() and
666 // takes the time out branch, which reads nothing from the pool any more.
667 if (p != nullptr && p != WaiterNode::gaveUp()) {
668 delete p;
669 releaseResourceCount();
670 }
671 }
672
673 // Wait for all the active resources to be returned
674 std::unique_lock<MutexType> lock(m_destroy_mutex);
675 m_destroy_waiting.store(true, std::memory_order_seq_cst);
676 m_destroy_cv.wait(lock, [this]() {
677 return m_count.load(std::memory_order_seq_cst) ==
678 m_idleCount.load(std::memory_order_seq_cst);
679 });
680 m_destroy_waiting.store(false, std::memory_order_relaxed);
681
682 // At this time all the resources are in the idle queue, release them
683 ResourceType *p = nullptr;
684 while (m_resourceList.pop(p)) {
685 if (p) {
686 delete p;
687 }
688 }

772 // 1. Try to acquire a resource from the idle queue; a resource whose version is too old is
773 // destroyed by takeIdleResource() itself and nullptr is returned then

818 awaitable<stdx::expected<ResourcePtr, std::string>> asyncGet(
819 std::chrono::steady_clock::duration timeout = std::chrono::seconds(5)) {
820 auto executor = co_await this_coro::executor;
821
822 // Try to acquire a resource from the idle queue; a resource whose version is too old is
823 // destroyed by takeIdleResource() itself and nullptr is returned then, a new resource is
824 // created below
825 ResourceType *p = takeIdleResource();
826 if (p) {
827 co_return stdx::expected<ResourcePtr, std::string>(
828 ResourcePtr(p, ResourceCloser(this)));
829 }
830
831 // The upper limit has not been reached, create a new resource. The slot is reserved with a
832 // CAS first, so the upper limit stays a hard one even when several coroutines create
833 // resources at the same time
834 if (reserveResourceSlot()) {
835 try {
836 Parameter current_param;
837 int current_version;
838 {
839 std::lock_guard<MutexType> lock(m_mutex);
840 current_param = m_param;
841 current_version = m_version.load();
842 }
843
844 p = new ResourceType(current_param);
845 p->setVersion(current_version);
846 } catch (const std::exception &e) {
847 releaseResourceCount();
848 co_return stdx::unexpected(
849 std::string(fmt::format("Failed create a new Resource! {}", e.what())));
850 } catch (...) {
851 releaseResourceCount();
852 co_return stdx::unexpected(
853 std::string("Failed create a new Resource! Unknown error!"));
854 }
855
856 co_return stdx::expected<ResourcePtr, std::string>(
857 ResourcePtr(p, ResourceCloser(this)));
858 }
859
860 // The upper limit has been reached, enter the waiting queue
861 auto waiter = std::make_shared<WaiterNode>();
862 waiter->timer = std::make_shared<net::steady_timer>(executor);
863 waiter->timer->expires_after(timeout);
864
865 // Both messages are built before the wait is started: nothing may be read from the pool on
866 // the failure branches, because the pool can already be gone when this coroutine resumes,
867 // and nothing between the wait being started and its co_await may throw (an awaitable with
868 // a pending operation must never be destroyed without being awaited)
869 auto timeout_msg =
870 fmt::format("ResourceAsioVersionPool get timeout, max_count={}, current_count={}",
871 m_maxCount, m_count.load(std::memory_order_relaxed));
872 auto queue_full_msg = fmt::format("Waiter queue is full (max={})", m_maxWaiters);
873
874 // Start the wait before joining the queue, so that every wake up done by a resource
875 // returning thread hits an outstanding operation
876 net::error_code ec;
877 auto wait_result = waiter->timer->async_wait(net::redirect_error(net::use_awaitable, ec));
878
879 // Keep our own reference to the waiting queue, so that unregistering stays valid even when
880 // the pool is destroyed while we are suspended
881 auto waiters = m_waiters;
882
883 if (!pushWaiter(waiters, waiter)) {
884 waiter->timer->expires_after(std::chrono::milliseconds(0));
885 co_await std::move(wait_result);
886 co_return stdx::unexpected(queue_full_msg);
887 }
888
889 // A resource may have been handed over between the checks above and joining the queue, take
890 // it instead of sleeping for a whole timeout
891 ResourceType *idle = takeIdleResource();
892 if (idle) {
893 waiter->timer->expires_after(std::chrono::milliseconds(0));
894 co_await std::move(wait_result);
895 ResourceType *reserved =
896 waiter->handed.exchange(WaiterNode::gaveUp(), std::memory_order_acq_rel);
897 removeWaiter(waiters, waiter);
898 if (reserved != nullptr && reserved != WaiterNode::gaveUp()) {
899 // Somebody handed a second resource over to us, give ours back
900 returnResource(idle, nullptr);
901 idle = reserved;
902 }
903 co_return stdx::expected<ResourcePtr, std::string>(
904 ResourcePtr(idle, ResourceCloser(this)));
905 }
906
907 // Note the error code is not looked at: it only tells how we were woken up, the result is
908 // decided by who wins the atomic exchange below
909 co_await std::move(wait_result);
910
911 ResourceType *reserved =
912 waiter->handed.exchange(WaiterNode::gaveUp(), std::memory_order_acq_rel);
913 removeWaiter(waiters, waiter);
914
915 if (reserved != nullptr && reserved != WaiterNode::gaveUp()) {
916 co_return stdx::expected<ResourcePtr, std::string>(
917 ResourcePtr(reserved, ResourceCloser(this)));
918 }
919
920 co_return stdx::unexpected(timeout_msg);
921 }

987 bool pushWaiter(const std::shared_ptr<WaiterRegistry> &waiters,
988 const std::shared_ptr<WaiterNode> &waiter) {
989 std::lock_guard<std::mutex> lock(waiters->mutex);
990 if (waiters->queue.size() >= m_maxWaiters) {
991 return false;
992 }
993 waiters->queue.emplace_back(waiter);
994 m_registeredWaiters.fetch_add(1, std::memory_order_relaxed);
995 return true;
996 }
997
999 void removeWaiter(const std::shared_ptr<WaiterRegistry> &waiters,
1000 const std::shared_ptr<WaiterNode> &waiter) {
1001 std::lock_guard<std::mutex> lock(waiters->mutex);
1002 for (auto iter = waiters->queue.begin(); iter != waiters->queue.end(); ++iter) {
1003 if (iter->lock() == waiter) {
1004 waiters->queue.erase(iter);
1005 m_registeredWaiters.fetch_sub(1, std::memory_order_relaxed);
1006 return;
1007 }
1008 }

1015 bool handToWaiter(ResourceType *p) {
1016 // The fast path of the resource returning: without any registered waiter there is nothing
1017 // to scan and no lock to take. A waiter registered just after this check finds the resource
1018 // by itself through the idle queue re-check below
1019 if (m_registeredWaiters.load(std::memory_order_acquire) == 0) {
1020 return false;
1021 }
1022
1023 while (true) {
1024 std::shared_ptr<WaiterNode> waiter;
1025 {
1026 std::lock_guard<std::mutex> lock(m_waiters->mutex);
1027 for (auto iter = m_waiters->queue.begin(); iter != m_waiters->queue.end();) {
1028 auto candidate = iter->lock();
1029 if (!candidate) {
1030 // The waiting coroutine is gone, clean its node up on the way
1031 iter = m_waiters->queue.erase(iter);
1032 m_registeredWaiters.fetch_sub(1, std::memory_order_relaxed);
1033 continue;
1034 }
1035 ResourceType *current = candidate->handed.load(std::memory_order_relaxed);
1036 if (current == WaiterNode::gaveUp()) {
1037 // It has given up, clean its node up on the way
1038 iter = m_waiters->queue.erase(iter);
1039 m_registeredWaiters.fetch_sub(1, std::memory_order_relaxed);
1040 continue;
1041 }
1042 if (current == nullptr) {
1043 waiter = std::move(candidate);
1044 break;
1045 }
1046 // A resource has been handed over to it but has not been taken yet
1047 ++iter;
1048 }
1049 if (!waiter) {
1050 return false;
1051 }
1052 }
1053
1054 ResourceType *expected = nullptr;
1055 if (waiter->handed.compare_exchange_strong(expected, p, std::memory_order_acq_rel)) {
1056 // Wake it up with cancel(): it never touches the expiry, which the waiting
1057 // coroutine may rearm at the very same moment
1058 waiter->timer->cancel();
1059 return true;
1060 }
1061
1062 // It has just given up, look for the next one
1063 }
1064 }

1099 void returnResource(ResourceType *p, ResourceCloser *closer) {
1100 if (!p) [[unlikely]] {
1101 HKU_WARN("ResourceAsioVersionPool::returnResource: nullptr");
1102 return;
1103 }
1104
1105 // If it is being destructed, delete the resource directly
1106 if (m_is_destroying.load(std::memory_order_acquire)) {
1107 delete p;
1108 releaseResourceCount();
1109 return;
1110 }
1111
1112 // The returned resource is accepted only when its version equals the resource pool version
1113 if (p->getVersion() != m_version.load(std::memory_order_acquire)) {
1114 delete p;
1115 releaseResourceCount();
1116 return;
1117 }
1118
1119 // Try to hand it over to a waiter that has not given up yet
1120 if (handToWaiter(p)) {
1121 return;
1122 }
1123
1124 // There is no waiter to hand it over to, put it back into the idle queue
1125 if (!m_resourceList.push(p)) {
1126 // The queue is full, delete it directly
1127 delete p;
1128 releaseResourceCount();
1129 return;
1130 }
1131
1132 m_idleCount.fetch_add(1, std::memory_order_seq_cst);
1133 notifyCountChanged();
1134 }
