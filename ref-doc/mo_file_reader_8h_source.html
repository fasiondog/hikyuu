339 eErrorCode ReadFile(const char *filename) {
340 // Opening the file.
341 std::ifstream stream(filename, std::ios_base::binary | std::ios_base::in);
342 if (!stream.is_open()) {
343 m_error = std::string("Cannot open File ") + std::string(filename);
344 return moFileReader::EC_FILENOTFOUND;
345 }
346
347 eErrorCode res = ReadStream(stream);
348 stream.close();
349
350 return res;
351 }

358 template <typename T>
359 eErrorCode ReadStream(T &stream) {
360 try {
361 // The actual data size bounds every offset and length the file claims
362 stream.seekg(0, std::ios::end);
363 std::streamoff endPos = stream.tellg();
364 stream.seekg(0, std::ios::beg);
365 if (!stream || endPos < 0) {
366 m_error = "Cannot determine the size of the .mo-file!";
367 return moFileReader::EC_FILEINVALID;
368 }
369 const int64_t dataSize = static_cast<int64_t>(endPos);
370
371 // Creating a file-description.
372 moFileInfo moInfo;
373
374 // Reference to the List inside moInfo.
375 moFileInfo::moTranslationPairList &TransPairInfo = moInfo.m_translationPairInformation;
376
377 // Read in all the 4 bytes of fire-magic, offsets and stuff...
378 stream.read((char *)&moInfo.m_magicNumber, 4);
379 stream.read((char *)&moInfo.m_fileVersion, 4);
380 stream.read((char *)&moInfo.m_numStrings, 4);
381 stream.read((char *)&moInfo.m_offsetOriginal, 4);
382 stream.read((char *)&moInfo.m_offsetTranslation, 4);
383 stream.read((char *)&moInfo.m_sizeHashtable, 4);
384 stream.read((char *)&moInfo.m_offsetHashtable, 4);
385
386 // A truncated header read sets failbit, not badbit
387 if (stream.fail()) {
388 m_error = "The .mo-file header is truncated!";
389 return moFileReader::EC_FILEINVALID;
390 }
391
392 // Checking the Magic Number
393 if (MagicNumber != moInfo.m_magicNumber) {
394 if (MagicReversed != moInfo.m_magicNumber) {
395 m_error = "The Magic Number does not match in all cases!";
396 return moFileReader::EC_MAGICNUMBER_NOMATCH;
397 } else {
398 moInfo.m_reversed = true;
399 m_error = "Magic Number is reversed. We do not support this yet!";
400 return moFileReader::EC_MAGICNUMBER_REVERSED;
401 }
402 }
403
404 // Validate counts and table offsets before any loop or allocation
405 if (moInfo.m_numStrings < 0 || moInfo.m_offsetOriginal < 0 ||
406 moInfo.m_offsetTranslation < 0) {
407 m_error = "Negative string count or table offset in the .mo-file!";
408 return moFileReader::EC_FILEINVALID;
409 }
410 const int64_t num = moInfo.m_numStrings;
411 const int64_t tableSize = num * 8; // one descriptor is 8 bytes
412 if (tableSize > dataSize ||
413 static_cast<int64_t>(moInfo.m_offsetOriginal) + tableSize > dataSize ||
414 static_cast<int64_t>(moInfo.m_offsetTranslation) + tableSize > dataSize) {
415 m_error = "The string descriptor table extends beyond the .mo-file!";
416 return moFileReader::EC_FILEINVALID;
417 }
418
419 auto readFully = [&stream](void *dst, std::streamsize count) -> bool {
420 stream.read(reinterpret_cast<char *>(dst), count);
421 return !stream.fail();
422 };
423
424 // Now we search all Length & Offsets of the original strings
425 stream.seekg(moInfo.m_offsetOriginal);
426 for (int64_t i = 0; i < num; i++) {
427 moTranslationPairInformation _str;
428 if (!readFully(&_str.m_orLength, 4) || !readFully(&_str.m_orOffset, 4)) {
429 m_error = "The original string descriptor table is truncated!";
430 return moFileReader::EC_FILEINVALID;
431 }
432 TransPairInfo.push_back(_str);
433 }
434
435 // Get all Lengths & Offsets of the translated strings
436 stream.seekg(moInfo.m_offsetTranslation);
437 for (int64_t i = 0; i < num; i++) {
438 moTranslationPairInformation &_str = TransPairInfo[i];
439 if (!readFully(&_str.m_trLength, 4) || !readFully(&_str.m_trOffset, 4)) {
440 m_error = "The translated string descriptor table is truncated!";
441 return moFileReader::EC_FILEINVALID;
442 }
443 }
444
445 // Normally you would read the hash-table here, but we don't use it. :)
446
447 // Now to the interesting part, we read the strings-pairs now
448 for (int64_t i = 0; i < num; i++) {
449 // +1 to catch the trailing \0
450 int64_t orLength = static_cast<int64_t>(TransPairInfo[i].m_orLength) + 1;
451 int64_t trLength = static_cast<int64_t>(TransPairInfo[i].m_trLength) + 1;
452 int64_t orOffset = TransPairInfo[i].m_orOffset;
453 int64_t trOffset = TransPairInfo[i].m_trOffset;
454
455 // Lengths and offsets must stay inside the data
456 if (orLength <= 0 || trLength <= 0 || orOffset < 0 || trOffset < 0 ||
457 orOffset + orLength > dataSize || trOffset + trLength > dataSize) {
458 m_error = "A string descriptor points beyond the .mo-file!";
459 return moFileReader::EC_FILEINVALID;
460 }
461
462 // Original (unique_ptr also removes the old leak on error returns)
463 std::unique_ptr<char[]> original(new char[static_cast<size_t>(orLength)]);
464 memset(original.get(), 0, sizeof(char) * static_cast<size_t>(orLength));
465 stream.seekg(orOffset);
466 if (!readFully(original.get(), orLength)) {
467 m_error = "Truncated original string in the .mo-file!";
468 return moFileReader::EC_FILEINVALID;
469 }
470
471 // Translation
472 std::unique_ptr<char[]> translation(new char[static_cast<size_t>(trLength)]);
473 memset(translation.get(), 0, sizeof(char) * static_cast<size_t>(trLength));
474 stream.seekg(trOffset);
475 if (!readFully(translation.get(), trLength)) {
476 m_error = "Truncated translated string in the .mo-file!";
477 return moFileReader::EC_FILEINVALID;
478 }
479
480 std::string original_str = original.get();
481 std::string translation_str = translation.get();
482 auto ctxSeparator = original_str.find(ContextSeparator);
483
484 // Store it in the map.
485 if (ctxSeparator == std::string::npos) {
486 m_lookup[original_str] = translation_str;
487 numStrings++;
488 } else {
489 // try-catch for handling out_of_range exceptions
490 try {
491 m_lookup_context[original_str.substr(0, ctxSeparator)][original_str.substr(
492 ctxSeparator + 1, original_str.length())] = translation_str;
493 numStrings++;
494 } catch (...) {
495 m_error =
496 "Stream bad during reading. The .mo-file seems to be invalid or has bad "
497 "descriptions!";
498 return moFileReader::EC_ERROR;
499 }
500 }
501 }
502
503 // Done :)
504 return moFileReader::EC_SUCCESS;
505 } catch (const std::bad_alloc &) {
506 // Keep the error-code contract instead of letting bad_alloc escape
507 m_error = "Out of memory while reading the .mo-file!";
508 return moFileReader::EC_FILEINVALID;
509 }
510 }

516 std::string Lookup(const char *id) const {
517 if (m_lookup.empty())
518 return id;
519 auto iterator = m_lookup.find(id);
520
521 return iterator == m_lookup.end() ? id : iterator->second;
522 }
523
529 std::string LookupWithContext(const char *context, const char *id) const {
530 if (m_lookup_context.empty())
531 return id;
532 auto iterator = m_lookup_context.find(context);
533
534 if (iterator == m_lookup_context.end())
535 return id;
536 auto iterator2 = iterator->second.find(id);
537
538 return iterator2 == iterator->second.end() ? id : iterator2->second;
539 }

569 unsigned long SwapBytes(unsigned long in) {
570 unsigned long b0 = (in >> 0) & 0xff;
571 unsigned long b1 = (in >> 8) & 0xff;
572 unsigned long b2 = (in >> 16) & 0xff;
573 unsigned long b3 = (in >> 24) & 0xff;
574
575 return (b0 << 24) | (b1 << 16) | (b2 << 8) | b3;
576 }
