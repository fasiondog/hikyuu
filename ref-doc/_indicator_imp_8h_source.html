40template <typename ValueType>
41inline ValueType parseLegacyValueText(const std::string& vstr) {
42 std::string s;
43 s.reserve(vstr.size());
44 for (char c : vstr) {
45 if (c == ' ' || c == '\t' || c == '(' || c == ')') {
46 continue;
47 }
48 s.push_back(static_cast<char>(std::tolower(static_cast<unsigned char>(c))));
49 }
50 if (s.empty()) {
51 return Null<ValueType>();
52 }
53 if (s.find("nan") != std::string::npos) {
54 return Null<ValueType>();
55 }
56 if (s.find("inf") != std::string::npos) {
57 return (s[0] == '-') ? -std::numeric_limits<ValueType>::infinity()
58 : std::numeric_limits<ValueType>::infinity();
59 }
60 try {
61 return boost::lexical_cast<ValueType>(vstr);
62 } catch (const std::exception&) {
63 // Unrecognizable legacy special value text, treated as invalid
64 return Null<ValueType>();
65 }
66}

513 void save(Archive& ar, const unsigned int version) const {
514 namespace bs = boost::serialization;
515 ar& BOOST_SERIALIZATION_NVP(m_name);

520 ar& BOOST_SERIALIZATION_NVP(m_old_context);
521 ar& BOOST_SERIALIZATION_NVP(m_need_context);
522 ar& BOOST_SERIALIZATION_NVP(m_is_python_object);
523 ar& BOOST_SERIALIZATION_NVP(m_need_self_alike_compare);
524 ar& BOOST_SERIALIZATION_NVP(m_is_serial);
525 ar& BOOST_SERIALIZATION_NVP(m_need_calculate);
526 ar& BOOST_SERIALIZATION_NVP(m_param_changed);
527 ar& BOOST_SERIALIZATION_NVP(m_optype);
528 ar& BOOST_SERIALIZATION_NVP(m_left);
529 ar& BOOST_SERIALIZATION_NVP(m_right);
530 ar& BOOST_SERIALIZATION_NVP(m_three);
531 ar& BOOST_SERIALIZATION_NVP(m_ind_params);
532
533 size_t act_result_num = 0;
534 for (size_t i = 0; i < m_result_num; i++) {
535 if (m_pBuffer[i]) {
536 act_result_num++;
537 }
538 }
539 ar& BOOST_SERIALIZATION_NVP(act_result_num);
540 if (act_result_num > 0) {
541 ar& boost::serialization::make_nvp("count", BIT_PATTERN_TAG);
542 // One raw bit pattern block per result: compact in a binary archive and lossless in
543 // every archive type
544 for (size_t i = 0; i < act_result_num; ++i) {
545 const buffer_t& values = *m_pBuffer[i];
546 size_t count = values.size();
547 ar& BOOST_SERIALIZATION_NVP(count);
548 std::vector<bit_t> bits(count);
549 std::memcpy(bits.data(), values.data(), count * sizeof(value_t));
550 ar& boost::serialization::make_nvp(
551 "values", boost::serialization::make_array(bits.data(), count));
552 }
553 }

556 template <class Archive>
557 void load(Archive& ar, const unsigned int version) {
558 namespace bs = boost::serialization;
559 ar& BOOST_SERIALIZATION_NVP(m_name);
560 ar& BOOST_SERIALIZATION_NVP(m_params);
561 ar& BOOST_SERIALIZATION_NVP(m_discard);
562 ar& BOOST_SERIALIZATION_NVP(m_result_num);
563 ar& BOOST_SERIALIZATION_NVP(m_context);
564 ar& BOOST_SERIALIZATION_NVP(m_old_context);
565 ar& BOOST_SERIALIZATION_NVP(m_need_context);
566 ar& BOOST_SERIALIZATION_NVP(m_is_python_object);
567 ar& BOOST_SERIALIZATION_NVP(m_need_self_alike_compare);
568 ar& BOOST_SERIALIZATION_NVP(m_is_serial);
569 ar& BOOST_SERIALIZATION_NVP(m_need_calculate);
570 ar& BOOST_SERIALIZATION_NVP(m_param_changed);
571 ar& BOOST_SERIALIZATION_NVP(m_optype);
572 ar& BOOST_SERIALIZATION_NVP(m_left);
573 ar& BOOST_SERIALIZATION_NVP(m_right);
574 ar& BOOST_SERIALIZATION_NVP(m_three);
575 ar& BOOST_SERIALIZATION_NVP(m_ind_params);
576
577 size_t act_result_num = 0;
578 ar& BOOST_SERIALIZATION_NVP(act_result_num);
579 _readyBuffer(0, act_result_num);
580 if (act_result_num == 0) {
581 return;
582 }
583
584 uint64_t tag = 0;
585 ar& boost::serialization::make_nvp("count", tag);
586 // The tag value discriminates the format (the archive version may be stale across DLLs):
587 // BIT_PATTERN_TAG for the bit pattern arrays, otherwise the legacy per-value texts
588 if (tag != BIT_PATTERN_TAG) {
589 // The legacy format stored every value as a string of 6 decimal places, reading it
590 // back loses precision; kept only to open old archives. Its first field is the value
591 // count of the first result, which the read above already consumed
592 _loadLegacyValues(ar, 0, static_cast<size_t>(tag));
593 for (size_t i = 1; i < act_result_num; ++i) {
594 size_t count = 0;
595 ar& BOOST_SERIALIZATION_NVP(count);
596 _loadLegacyValues(ar, i, count);
597 }
598 return;
599 }
600
601 for (size_t i = 0; i < act_result_num; ++i) {
602 size_t count = 0;
603 ar& BOOST_SERIALIZATION_NVP(count);
604 buffer_t& values = *m_pBuffer[i];
605 values.resize(count);
606 std::vector<bit_t> bits(count);
607 ar& boost::serialization::make_nvp(
608 "values", boost::serialization::make_array(bits.data(), count));
609 std::memcpy(values.data(), bits.data(), count * sizeof(value_t));
610 }
611 }

614 template <class Archive>
615 void _loadLegacyValues(Archive& ar, size_t result_idx, size_t count) {
616 buffer_t& values = *m_pBuffer[result_idx];
617 values.resize(count);
618 for (size_t j = 0; j < count; j++) {
619 std::string vstr;
620 ar >> boost::serialization::make_nvp<string>("item", vstr);
621 values[j] = detail::parseLegacyValueText<value_t>(vstr);
622 }
623 }

739inline bool IndicatorImp::equalValue(value_t v1, value_t v2) noexcept {
740 if (v1 == v2)
741 return true;
742 bool nan1 = std::isnan(v1);
743 if (nan1 || std::isnan(v2))
744 return nan1 == std::isnan(v2);
745 if (std::isinf(v1) || std::isinf(v2))
746 return false;
747 double a1 = std::abs(v1);
748 double a2 = std::abs(v2);
749 double scale = a1 > a2 ? a1 : a2;
750 return std::abs(v1 - v2) <= IND_EQ_THRESHOLD * (scale > 1.0 ? scale : 1.0);
751}
