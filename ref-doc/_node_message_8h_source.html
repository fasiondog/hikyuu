44inline json decodeMsg(nng_msg *msg) {
45 HKU_ASSERT(msg != nullptr);
46 size_t len = nng_msg_len(msg);
47 uint8_t *data = (uint8_t *)nng_msg_body(msg);
48 json result = json::from_msgpack(data, data + len);
49 return result;
50}

60inline void encodeMsg(nng_msg *msg, const json &in) {
61 HKU_ASSERT(msg != nullptr);
62 nng_msg_clear(msg);
63
64 std::vector<std::uint8_t> v = json::to_msgpack(in);
65 int rv = nng_msg_append(msg, v.data(), v.size());
66 NODE_NNG_CHECK(rv, "Failed nng_msg_append!");
67}
