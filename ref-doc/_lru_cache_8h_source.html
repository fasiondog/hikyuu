39template <typename KeyType, typename ValueType, class Lock = NullLock>
40class LruCache final {
41public:
42 using key_type = KeyType;
43 using value_type = ValueType;
44 using size_type = size_t;
45 typedef Lock lock_type;
46 using UniqueGuard = std::unique_lock<lock_type>;
47 using SharedGuard = std::shared_lock<lock_type>;
48
49 // Storage structure: the value + an atomic dirty flag (marking whether it has been accessed by
50 // get and the LRU order needs to be updated)
51 using CacheValue = std::pair<value_type, std::atomic<bool>>;
52 using LruList = std::list<key_type>;
53 using CacheMap =
54 std::unordered_map<key_type, std::pair<typename LruList::iterator, CacheValue>>;
55
63 explicit LruCache(size_type capacity = 64, size_type overflow = 8)
64 : m_capacity(capacity), m_overflow(overflow) {}
65
66 ~LruCache() {
67 UniqueGuard lock(m_mutex);
68 m_cache.clear();
69 m_lru_list.clear();
70 }

77 void insert(const key_type& key, const value_type& value) {
78 UniqueGuard lock(m_mutex);
79 _batch_update_dirty_nodes();
80 auto it = m_cache.find(key);
81 if (it != m_cache.end()) {
82 it->second.second.first = value;
83 it->second.second.second.store(false, std::memory_order_relaxed);
84 m_lru_list.splice(m_lru_list.begin(), m_lru_list, it->second.first);
85 } else {
86 m_lru_list.emplace_front(key);
87 m_cache.emplace(key, std::make_pair(m_lru_list.begin(),
88 std::make_pair(value, false) // The initial dirty
89 // flag is false
90 ));
91 _prune_if_needed();
92 }
93 }

100 void insert(const key_type& key, value_type&& value) {
101 UniqueGuard lock(m_mutex);
102 _batch_update_dirty_nodes();
103 auto it = m_cache.find(key);
104 if (it != m_cache.end()) {
105 it->second.second.first = std::move(value);
106 it->second.second.second.store(false, std::memory_order_relaxed);
107 m_lru_list.splice(m_lru_list.begin(), m_lru_list, it->second.first);
108 } else {
109 m_lru_list.emplace_front(key);
110 m_cache.emplace(

113 }
114 }

122 value_type get(const key_type& key) {
123 SharedGuard lock(m_mutex);
124 auto it = m_cache.find(key);

240private:
241 // If the cache is full, the least recently used item is removed
242 size_t _prune_if_needed() {
243 size_t maxAllowed = m_capacity + m_overflow;
244 if (m_capacity == 0 || m_cache.size() <= maxAllowed) {
245 return 0;
246 }
247 size_t count = 0;
248 while (m_cache.size() > m_capacity) {
249 m_cache.erase(m_lru_list.back());
250 m_lru_list.pop_back();
251 ++count;
252 }
253 return count;
254 }
255
256 // Update all the dirty nodes in a batch: move them to the head of the list and clear the dirty
257 // flags
258 void _batch_update_dirty_nodes() {
259 // Fix 1: avoid the default construction and use a pointer/reference to track the latest
260 // dirty node
261 typename LruList::reverse_iterator latest_dirty_it;
262 bool has_dirty = false;
263
264 // Traverse the list backward: from the tail to the head, find the first dirty node (the
265 // most recently accessed node)
266 for (auto it = m_lru_list.rbegin(); it != m_lru_list.rend(); ++it) {
267 const key_type& key = *it;
268 auto cache_it = m_cache.find(key);
269 if (cache_it == m_cache.end()) {
270 continue;
271 }
272
273 auto& dirty_flag = cache_it->second.second.second;
274 // Fix 2: the atomic load judgment avoids an uninitialized access
275 if (dirty_flag.load(std::memory_order_relaxed)) {
276 latest_dirty_it = it;
277 has_dirty = true;
278 break; // Only the most recently accessed dirty node is handled, one is kept at the
279 // tail
280 }
281 }
282
283 // Only the most recently accessed dirty node is moved to the head (fix 3: the reverse
284 // iterator is converted to a forward iterator)
285 if (has_dirty) {
286 // C++17: convert the reverse iterator to a forward iterator (the base() method)
287 auto forward_it = latest_dirty_it.base();
288 --forward_it; // base() of the reverse iterator returns the next forward iterator, it
289 // needs to be decreased by 1
290
291 const key_type& key = *forward_it;
292 auto cache_it = m_cache.find(key);
293 if (cache_it != m_cache.end()) {
294 // Clear the dirty flag
295 cache_it->second.second.second.store(false, std::memory_order_relaxed);
296 // Move the node to the head of the list (splice supports the forward iterators
297 // only)
298 m_lru_list.splice(m_lru_list.begin(), m_lru_list, forward_it);
299 }
300 }
301 }
302
303private:
304 size_type m_capacity;
305 size_type m_overflow;
306 LruList m_lru_list;
307 CacheMap m_cache;
308 mutable lock_type m_mutex;
309};
