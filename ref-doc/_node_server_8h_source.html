58 void start(size_t max_parrel = 128) {
59 CLS_CHECK(!m_addr.empty(), "You must set NodeServer's addr first!");
60
61 // Start the node server
62 int rv = nng_rep0_open(&m_socket);
63 CLS_CHECK(0 == rv, "Failed open server socket! {}", nng_strerror(rv));
64 rv = nng_listen(m_socket, m_addr.c_str(), &m_listener, 0);
65 CLS_CHECK(0 == rv, "Failed listen node server socket ({})! {}", m_addr, nng_strerror(rv));
66 CLS_TRACE("channel lisenter server: {}", m_addr);
67
68 m_works.resize(max_parrel);
69 for (size_t i = 0, total = m_works.size(); i < total; i++) {
70 Work* w = &m_works[i];
71 rv = nng_aio_alloc(&w->aio, _serverCallback, w);
72 CLS_CHECK(0 == rv, "Failed create work {}! {}", i, nng_strerror(rv));
73 rv = nng_ctx_open(&w->ctx, m_socket);
74 CLS_CHECK(0 == rv, "Failed open ctx {}! {}", i, nng_strerror(rv));
75 w->state = Work::INIT;
76 w->server = this;
77 }
78
79 for (size_t i = 0, total = m_works.size(); i < total; i++) {
80 _serverCallback(&m_works[i]);
81 }
82 }

84 void loop() {
85 while (true) {
86 std::this_thread::sleep_for(std::chrono::seconds(1));

90 void stop() {
91 HKU_IF_RETURN(m_works.empty(), void());
92 for (size_t i = 0, total = m_works.size(); i < total; i++) {
93 Work* w = &m_works[i];
94 w->server = nullptr;
95 w->state = Work::FINISH;
96 if (w->aio) {
97 nng_aio_stop(w->aio);
98 nng_aio_free(w->aio);
99 nng_ctx_close(w->ctx);
100 w->aio = nullptr;
101 }
102 }
103
104 // Close the socket service node
105 nng_listener_close(m_listener);
106 nng_close(m_socket);
107 m_works.clear();
108 CLS_INFO("stopped node server.");
109 }

111private:
112 struct Work {
113 enum { INIT, RECV, SEND, FINISH } state = INIT;
114 nng_aio* aio{nullptr};
115 nng_ctx ctx;
116 NodeServer* server{nullptr};
117 };
118
119 static void _serverCallback(void* arg) {
120 Work* work = static_cast<Work*>(arg);
121 int rv = 0;
122 switch (work->state) {
123 case Work::INIT:
124 work->state = Work::RECV;
125 nng_ctx_recv(work->ctx, work->aio);
126 break;
127
128 case Work::RECV:
129 _processRequest(work);
130 break;
131
132 case Work::SEND:
133 if ((rv = nng_aio_result(work->aio)) != 0) {
134 CLS_FATAL("Failed nng_ctx_send! {}", nng_strerror(rv));
135 work->state = Work::FINISH;
136 return;
137 }
138 work->state = Work::RECV;
139 nng_ctx_recv(work->ctx, work->aio);
140 break;
141
142 case Work::FINISH:
143 break;
144
145 default:
146 CLS_FATAL("nng bad state!");
147 break;
148 }
149 }
150
151 static void _processRequest(Work* work) {
152 NodeServer* server = work->server;
153 CLS_IF_RETURN(!server || !work->aio, void());
154 nng_msg* msg = nullptr;
155 json res;
156
157 // A failed receive carries no message: an error reply built below would call encodeMsg
158 // with a null msg, whose HKU_ASSERT rethrows from inside the catch handler and escapes
159 // into the nng C callback (std::terminate). Re-arm the receive on the context and keep this
160 // work slot alive instead, so only a fully received request reaches the try block
161 int rv = nng_aio_result(work->aio);
162 if (rv != 0) {
163 CLS_WARN("Failed to receive a request! {}", nng_strerror(rv));
164 work->state = Work::RECV;
165 nng_ctx_recv(work->ctx, work->aio);
166 return;
167 }
168
169 msg = nng_aio_get_msg(work->aio);
170 try {
171 json req = decodeMsg(msg);
172 NODE_CHECK(req.contains("cmd"), NodeErrorCode::MISSING_CMD, "Missing command!");
173
174 // Compatible with the old version numeric cmd
175 std::string cmd = req["cmd"].is_number() ? fmt::format("{}", req["cmd"].get<int>())
176 : req["cmd"].get<std::string>();
177 auto iter = server->m_handles.find(cmd);
178 NODE_CHECK(iter != server->m_handles.end(), NodeErrorCode::INVALID_CMD,
179 "The server does not know how to process the message: {}", cmd);
180
181 // For a tcp connection it tries to get the client address and port into req
182 req["remote_host"] = "";
183 req["remote_port"] = 0;
184 nng_pipe p = nng_msg_get_pipe(msg);
185 if (nng_pipe_id(p) > 0) {
186 uint16_t port = 0;
187 nng_sockaddr ra;
188 rv = nng_pipe_get_addr(p, NNG_OPT_REMADDR, &ra);
189 if (rv == 0) {
190 if (ra.s_family == NNG_AF_INET) {
191 char ipAddr[INET_ADDRSTRLEN];
192 inet_ntop(AF_INET, (void*)&ra.s_in.sa_addr, ipAddr, INET_ADDRSTRLEN);
193 port = ntohs(ra.s_in.sa_port);
194 req["remote_host"] = ipAddr;
195 req["remote_port"] = port;
196 } else if (ra.s_family == NNG_AF_INET6) {
197 char ipAddr[INET6_ADDRSTRLEN];
198 inet_ntop(AF_INET6, (void*)&ra.s_in.sa_addr, ipAddr, INET6_ADDRSTRLEN);
199 port = ntohs(ra.s_in6.sa_port);
200 req["remote_host"] = ipAddr;
201 req["remote_port"] = port;
202 }
203 }
204 }
205
206 res = iter->second(std::move(req));
207 res["ret"] = NodeErrorCode::SUCCESS;
208 encodeMsg(msg, res);
209
210 nng_aio_set_msg(work->aio, msg);
211 work->state = Work::SEND;
212 nng_ctx_send(work->ctx, work->aio);
213
214 } catch (const NodeNngError& e) {
215 CLS_FATAL("{}", e.what());
216 work->state = Work::FINISH;
217
218 } catch (const NodeError& e) {
219 CLS_ERROR("{}", e.what());
220 res["ret"] = e.errcode();
221 res["msg"] = e.what();
222 encodeMsg(msg, res);
223 nng_aio_set_msg(work->aio, msg);
224 work->state = Work::SEND;
225 nng_ctx_send(work->ctx, work->aio);
226
227 } catch (const std::exception& e) {
228 CLS_ERROR("{}", e.what());
229 res["ret"] = NodeErrorCode::UNKNOWN_ERROR;
230 res["msg"] = e.what();
231 encodeMsg(msg, res);
232 nng_aio_set_msg(work->aio, msg);
233 work->state = Work::SEND;
234 nng_ctx_send(work->ctx, work->aio);
235
236 } catch (...) {
237 std::string errmsg = "Unknown error!";
238 CLS_ERROR("{}", errmsg);
239 res["ret"] = NodeErrorCode::UNKNOWN_ERROR;
240 res["msg"] = errmsg;
241 encodeMsg(msg, res);
242 nng_aio_set_msg(work->aio, msg);
243 work->state = Work::SEND;
244 nng_ctx_send(work->ctx, work->aio);
245 }
246 }

254};
