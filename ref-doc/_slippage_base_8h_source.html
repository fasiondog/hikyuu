206// For RNG-based slippages: copy the source (so the cloned params/seed are carried over), then
207// reset() to reseed the RNG from the (copied) seed. The base clone() then restores the common
208// members including m_kdata and clears m_calculated, so the clone keeps the source's kdata and
209// recalculates on the next setTO.
210#define SLIPPAGE_IMP_RNG(classname) \
211public: \
212 virtual SlippagePtr _clone() override { \
213 auto p = std::make_shared<classname>(*this); \
214 p->reset(); \
215 return p; \
216 } \
217 virtual price_t _getRealBuyPrice(const Datetime&, price_t) override; \
218 virtual price_t _getRealSellPrice(const Datetime&, price_t) override; \
