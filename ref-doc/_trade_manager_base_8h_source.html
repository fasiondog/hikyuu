220 FundsList funds_list = getFundsList(dates, ktype);

254 PriceList getProfitCumChangeCurve(const DatetimeList& dates,
255 const KQuery::KType& ktype = KQuery::DAY) {
256 FundsList funds_list = getFundsList(dates, ktype);
257 PriceList ret(funds_list.size());
258 int precision = getParam<int>("precision");
259 for (size_t i = 0, total = funds_list.size(); i < total; i++) {
260 price_t base = funds_list[i].total_base();
261 ret[i] = base != 0.0 ? roundEx(funds_list[i].total_assets() / base, precision)
262 : Null<price_t>();
263 }
264 return ret;
265 }
