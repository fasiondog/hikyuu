71 void start() {
72 // It is already in the executing state, return directly
73 HKU_IF_RETURN(!m_stop, void());
74
75 // Set the executing state
76 m_stop = false;
77
78 std::unique_lock<std::mutex> lock(m_mutex);
79
80 std::priority_queue<IntervalS> new_queue;
81 m_queue.swap(new_queue);
82 if (!m_tg) {
83 m_tg = new ThreadPool(m_work_num);
84 }
85
86 /*
87 * Rebuild the execution queue according to the existing timers and delete the invalid ones
88 */
89
90 std::forward_list<int> invalid_timers; // Records the invalid timers
91 for (auto iter = m_timers.begin(); iter != m_timers.end(); ++iter) {
92 int time_id = iter->first;
93 const Timer* timer = iter->second;
94 Datetime now = Datetime::now();
95
96 // Record the invalid timer id
97 if (timer->m_repeat_num <= 0 || (timer->m_end_date != Datetime::max() &&
98 timer->m_end_date + timer->m_end_time < now)) {
99 invalid_timers.push_front(time_id);
100 continue;
101 }
102
103 IntervalS s;
104 s.m_timer_id = time_id;
105 if (timer->m_start_time < TimeDelta()) {
106 Datetime first_start_time = timer->m_start_date + timer->m_end_time;
107 if (first_start_time >= now) {
108 s.m_time_point = first_start_time;
109 } else {
110 if (timer->m_repeat_num <= 1) {
111 invalid_timers.push_front(time_id);
112 continue;
113 }
114 s.m_time_point = now.startOfDay() + timer->m_end_time;
115 if (s.m_time_point < now) {
116 s.m_time_point = s.m_time_point + TimeDelta(1);
117 }
118 }
119
120 } else {
121 s.m_time_point = timer->m_start_date >= now.startOfDay()
122 ? timer->m_start_date + timer->m_start_time + timer->m_duration
123 : now + timer->m_duration;
124 if (timer->m_start_time != timer->m_end_time) {
125 Datetime point_date = s.m_time_point.startOfDay();
126 TimeDelta point = s.m_time_point - point_date;
127 if (point < timer->m_start_time) {
128 s.m_time_point = point_date + timer->m_start_time;
129 } else if (point > timer->m_end_time) {
130 s.m_time_point = point_date + timer->m_start_time + TimeDelta(1);
131 } else {
132 TimeDelta gap = point - timer->m_start_time;
133 if (gap % timer->m_duration != TimeDelta()) {
134 int x = int(gap / timer->m_duration) + 1;
135 s.m_time_point =
136 point_date + timer->m_start_time + timer->m_duration * double(x);
137 }
138 }
139 }
140 }
141
142 m_queue.push(s);
143 }
144
145 // Clear the invalid timers
146 for (auto id : invalid_timers) {
147 _removeTimer(id);
148 }
149
150 lock.unlock();
151 m_cond.notify_all();
152
153 m_detect_thread = std::thread([this]() { detectThread(); });
154 }

157 void stop() {
158 if (!m_stop) {
159 std::unique_lock<std::mutex> lock(m_mutex);
160 std::priority_queue<IntervalS> queue;
161 m_queue.swap(queue);
162 m_stop = true;
163 lock.unlock();
164 m_cond.notify_all();
165 }
166
167 if (m_detect_thread.joinable()) {
168 m_detect_thread.join();
169 }
170
171 if (!m_use_extend_tg && m_tg) {
172 m_tg->stop();
173 delete m_tg;
174 m_tg = nullptr;
175 }
176 }

209 template <typename F, typename... Args>
210 int addFunc(Datetime start_date, Datetime end_date, TimeDelta start_time, TimeDelta end_time,
211 int repeat_num, TimeDelta duration, F&& f, Args&&... args) {
212 HKU_CHECK(!start_date.isNull(), "Invalid start_date!");
213 HKU_CHECK(!end_date.isNull(), "Invalid end_date!");
214 Datetime start = start_date.startOfDay();
215 Datetime end = end_date.startOfDay();
216 HKU_CHECK(end >= start, "end_date({}) need > start_date({})!", end, start);
217 HKU_CHECK(start_time >= TimeDelta(0) && start_time <= TimeDelta(0, 23, 59, 59, 999, 999),
218 "Invalid start_time: {}", start_time.repr());
219 HKU_CHECK(end_time >= TimeDelta(0) && end_time <= TimeDelta(0, 23, 59, 59, 999, 999),
220 "Invalid end_time: {}", end_time.repr());
221 HKU_CHECK(end_time >= start_time, "end_time({}) need >= start_time({})!", end_time,
222 start_time);
223 HKU_CHECK(repeat_num > 0, "Invalid repeat_num: {}", repeat_num);
224 HKU_CHECK(duration > TimeDelta(0), "Invalid duration: {}", duration.repr());
225 return _addFunc(start, end, start_time, end_time, repeat_num, duration, std::forward<F>(f),
226 std::forward<Args>(args)...);

347 void detectThread() {
348 try {
349 while (!m_stop) {
350 Datetime now = Datetime::now();
351 std::unique_lock<std::mutex> lock(m_mutex);
352 if (m_queue.empty()) {
353 m_cond.wait(lock);
354 continue;
355 }
356
357 IntervalS s = m_queue.top();
358 if (s.m_time_point == Datetime::min()) {
359 break; // End the detection thread so that the dll can exit safely, because the
360 // atomic may be invalid when the dll exits
361 }
362
363 TimeDelta diff = s.m_time_point - now;
364 if (diff > TimeDelta()) {
365 m_cond.wait_for(lock, std::chrono::duration<int64_t, std::micro>(diff.ticks()));
366 continue;
367 }
368
369 m_queue.pop();
370
371 // Get the current time again
372 now = Datetime::now();
373
374 auto timer_iter = m_timers.find(s.m_timer_id);
375 if (timer_iter == m_timers.end()) {
376 continue;
377 }
378
379 auto timer = timer_iter->second;
380 try {
381 m_tg->submit(timer->m_func);
382 } catch (const std::exception& e) {
383 // The worker pool may have been stopped already (e.g. an externally
384 // supplied pool shut down first). Letting the exception escape the
385 // detection thread would call std::terminate, so drop this timer and
386 // keep draining the queue instead.
387 HKU_WARN("Failed to submit timer task (id={}): {}", s.m_timer_id, e.what());
388 _removeTimer(s.m_timer_id);
389 continue;
390 }
391
392 if (timer->m_repeat_num != std::numeric_limits<int>::max()) {
393 timer->m_repeat_num--;
394 }
395
396 if (timer->m_repeat_num <= 0) {
397 _removeTimer(s.m_timer_id);
398 continue;
399 }
400
401 // Calculate the time point of the next execution
402 Datetime today = now.startOfDay();
403 if (timer->m_start_time >= TimeDelta()) {
404 // The timer not executed at the given moment
405 s.m_time_point = s.m_time_point + timer->m_duration;
406 if (s.m_time_point < now) {
407 // The system time is adjusted forward
408 s.m_time_point = now;
409 }
410
411 // If the executable time range of the day is limited and the next execution
412 // moment exceeds the limit of the day
413 if (timer->m_start_time != timer->m_end_time &&
414 s.m_time_point > today + timer->m_end_time) {
415 s.m_time_point = today + timer->m_start_time + TimeDelta(1);
416 }
417
418 } else {
419 // The timer with the given daily running time
420 s.m_time_point =
421 s.m_time_point + (today - s.m_time_point.startOfDay() + TimeDelta(1));
422 }
423
424 if (timer->m_end_date != Datetime::max() &&
425 s.m_time_point > timer->m_end_date + timer->m_end_time) {
426 _removeTimer(s.m_timer_id);
427 continue;
428 }
429
430 // Push the next running time into the queue
431 m_queue.push(s);
432 }
433 } catch (const std::exception& e) {
434 // Last-resort guard: any uncaught exception in the detection thread would
435 // terminate the process. Log and let the thread exit cleanly.
436 HKU_ERROR("Unexpected exception in TimerManager detection thread: {}", e.what());
437 } catch (...) {
438 HKU_ERROR("Unknown exception in TimerManager detection thread");
439 }
440 }

442 // Allocate the timer_id
443 int getNewTimerId() {
444 int max_int = std::numeric_limits<int>::max();
445 HKU_WARN_IF_RETURN(m_timers.size() >= size_t(max_int), -1, "Timer queue is full!");
446
447 if (m_current_timer_id >= max_int) {
448 m_current_timer_id = 0;
449 } else {
450 m_current_timer_id++;
451 }
452
453 while (true) {
454 if (m_timers.find(m_current_timer_id) != m_timers.end()) {
455 if (m_current_timer_id >= max_int) {
456 m_current_timer_id = 0;
457 } else {
458 m_current_timer_id++;
459 }
460 } else {
461 break;
462 }
463 }
464 return m_current_timer_id;
465 }

489 struct IntervalS {
490 Datetime m_time_point; // The exact time point of the execution
491 int m_timer_id = -1; // The corresponding Timer, a negative value is invalid
492 bool operator<(const IntervalS& other) const {
493 return m_time_point > other.m_time_point;
494 }

497 template <typename F, typename... Args>
498 int _addFunc(Datetime start_date, Datetime end_date, TimeDelta start_time, TimeDelta end_time,
499 int repeat_num, TimeDelta duration, F&& f, Args&&... args) {
500 Datetime now = Datetime::now();
501 Datetime today = now.startOfDay();
502 HKU_CHECK(end_date >= today, "Invalid end_date {}, because today is {}", end_date, today);
503 if (end_date != Datetime::max()) {
504 HKU_CHECK(end_date + end_time >= now,
505 "Invalid param! You want end time is {}, but now is {}", end_date + end_time,
506 now);
507 }
508
509 Timer* timer = new Timer;
510 timer->m_start_date = start_date;
511 timer->m_end_date = end_date;
512 timer->m_start_time = start_time;
513 timer->m_end_time = end_time;
514 timer->m_repeat_num = repeat_num;
515 timer->m_duration = duration;
516 timer->m_func = std::bind(std::forward<F>(f), std::forward<Args>(args)...);
517
518 IntervalS s;
519 if (start_time < TimeDelta()) {
520 Datetime first_start_time = start_date + end_time;
521 if (first_start_time >= now) {
522 s.m_time_point = first_start_time;
523 } else {
524 HKU_CHECK(repeat_num > 1, "The time has expired! expect time {}, but now is {}",
525 first_start_time, now);
526 s.m_time_point = today + end_time;
527 if (s.m_time_point < now) {
528 s.m_time_point = s.m_time_point + TimeDelta(1);
529 }
530 }
531
532 } else {
533 s.m_time_point =
534 start_date >= today ? start_date + start_time + duration : now + duration;
535 if (timer->m_start_time != timer->m_end_time) {
536 Datetime point_date = s.m_time_point.startOfDay();
537 TimeDelta point = s.m_time_point - point_date;
538 if (point < timer->m_start_time) {
539 s.m_time_point = point_date + timer->m_start_time;
540 } else if (point > timer->m_end_time) {
541 s.m_time_point = point_date + timer->m_start_time + TimeDelta(1);
542 } else {
543 TimeDelta gap = point - timer->m_start_time;
544 if (gap % timer->m_duration != TimeDelta()) {
545 int x = int(gap / timer->m_duration) + 1;
546 s.m_time_point =
547 point_date + timer->m_start_time + timer->m_duration * double(x);
548 }
549 }
550 }
551 }
552
553 std::unique_lock<std::mutex> lock(m_mutex);
554 int id = getNewTimerId();
555 if (id < 0) {
556 delete timer;
557 lock.unlock();
558 HKU_THROW("Failed to get new id, maybe too timers!");
559 }
560
561 m_timers[id] = timer;
562 s.m_timer_id = id;
563 // HKU_TRACE("s.m_time_point: {}", s.m_time_point.repr());
564 m_queue.push(s);
565 lock.unlock();
566 m_cond.notify_all();
567 return id;
568 }
