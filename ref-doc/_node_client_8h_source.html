40 bool dial() noexcept {
41 std::lock_guard<std::mutex> lock(m_mutex);
42 _close();
43 // HKU_TRACE("dial: {}", m_server_addr);
44 int rv = nng_req0_open(&m_socket);
45 // HKU_ERROR_IF_RETURN(rv != 0, false, "Failed open req socket! {}", nng_strerror(rv));
46 HKU_IF_RETURN(rv != 0, false);
47 m_connected = true;
48
49 try {
50 // Set the socket connection parameters for sending the result
51 rv = nng_socket_set_ms(m_socket, NNG_OPT_RECONNMINT, 10);
52 NODE_NNG_CHECK(rv, "Failed nng_socket_set_ms!");
53
54 rv = nng_socket_set_ms(m_socket, NNG_OPT_RECONNMAXT, 15000);
55 NODE_NNG_CHECK(rv, "Failed nng_socket_set_ms!");
56
57 rv = nng_socket_set_ms(m_socket, NNG_OPT_SENDTIMEO, 10000);
58 NODE_NNG_CHECK(rv, "Failed nng_socket_set_ms!");
59
60 rv = nng_socket_set_ms(m_socket, NNG_OPT_RECVTIMEO, 10000);
61 NODE_NNG_CHECK(rv, "Failed nng_socket_set_ms!");
62
63 rv = nng_dial(m_socket, m_server_addr.c_str(), NULL, 0);
64 NODE_NNG_CHECK(rv, "Failed dial server: {}!", m_server_addr);
65
66 return true;
67
68 } catch (const std::exception& e) {
69 HKU_ERROR_IF(m_show_log, "Failed dail server: {}! {}", m_server_addr, e.what());
70 } catch (...) {
71 HKU_ERROR_IF(m_show_log, "Failed dail server: {}! Unknown error!", m_server_addr);
72 }
73
74 m_connected = false;
75 nng_close(m_socket);
76 return false;
77 }

120 bool _send(const json& req) const noexcept {
121 bool success = false;
122 // HKU_ERROR_IF_RETURN(!m_connected, success, "Not connected!");
123 HKU_IF_RETURN(!m_connected, success);
124
125 nng_msg* msg = nullptr;
126 int rv = nng_msg_alloc(&msg, 0);
127 // HKU_ERROR_IF_RETURN(rv != 0, success, "Failed nng_msg_alloc! {}", nng_strerror(rv));
128 HKU_IF_RETURN(rv != 0, success);
129
130 try {
131 encodeMsg(msg, req);
132 rv = nng_sendmsg(m_socket, msg, 0);
133 NODE_NNG_CHECK(rv, "Failed nng_sendmsg!");
134 success = true;
135
136 } catch (const std::exception& e) {
137 HKU_ERROR_IF(m_show_log, "Failed send result! {}", e.what());
138 } catch (...) {
139 HKU_ERROR_IF(m_show_log, "Failed send result! Unknown error!");
140 }
141
142 if (!success) {
143 nng_msg_free(msg);
144 }
145
146 return success;
147 }
148
149 bool _recv(json& res) noexcept {
150 bool success = false;
151 nng_msg* msg{nullptr};
152 int rv = nng_recvmsg(m_socket, &msg, 0);
153 if (rv != 0) {
154 HKU_ERROR_IF(m_show_log, "Failed nng_recvmsg! {}", nng_strerror(rv));
155 return success;
156 }
157
158 m_last_ack_time = Datetime::now();
159
160 try {
161 res = decodeMsg(msg);
162 success = true;
163
164 } catch (const std::exception& e) {
165 HKU_ERROR_IF(m_show_log, "Failed recv response! {}", e.what());
166 } catch (...) {
167 HKU_ERROR_IF(m_show_log, "Failed recv response! Unknown error!");
168 }
169
170 nng_msg_free(msg);
171 return success;
172 }
173
174private:
175 mutable std::mutex m_mutex;
176 std::string m_server_addr; // Server address
177 nng_socket m_socket;
178 Datetime m_last_ack_time{Datetime::now()}; // The time of the last received server response
179 std::atomic_bool m_connected{false};
180 std::atomic_bool m_show_log{true};
