459template <typename Duration>
460inline Datetime Datetime::fromLocalTime(std::chrono::local_time<Duration> local_time) {
461 if (local_time == std::chrono::local_time<Duration>::max()) {
462 return Datetime();
463 }
464 auto duration_us =
465 std::chrono::duration_cast<std::chrono::microseconds>(local_time.time_since_epoch());
466 return fromTimestamp(static_cast<int64_t>(duration_us.count()));
467}
468
469template <typename Clock, typename Duration>
470inline Datetime Datetime::fromTimePointUTC(std::chrono::time_point<Clock, Duration> time_point) {
471 if (time_point == std::chrono::time_point<Clock, Duration>::max()) {
472 return Datetime();
473 }
474 auto duration_us =
475 std::chrono::duration_cast<std::chrono::microseconds>(time_point.time_since_epoch());
476 return fromTimestampUTC(static_cast<int64_t>(duration_us.count()));
477}

492template <typename Duration>
493inline std::chrono::local_time<Duration> Datetime::to_local_time() const {
494 if (isNull()) {
495 return std::chrono::local_time<Duration>::max();
496 }
497 auto timestamp_us = timestamp();
498 auto target_duration =
499 std::chrono::duration_cast<Duration>(std::chrono::microseconds(timestamp_us));
500 return std::chrono::local_time<Duration>(target_duration);
501}
