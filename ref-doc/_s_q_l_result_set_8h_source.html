40template <class TableT, size_t page_size = 100>
41class SQLResultSet {
42 friend class SQLResultSetIterator<TableT, page_size>;
43
44public:
45#if CPP_STANDARD >= CPP_STANDARD_17

48 static const int PSIZE = page_size;

51 SQLResultSet() = default;

58 SQLResultSet(const DBConnectPtr& connect, const std::string& sql)
59 : m_connect(connect),
60 m_sql_template("id IN (SELECT id FROM {} WHERE {} {} LIMIT {} OFFSET {}) {}") {
61 // the plain path: the clauses can only be located in the text the caller wrote
62 WhereParts parts = splitWhereParts(sql);
63 _setup(parts.where, parts.orderBy, parts.limit, BoundValues{});

71 SQLResultSet(const DBConnectPtr& connect, const DBCondition& cond)
72 : m_connect(connect),
73 m_sql_template("id IN (SELECT id FROM {} WHERE {} {} LIMIT {} OFFSET {}) {}") {
74 // a condition already carries its parts, nothing has to be parsed out of the text
75 _setup(cond.sql(), cond.getOrderBy(), cond.getLimit(), cond.params());

107 size_t size() const {
108 HKU_IF_RETURN(!m_connect, 0);
109 std::string sql =
110 fmt::format("select count(1) from {} where {}", TableT::getTableName(), m_where);
111 size_t total = m_connect->queryNumber<size_t>(sql, 0, m_params);
112 // the row limit of the condition caps the reported size, the pages are cut the same way
113 return m_limit >= 0 ? std::min(total, static_cast<size_t>(m_limit)) : total;
114 }

131 size_t getPageCount() {
132 size_t total = size();
133 size_t n = total / page_size;
134 // the last page counts only when it holds rows: an exact multiple needs no extra page
135 return n * page_size >= total ? n : n + 1;
136 }

143 std::vector<TableT> getPage(size_t page) {
144 std::vector<TableT> result;
145 HKU_IF_RETURN(!m_connect || _pageLimit(page) <= 0, result);
146 m_connect->batchLoadView(result, _selectSQL(page), m_params);
147 return result;
148 }

150 TableT operator[](size_t index) {
151 return get(index);
152 }

154 TableT at(size_t index) {
155 TableT result = get(index);
156 HKU_CHECK_THROW(result.valid(), std::out_of_range, "Index is over");
157 return result;
158 }
159
160private:
162 void _setup(const std::string& where, const std::string& orderBy, int limit,
163 BoundValues params) {
164 m_params = std::move(params);
165 m_limit = limit;
166
167 std::string text = where;
168 trim(text);
169 m_where = text.empty() ? "1=1" : text;
170
171 if (orderBy.empty()) {
172 m_orderby_inner = "ORDER BY id";
173 } else {
174 m_orderby_inner = fmt::format("{}, id ASC", orderBy);
175 }
176
177 // the page rows are picked by id in the inner subquery, but the outer select reads them
178 // back with `id IN (...)`, whose result order is undefined without an ORDER BY of its own;
179 // without it the row sequence within a page (and hence the index-to-row mapping) is not
180 // guaranteed on every driver, so the outer query must re-apply the same order as the inner
181 m_orderby_outer = m_orderby_inner;
182 }
183
188 size_t _pageLimit(size_t page) const {
189 if (m_limit < 0) {
190 return page_size;
191 }
192
193 const size_t offset = page * page_size;
194 const size_t left = static_cast<size_t>(m_limit);
195 return offset >= left ? 0 : std::min(page_size, left - offset);
196 }
197
199 std::string _selectSQL(size_t page) const {
200 return fmt::format(
201 "{} where {}", TableT::getSelectSQL(),
202 fmt::format(fmt::runtime(m_sql_template), TableT::getTableName(), m_where,
203 m_orderby_inner, _pageLimit(page), page * page_size, m_orderby_outer));
204 }
205
206 TableT get(size_t index) {
207 TableT result{Null<TableT>()};
208 HKU_IF_RETURN(index == Null<size_t>(), result);
209
210 size_t page = index / page_size;
211 if (m_connect && page != m_current_page) {
212 m_buffer.clear();
213 if (_pageLimit(page) > 0) {
214 m_connect->batchLoadView(m_buffer, _selectSQL(page), m_params);
215 }
216 m_current_page = page;
217 }
218
219 HKU_IF_RETURN(m_buffer.empty(), result);
220
221 size_t pos = index - page * page_size;
222 HKU_IF_RETURN(pos >= m_buffer.size(), result);
223
224 result = m_buffer[index - page * page_size];
225 return result;
226 }
227
228private:
229 DBConnectPtr m_connect;
230 std::vector<TableT> m_buffer;
231 std::string m_where;
232 std::string m_sql_template;
233 std::string m_orderby_inner;
234 std::string m_orderby_outer;
235 BoundValues m_params;
236 int m_limit = -1;
237 size_t m_current_page = Null<size_t>();
238};

240template <class TableT, size_t page_size>
241class SQLResultSetIterator {
242public:
243 using ResultSet = SQLResultSet<TableT, page_size>;
244
245 SQLResultSetIterator() = default;
246 ~SQLResultSetIterator() = default;
247
248 explicit SQLResultSetIterator(ResultSet* result_set, size_t index)
249 : m_set(result_set), m_index(index) {
250 if (m_index != Null<size_t>()) {
251 m_value = std::move(m_set->get(index));
252 if (!m_value.valid()) {
253 m_index = Null<size_t>();
254 }
255 }
256 }
257
258 SQLResultSetIterator(const SQLResultSetIterator& other)
259 : m_set(other.m_set), m_index(other.m_index), m_value(other.m_value) {}
260
261 SQLResultSetIterator& operator=(const SQLResultSetIterator& other) {
262 if (this == &other)
263 return *this;
264 m_index = other.m_index;
265 m_set = other.m_set;
266 m_value = other.m_value;
267 return *this;
268 }
269
270 const TableT& operator*() const {
271 return m_value;
272 }
273
274 TableT& operator*() {
275 return m_value;
276 }
277
278 const TableT* const operator->() const {
279 return &m_value;
280 }
281
282 TableT* operator->() {
283 return &m_value;
284 }
285
286 // Prefix increment operator
287 SQLResultSetIterator& operator++() {
288 HKU_CHECK_THROW(m_index != Null<size_t>(), std::logic_error,
289 "Cannot increment an end iterator.");
290 m_index++;
291 m_value = std::move(m_set->get(m_index));
292 if (!m_value.valid()) {
293 m_index = Null<size_t>();
294 }
295 return *this;
296 }
297
298 bool operator!=(const SQLResultSetIterator& iter) const {
299 return m_index != iter.m_index;
300 }
301
302private:
303 ResultSet* m_set = nullptr;
304 size_t m_index = Null<size_t>();
305 TableT m_value;
306};
