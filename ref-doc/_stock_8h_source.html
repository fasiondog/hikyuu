340struct HKU_API Stock::Data {
341 string m_market; // The market abbreviation it belongs to
342 string m_code; // Security code
343 string m_market_code; // Market abbreviation + security code
344 string m_name; // Security name
345 uint32_t m_type; // Security type
346 bool m_valid; // Whether the security is currently valid
347 Datetime m_startDate; // Start date of the security
348 Datetime m_lastDate; // Last date of the security
349
350 StockWeightList m_weightList; // Equity/dividend adjustment (weight) record list
351 std::shared_mutex m_weight_mutex;
352 // Whether the weight data has been initialized (materialized at startup or lazily loaded as a
353 // fallback; it may be empty). When it is set:
354 // - Client mode: set when materialized at startup while load_stock_weight is enabled; set on
355 // the
356 // first lazy load for securities that were not materialized, e.g. added by addStock or newly
357 // constructed, when the config is disabled;
358 // - shm server role: set to false by releaseShmServerBaseInfoCache() after the snapshot
359 // containing finance data is published (the cache has been released and the memory returned),
360 // then set again after the next access lazily reloads it through the driver;
361 // An empty result sets it as well, so securities without weight data do not hit the driver
362 // again on every query.
363 mutable std::atomic_bool m_weight_ready{false};
364
365 // The equal-ratio adjustment multipliers of the effective ex-rights records, built lazily from
366 // m_weightList (the reference close of a record date comes from the daily line) and guarded by
367 // m_weight_mutex; released together with the weight cache.
368 mutable vector<EqualRecoverFactor> m_recover_factors;
369 mutable std::atomic_bool m_recover_ready{false};
370
371 mutable vector<HistoryFinanceInfo>
372 m_history_finance; // Historical financial info [report date, field 1, field 2, ...]
373 // Whether the historical finance data has been initialized (set when the main process preloads
374 // it at startup; set to false after it is released by releaseShmServerBaseInfoCache(), and set
375 // again after the next access lazily reloads it; in client mode it is never materialized
376 // locally, so it stays false)
377 mutable std::atomic_bool m_history_finance_ready{false};
378 mutable std::shared_mutex m_history_finance_mutex;
379
380 price_t m_tick;
381 price_t m_tickValue;
382 price_t m_unit;
383 int m_precision;
384 double m_minTradeNumber;
385 double m_maxTradeNumber;
386
387 std::unordered_set<string> m_ktype_preload; // Records whether the K-line data of this security
388 // needs to be preloaded
389 unordered_map<string, std::shared_ptr<KRecordList>> pKData;
390 unordered_map<string, std::shared_mutex*> pMutex;
391 unordered_map<string, Datetime> m_lastUpdate; // Last update time of each K-line data type
392
393 Data();
394 Data(const string& market, const string& code, const string& name, uint32_t type, bool valid,
