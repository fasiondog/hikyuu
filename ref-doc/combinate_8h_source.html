27template <class T>
28std::vector<std::vector<size_t>> combinateIndex(const std::vector<T>& inputs) {
29 size_t total = inputs.size();
30 HKU_CHECK(total <= 15, "The length of inputs exceeds the maximum limit!");
31
32 std::vector<std::vector<size_t>> result;
33 std::vector<size_t> current;
34 for (size_t i = 0; i < total; i++) {
35 for (size_t j = 0, len = result.size(); j < len; j++) {
36 current.resize(result[j].size());
37 std::copy(result[j].cbegin(), result[j].cend(), current.begin());
38 current.push_back(i);
39 result.push_back(std::move(current));
40 }
41 current.clear();
42 current.push_back(i);
43 result.push_back(std::move(current));
44 }
45
46 return result;
47}
