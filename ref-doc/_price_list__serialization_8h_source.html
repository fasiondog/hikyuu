37template <class Archive>
38void load(Archive& ar, hku::PriceList& values, unsigned int version) {
39 size_t count = 0;
40 unsigned int item_version = 0;
41 ar& BOOST_SERIALIZATION_NVP(count);
42 ar& BOOST_SERIALIZATION_NVP(item_version);
43 values.resize(count);
44 for (size_t i = 0; i < count; i++) {
45 std::string vstr;
46 ar >> boost::serialization::make_nvp("item", vstr);
47 if (vstr == "nan") {
48 values[i] = std::numeric_limits<double>::quiet_NaN();
49 } else if (vstr == "+inf") {
50 values[i] = std::numeric_limits<double>::infinity();
51 } else if (vstr == "-inf") {
52 values[i] = 0.0 - std::numeric_limits<double>::infinity();
53 } else {
54 values[i] = std::atof(vstr.c_str());
55 }
56 }
