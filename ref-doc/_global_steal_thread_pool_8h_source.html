52 explicit GlobalStealThreadPool(size_t n, bool until_empty = true)
53 : m_done(false), m_worker_num(n), m_running_until_empty(until_empty), m_sleep_count(0) {
54 try {
55 m_interrupt_flags.resize(m_worker_num, nullptr);
56 for (int i = 0; i < m_worker_num; i++) {
57 // Create the worker threads and their task queues
58 m_queues.emplace_back(new WorkStealQueue);
59 }
60 // The threads are started after all the thread resources have been initialized
61 for (int i = 0; i < m_worker_num; i++) {
62 m_threads.emplace_back(&GlobalStealThreadPool::worker_thread, this, i);
63 }
64 } catch (...) {
65 m_done.store(true, std::memory_order_release);
66 // Wake up and join the already started worker threads before the members are
67 // destroyed, otherwise the workers would access the destroyed members
68 m_cv.notify_all();
69 size_t started = m_threads.size();
70 for (size_t i = 0; i < started; i++) {
71 if (m_threads[i].joinable()) {
72 m_threads[i].join();
73 }
74 }
75 throw;
76 }
77 }

104 int wake_up() {
105 HKU_IF_RETURN(m_done.load(std::memory_order_acquire), 0);
106 int sleeping_count = m_sleep_count.load(std::memory_order_acquire);
107 if (sleeping_count <= 0) {
108 return 0;
109 }
110
111 // Get the current number of the remaining tasks
112 size_t remaining_tasks = remain_task_count();
113 if (remaining_tasks == 0) {
114 // There is no remaining task, no wake-up is needed
115 return 0;
116 }
117
118 // The intelligent wake-up strategy:
119 // 1. If the number of the tasks is greater than or equal to the number of the sleeping
120 // threads, use notify_all to wake up all the threads (more efficient)
121 // 2. If the number of the tasks is less than the number of the sleeping threads, wake up
122 // precisely the needed number of the threads
123 int threads_to_wake = 0;
124 if (remaining_tasks >= static_cast<size_t>(sleeping_count)) {
125 // The tasks are sufficient, use notify_all to wake up all the sleeping threads (better
126 // performance)
127 m_cv.notify_all();
128 threads_to_wake = sleeping_count;
129 } else {
130 // There are few tasks, wake up precisely as needed
131 threads_to_wake = static_cast<int>(remaining_tasks);
132 // Ensure that at least one thread is woken up to handle the task
133 threads_to_wake = std::max(threads_to_wake, 1);
134
135 // Wake up precisely the given number of the threads
136 for (int i = 0; i < threads_to_wake; ++i) {
137 m_cv.notify_one();
138 }
139 }
140
141 return threads_to_wake;
142 }

171 template <typename FunctionType>
172 auto submit(FunctionType&& f) {
173 if (thread_need_stop().isSet() || m_done.load(std::memory_order_acquire)) {
174 throw std::logic_error(
175 "You can't submit a task to the stopped GlobalStealThreadPool!!");
176 }
177
178 // The counter is incremented before the task is pushed so that join() can observe an
179 // in-flight task even while it is between pop and execution. A second shutdown check after
180 // the increment guards against the race where join() flips m_done right before the push.
181 m_running_task_count.fetch_add(1, std::memory_order_relaxed);
182 if (thread_need_stop().isSet() || m_done.load(std::memory_order_acquire)) {
183 _releaseRunningTask();
184 throw std::logic_error(
185 "You can't submit a task to the stopped GlobalStealThreadPool!!");
186 }
187
188 typedef typename std::invoke_result<FunctionType>::type result_type;
189 std::packaged_task<result_type()> task(std::forward<FunctionType>(f));
190 task_handle<result_type> res(task.get_future());
191
192 try {
193 std::thread::id id = std::this_thread::get_id();
194 if (local_work_queue() && id == thread_id()) {
195 // The local thread tasks enter the queue from the front (recursion becomes a stack)
196 local_work_queue()->push_front(std::move(task));
197 } else {
198 m_master_work_queue.push(std::move(task));
199 m_cv.notify_one();
200 }
201 } catch (...) {
202 _releaseRunningTask();
203 throw;
204 }
205
206 return res;
207 }

221 void stop() {
222 // Serialize with join(): two threads must never join the same worker concurrently
223 std::lock_guard<std::mutex> lock(m_join_mutex);
224 if (m_done.exchange(true, std::memory_order_acq_rel)) {
225 return;
226 }
227
228 // Wake a concurrent joiner sleeping on the drain wait (stop discards the queued tasks,
229 // so the counter alone would never reach zero)
230 {
231 std::lock_guard<std::mutex> lk(m_idle_mutex);
232 m_idle_cv.notify_all();
233 }
234
235 // Reset the sleep count
236 m_sleep_count.store(0, std::memory_order_release);
237
238 // At the same time the end task indication is added, so that it can also be terminated when
239 // the dll exits. The state change is done under m_cv_mutex so a worker cannot evaluate the
240 // wait predicate before m_done becomes visible and then miss the notify (lost wakeup).
241 {
242 std::lock_guard<std::mutex> lk(m_cv_mutex);
243 for (size_t i = 0; i < m_worker_num; i++) {
244 if (m_interrupt_flags[i]) {
245 m_interrupt_flags[i]->set();
246 }
247 m_queues[i]->push_front(FuncWrapper());
248 }
249 }
250
251 m_cv.notify_all(); // Wake up all the worker threads
252 for (size_t i = 0; i < m_worker_num; i++) {
253 if (m_threads[i].joinable()) {
254 m_threads[i].join();
255 }
256 }
257
258 m_master_work_queue.clear();
259 for (size_t i = 0; i < m_worker_num; i++) {
260 m_queues[i]->clear();
261 }
262 m_threads.clear();
263 }

269 void join() {
270 // It instructs every worker thread to stop running when no work task is got
271 if (m_running_until_empty) {
272 // Sleep-wait until there is no queued nor in-flight task instead of spinning (a long
273 // running task would otherwise burn a core for the whole join). The counter already
274 // covers queued tasks (incremented at submit time), so a zero counter means the pool
275 // is truly idle. stop() discards queued tasks without decrementing the counter, so
276 // m_done must also break the wait (stop notifies m_idle_cv when it sets m_done).
277 std::unique_lock<std::mutex> lk(m_idle_mutex);
278 m_idle_cv.wait(lk, [this] {
279 return m_done.load(std::memory_order_acquire) ||
280 m_running_task_count.load(std::memory_order_acquire) == 0;
281 });
282 }
283
284 // Serialize with other join()/stop() calls: the workers must be joined exactly once
285 std::lock_guard<std::mutex> lock(m_join_mutex);
286 if (m_done.load(std::memory_order_acquire)) {
287 return;
288 }
289
290 m_done.store(true, std::memory_order_release);
291
292 // Wake a concurrent joiner sleeping on the drain wait
293 {
294 std::lock_guard<std::mutex> lk(m_idle_mutex);
295 m_idle_cv.notify_all();
296 }
297
298 // The state change and the sentinel pushes are done under m_cv_mutex so a worker cannot
299 // evaluate the wait predicate before m_done becomes visible and then miss the notify
300 // (lost wakeup).
301 {
302 std::lock_guard<std::mutex> lk(m_cv_mutex);
303 if (m_running_until_empty) {
304 for (size_t i = 0; i < m_worker_num; i++) {
305 if (m_interrupt_flags[i]) {
306 m_interrupt_flags[i]->set();
307 }
308 }
309 }
310
311 for (size_t i = 0; i < m_worker_num; i++) {
312 m_master_work_queue.push(FuncWrapper());
313 }
314 }
315
316 // Wake up all the worker threads
317 m_cv.notify_all();
318
319 // Wait for the threads to be finished
320 for (size_t i = 0; i < m_worker_num; i++) {
321 if (m_threads[i].joinable()) {
322 m_threads[i].join();
323 }
324 }
325
326 m_done.store(true, std::memory_order_release);
327 m_master_work_queue.clear();
328 for (size_t i = 0; i < m_worker_num; i++) {
329 m_queues[i]->clear();
330 }
331 m_threads.clear();
332 }

353 bool run_available_task_once(bool local_only = false) {
354 HKU_IF_RETURN(m_done.load(std::memory_order_acquire) || thread_need_stop().isSet(), false);
355 bool task_run = false;
356 task_type task;
357 if (local_work_queue()) {
358 if (pop_task_from_local_queue(task)) {
359 if (!task.isNullTask()) {
360 RunningTaskGuard guard(*this);
361 task();
362 task_run = true;
363 } else {
364 thread_need_stop().set();
365 }
366 } else if (!local_only && pop_task_from_other_thread_queue(task)) {
367 RunningTaskGuard guard(*this);
368 task();
369 task_run = true;
370 } else if (!local_only && pop_task_from_master_queue(task)) {
371 if (!task.isNullTask()) {
372 RunningTaskGuard guard(*this);
373 task();
374 task_run = true;
375 } else {
376 thread_need_stop().set();
377 }
378 }
379 } else if (!local_only && pop_task_from_master_queue(task)) {
380 if (!task.isNullTask()) {
381 RunningTaskGuard guard(*this);
382 task();
383 task_run = true;
384 }
385 }
386 return task_run;

392 // Balances the increment done in submit(). Wakes a joiner sleeping on the drain wait when
393 // the last in-flight task finishes. The notify is done under m_idle_mutex so it cannot slip
394 // through the window between the waiter's predicate evaluation and its blocking.
395 void _releaseRunningTask() {
396 if (m_running_task_count.fetch_sub(1, std::memory_order_relaxed) == 1) {
397 std::lock_guard<std::mutex> lk(m_idle_mutex);
398 m_idle_cv.notify_all();
399 }
400 }

402 // Decrements the in-flight task counter on scope exit. Used to balance the increment done in
403 // submit() once a popped task has finished executing (including when it throws).
404 struct RunningTaskGuard {
405 GlobalStealThreadPool& pool;
406 explicit RunningTaskGuard(GlobalStealThreadPool& p) : pool(p) {}
407 ~RunningTaskGuard() {
408 pool._releaseRunningTask();
409 }

454 void worker_thread(int index) {
455 thread_id() = std::this_thread::get_id();
456 m_interrupt_flags[index] = &thread_need_stop();
457 thread_index() = index;
458 local_work_queue() = m_queues[index].get();
459 while (!thread_need_stop().isSet() && !m_done.load(std::memory_order_acquire)) {
460 run_pending_task();
461 }
462 local_work_queue() = nullptr;
463 m_interrupt_flags[index] = nullptr;
464 }

466 void run_pending_task() {
467 // Take the work task from the local queue first; if there is no local task, take it from
468 // the master queue If the task taken from the master queue is an empty task, this thread is
469 // considered to be ended; otherwise a task is stolen from the other work queues
470 task_type task;
471 if (pop_task_from_local_queue(task)) {
472 if (!task.isNullTask()) {
473 RunningTaskGuard guard(*this);
474 task();
475 } else {
476 thread_need_stop().set();
477 }
478 } else if (pop_task_from_master_queue(task)) {
479 if (!task.isNullTask()) {
480 RunningTaskGuard guard(*this);
481 task();
482 } else {
483 thread_need_stop().set();
484 }
485 } else if (pop_task_from_other_thread_queue(task)) {
486 RunningTaskGuard guard(*this);
487 task();
488 } else {
489 // Increase the sleep count before entering the waiting state
490 m_sleep_count.fetch_add(1, std::memory_order_acq_rel);
491
492 // std::this_thread::yield();
493 std::unique_lock<std::mutex> lk(m_cv_mutex);
494 m_cv.wait(lk, [this] {
495 return this->m_done.load(std::memory_order_acquire) ||
496 !this->m_master_work_queue.empty() ||
497 (local_work_queue() && !local_work_queue()->empty()) ||
498 has_other_remain_task();
499 });
500
501 // Decrease the sleep count after being woken up
502 m_sleep_count.fetch_sub(1, std::memory_order_acq_rel);
503 }
504 }

506 bool pop_task_from_master_queue(task_type& task) {
507 return m_master_work_queue.try_pop(task);
508 }

510 // cppcheck-suppress functionStatic // Suppress the cppcheck suggestion of converting it into a
511 // static function
512 bool pop_task_from_local_queue(task_type& task) {
513 return local_work_queue() && local_work_queue()->try_pop(task);
514 }
515
516 bool pop_task_from_other_thread_queue(task_type& task) {
517 for (int i = 0; i < m_worker_num; ++i) {
518 int index = (thread_index() + i + 1) % m_worker_num;
519 if (index != thread_index() && m_queues[index]->try_steal(task)) {
520 return true;
521 }
522 }
523 return false;
524 }
525
526 bool has_other_remain_task() {
527 for (int i = 0; i < m_worker_num; ++i) {
528 if (i != thread_index() && m_queues[i] && !m_queues[i]->empty()) {
529 return true;
530 }
531 }
532 return false;
533 }
