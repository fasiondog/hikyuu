77 // Find the index of the KRecord whose datetime equals `datetime` within the contiguous
78 // [first, first + count) range, or return Null<size_t>() when absent. The range must be
79 // sorted ascending by datetime.
80 static size_t getKRecordPosByDatetime(const KRecord* first, size_t count,
81 const Datetime& datetime) noexcept {
82 const KRecord* last = first + count;
83 const KRecord* it = std::lower_bound(
84 first, last, datetime,
85 [](const KRecord& record, const Datetime& dt) { return record.datetime < dt; });
86 if (it == last || it->datetime != datetime) {
87 return Null<size_t>();
88 }
89 return static_cast<size_t>(it - first);
90 }
