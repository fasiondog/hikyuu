272 std::shared_ptr<ResourceType> get() {
273 // If the TLS Pool is disabled, acquire it from the global pool directly
274 if (m_max_tls_pool_size == 0) {
275 return m_global_pool->get();
276 }
277
278 // 1. Check whether the TLS Pool version needs to be synchronized
279 auto &tls_pool = TlsPoolType::getInstance();
280 int current_version = m_version.load(std::memory_order_acquire);
281
282 if (tls_pool.getVersion() != current_version) {
283 // The version does not match, synchronize the parameters and the version
284 Parameter current_param;
285 {
286 std::shared_lock<std::shared_mutex> lock(m_param_mutex);
287 current_param = m_param;
288 }
289 tls_pool.syncVersion(current_version, current_param);
290 }
291
292 // 2. Acquire it from the TLS Pool preferentially (the fast path)
293 auto tls_result = tls_pool.get();
294 if (tls_result) {
295 return tls_result.value(); // The TLS Pool succeeded, extract the shared_ptr
296 }
297
298 // 3. The TLS Pool failed, try to acquire it from the global shared pool
299 return m_global_pool->get(); // It may return nullptr or throw CreateResourceException
300 }

320 std::shared_ptr<ResourceType> getWaitFor(uint64_t ms_timeout) {
321 // If the TLS Pool is disabled (max_tls_pool_size == 0), wait and acquire it from the global
322 // pool directly
323 if (m_max_tls_pool_size == 0) {
324 return m_global_pool->getWaitFor(ms_timeout); // It may throw an exception
325 }
326
327 // 1. Check whether the TLS Pool version needs to be synchronized
328 auto &tls_pool = TlsPoolType::getInstance();
329 int current_version = m_version.load(std::memory_order_acquire);
330
331 if (tls_pool.getVersion() != current_version) {
332 // The version does not match, synchronize the parameters and the version
333 Parameter current_param;
334 {
335 std::shared_lock<std::shared_mutex> lock(m_param_mutex);
336 current_param = m_param;
337 }
338 tls_pool.syncVersion(current_version, current_param);
339 }
340
341 // 2. Acquire it from the TLS Pool preferentially (the fast path)
342 auto tls_result = tls_pool.get();
343 if (tls_result) {
344 return tls_result.value(); // The TLS Pool succeeded, extract the shared_ptr
345 }
346
347 // 3. The TLS Pool failed, try to wait and acquire it from the global shared pool (with a
348 // timeout)

386 std::shared_ptr<ResourceType> getFromTlsPool() {
387 // Synchronize the version first
388 auto &tls_pool = TlsPoolType::getInstance();
389 int current_version = m_version.load(std::memory_order_acquire);
390
391 if (tls_pool.getVersion() != current_version) {
392 Parameter current_param;
393 {
394 std::shared_lock<std::shared_mutex> lock(m_param_mutex);
395 current_param = m_param;
396 }
397 tls_pool.syncVersion(current_version, current_param);
398 }
399
400 auto result = tls_pool.get();
401 return result ? result.value() : nullptr;
402 }
