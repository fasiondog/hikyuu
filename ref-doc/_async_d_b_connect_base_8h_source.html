357template <typename NumberType>
358net::awaitable<NumberType> AsyncDBConnectBase::queryNumber(const std::string &query,
359 NumberType default_val,
360 const BoundValues &params) {
361 auto st = co_await getStatementWithParams(query, params);
362 co_await st->exec();
363
364 if (!(co_await st->moveNext() && st->getNumColumns() == 1)) {
365 HKU_CHECK(default_val != Null<NumberType>(), "query doesn't result in exactly 1 element");
366 co_return default_val;
367 }
368
369 NumberType result = 0;
370 st->getColumn(0, result); // getColumn is a synchronous method
371
372 if (co_await st->moveNext()) {
373 HKU_CHECK(default_val != Null<NumberType>(), "query doesn't result in exactly 1 element");
374 co_return default_val;
375 }
376
377 co_return result;
378}

384template <typename T>
385net::awaitable<void> AsyncDBConnectBase::save(T &item, bool autotrans) {
386 auto st =
387 co_await (item.valid() ? getStatement(T::getUpdateSQL()) : getStatement(T::getInsertSQL()));
388
389 if (autotrans) {
390 co_await transaction();
391 }
392
393 std::exception_ptr saved_exception;
394 try {
395 if (item.valid()) {
396 item.update(st);
397 co_await st->exec();
398 } else {
399 item.save(st);
400 co_await st->exec();
401 item.rowid(st->getLastRowid()); // getLastRowid is a synchronous method
402 }
403
404 if (autotrans) {
405 co_await commit();
406 }
407 } catch (...) {
408 saved_exception = std::current_exception();
409 }
410
411 // Handle the rollback outside the try-catch
412 if (saved_exception) {
413 if (autotrans) {
414 try {
415 co_await rollback();
416 } catch (...) {
417 // Ignore the rollback exception and keep the original exception
418 }
419 }
420 std::rethrow_exception(saved_exception);
421 }
422 co_return;
423}

622template <class InputIterator, class Filter>
623net::awaitable<void> AsyncDBConnectBase::_batchProcessRange(InputIterator first, InputIterator last,
624 Filter match, bool save_mode,
625 bool autotrans) {
626 auto st = co_await getStatement(save_mode ? InputIterator::value_type::getInsertSQL()
627 : InputIterator::value_type::getUpdateSQL());
628
629 if (autotrans) {
630 co_await transaction();
631 }
632
633 std::exception_ptr saved_exception;
634 try {
635 for (InputIterator iter = first; iter != last; ++iter) {
636 if (!match(*iter)) {
637 continue;
638 }
639
640 if (save_mode) {
641 iter->save(st);
642 } else {
643 iter->update(st);
644 }
645 co_await st->exec();
646
647 if (save_mode) {
648 iter->rowid(st->getLastRowid()); // getLastRowid is a synchronous method
649 }
650 }
651
652 if (autotrans) {
653 co_await commit();
654 }
655 } catch (...) {
656 saved_exception = std::current_exception();
657 }
658
659 // Handle the rollback outside the try-catch
660 if (saved_exception) {
661 if (autotrans) {
662 try {
663 co_await rollback();
664 } catch (...) {
665 // Ignore the rollback exception and keep the original exception
666 }
667 }
668 std::rethrow_exception(saved_exception);
669 }
670 co_return;
671}

673template <class InputIterator>
674net::awaitable<void> AsyncDBConnectBase::batchSaveOrUpdate(InputIterator first, InputIterator last,
675 bool autotrans) {
676 if (first == last) {
677 co_return;
678 }
679
680 bool has_save = false;
681 bool has_update = false;
682 for (auto iter = first; iter != last; ++iter) {
683 if (iter->valid()) {
684 has_update = true;
685 } else {
686 has_save = true;
687 }
688 }
689
690 if (has_save) {
691 co_await _batchProcessRange(
692 first, last, [](const auto &item) { return !item.valid(); }, true, autotrans);
693 }
694 if (has_update) {
695 co_await _batchProcessRange(
696 first, last, [](const auto &item) { return item.valid(); }, false, autotrans);
697 }
698}
