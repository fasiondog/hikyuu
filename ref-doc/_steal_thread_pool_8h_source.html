57 explicit StealThreadPool(size_t n, bool until_empty = true)
58 : m_done(false), m_worker_num(n), m_running_until_empty(until_empty) {
59 try {
60 m_interrupt_flags.resize(m_worker_num);
61 for (int i = 0; i < m_worker_num; i++) {
62 // Create the worker threads and their task queues
63 m_queues.emplace_back(new WorkStealQueue);
64 }
65 // The threads are started after all the thread resources have been initialized
66 for (int i = 0; i < m_worker_num; i++) {
67 m_threads.emplace_back(&StealThreadPool::worker_thread, this, i);
68 m_thread_index[m_threads.back().get_id()] = i;
69 }
70 } catch (...) {
71 m_done = true;
72 // Wake up and join the already started worker threads before the members are
73 // destroyed, otherwise the workers would access the destroyed members
74 m_cv.notify_all();
75 size_t started = m_threads.size();
76 for (size_t i = 0; i < started; i++) {
77 m_interrupt_flags[i].set();
78 if (m_threads[i].joinable()) {
79 m_threads[i].join();
80 }
81 }
82 throw;
83 }
84 }

89 ~StealThreadPool() {
90 if (!m_done) {
91 join();
92 }
93 m_threads.clear();
94 }

123 template <typename FunctionType>
124 auto submit(FunctionType&& f) {
125 if (m_done) {
126 throw std::logic_error("You can't submit a task to the stopped StealThreadPool!!");
127 }
128
129 // The counter is incremented before the task is pushed so that join() can observe an
130 // in-flight task even while it is between pop and execution. A second m_done check after
131 // the increment guards against the race where join() flips m_done right before the push.
132 m_running_task_count.fetch_add(1, std::memory_order_relaxed);
133 if (m_done) {
134 _releaseRunningTask();
135 throw std::logic_error("You can't submit a task to the stopped StealThreadPool!!");
136 }
137
138 int index = -1;
139 auto iter = m_thread_index.find(std::this_thread::get_id());
140 if (iter != m_thread_index.end()) {
141 index = iter->second;
142 }
143
144 typedef typename std::invoke_result<FunctionType>::type result_type;
145 std::packaged_task<result_type()> task(std::forward<FunctionType>(f));
146 task_handle<result_type> res(task.get_future());
147 try {
148 if (index != -1 && !m_interrupt_flags[index]) {
149 // The local thread tasks enter the queue from the front (recursion becomes a stack)
150 m_queues[index]->push_front(std::move(task));
151 } else {
152 m_master_work_queue.push(std::move(task));
153 m_cv.notify_one();
154 }
155 } catch (...) {
156 _releaseRunningTask();
157 throw;
158 }
159 return res;
160 }

174 void stop() {
175 // Serialize with join(): two threads must never join the same worker concurrently
176 std::lock_guard<std::mutex> lock(m_join_mutex);
177 if (m_done.exchange(true, std::memory_order_acq_rel)) {
178 return;
179 }
180
181 // Wake a concurrent joiner sleeping on the drain wait (stop discards the queued tasks,
182 // so the counter alone would never reach zero)
183 {
184 std::lock_guard<std::mutex> lk(m_idle_mutex);
185 m_idle_cv.notify_all();
186 }
187
188 // At the same time the end task indication is added, so that it can also be terminated when
189 // the dll exits. The state change is done under m_cv_mutex so a worker cannot evaluate the
190 // wait predicate before m_done becomes visible and then miss the notify (lost wakeup).
191 {
192 std::lock_guard<std::mutex> lk(m_cv_mutex);
193 for (size_t i = 0; i < m_worker_num; i++) {
194 m_interrupt_flags[i].set();
195 m_queues[i]->push_front(FuncWrapper());
196 }
197 }
198
199 m_cv.notify_all(); // Wake up all the worker threads
200 for (size_t i = 0; i < m_worker_num; i++) {
201 if (m_threads[i].joinable()) {
202 m_threads[i].join();
203 }
204 }
205
206 m_master_work_queue.clear();
207 for (size_t i = 0; i < m_worker_num; i++) {
208 m_queues[i]->clear();
209 }
210 }

216 void join() {
217 // It instructs every worker thread to stop running when no work task is got
218 if (m_running_until_empty) {
219 // Sleep-wait until there is no queued nor in-flight task instead of spinning (a long
220 // running task would otherwise burn a core for the whole join). The counter already
221 // covers queued tasks (incremented at submit time), so a zero counter means the pool
222 // is truly idle. stop() discards queued tasks without decrementing the counter, so
223 // m_done must also break the wait (stop notifies m_idle_cv when it sets m_done).
224 std::unique_lock<std::mutex> lk(m_idle_mutex);
225 m_idle_cv.wait(lk, [this] {
226 return m_done.load(std::memory_order_acquire) ||
227 m_running_task_count.load(std::memory_order_acquire) == 0;
228 });
229 }
230
231 // Serialize with other join()/stop() calls: the workers must be joined exactly once
232 std::lock_guard<std::mutex> lock(m_join_mutex);
233 if (m_done) {
234 return;
235 }
236
237 m_done = true;
238
239 // Wake a concurrent joiner sleeping on the drain wait
240 {
241 std::lock_guard<std::mutex> lk(m_idle_mutex);
242 m_idle_cv.notify_all();
243 }
244
245 // The state change and the sentinel pushes are done under m_cv_mutex so a worker cannot
246 // evaluate the wait predicate before m_done becomes visible and then miss the notify
247 // (lost wakeup).
248 {
249 std::lock_guard<std::mutex> lk(m_cv_mutex);
250 if (m_running_until_empty) {
251 for (size_t i = 0; i < m_worker_num; i++) {
252 m_interrupt_flags[i].set();
253 }
254 }
255
256 for (size_t i = 0; i < m_worker_num; i++) {
257 m_master_work_queue.push(FuncWrapper());
258 }
259 }
260
261 // Wake up all the worker threads
262 m_cv.notify_all();
263
264 // Wait for the threads to be finished
265 for (size_t i = 0; i < m_worker_num; i++) {
266 if (m_threads[i].joinable()) {
267 m_threads[i].join();
268 }
269 }
270
271 m_master_work_queue.clear();
272 for (size_t i = 0; i < m_worker_num; i++) {
273 m_queues[i]->clear();
274 }
275 }

293 // Balances the increment done in submit(). Wakes a joiner sleeping on the drain wait when
294 // the last in-flight task finishes. The notify is done under m_idle_mutex so it cannot slip
295 // through the window between the waiter's predicate evaluation and its blocking.
296 void _releaseRunningTask() {
297 if (m_running_task_count.fetch_sub(1, std::memory_order_relaxed) == 1) {
298 std::lock_guard<std::mutex> lk(m_idle_mutex);
299 m_idle_cv.notify_all();
300 }
301 }

303 // Decrements the in-flight task counter on scope exit. Used to balance the increment done in
304 // submit() once a popped task has finished executing (including when it throws).
305 struct RunningTaskGuard {
306 StealThreadPool& pool;
307 explicit RunningTaskGuard(StealThreadPool& p) : pool(p) {}
308 ~RunningTaskGuard() {
309 pool._releaseRunningTask();
310 }
311 };

333 void worker_thread(int index) {
334 while (!m_done && !m_interrupt_flags[index]) {
335 run_pending_task(index);
336 }
337 }
338
339 void run_pending_task(int index) {
340 // Take the work task from the local queue first; if there is no local task, take it from
341 // the master queue If the task taken from the master queue is an empty task, this thread is
342 // considered to be ended; otherwise a task is stolen from the other work queues
343 task_type task;
344 if (pop_task_from_local_queue(task, index)) {
345 if (!task.isNullTask()) {
346 RunningTaskGuard guard(*this);
347 task();
348 } else {
349 m_interrupt_flags[index].set();
350 }
351 } else if (pop_task_from_master_queue(task)) {
352 if (!task.isNullTask()) {
353 RunningTaskGuard guard(*this);
354 task();
355 } else {
356 m_interrupt_flags[index].set();
357 }
358 } else if (pop_task_from_other_thread_queue(task, index)) {
359 RunningTaskGuard guard(*this);
360 task();
361 } else {
362 std::unique_lock<std::mutex> lk(m_cv_mutex);
363 m_cv.wait(lk, [this] { return this->m_done || !this->m_master_work_queue.empty(); });
364 }
365 }
366
367 bool pop_task_from_master_queue(task_type& task) {
368 return m_master_work_queue.try_pop(task);
369 }
370
371 // cppcheck-suppress functionStatic // Suppress the cppcheck suggestion of converting it into a
372 // static function
373 bool pop_task_from_local_queue(task_type& task, int index) {
374 return m_queues[index]->try_pop(task);
375 }
376
377 bool pop_task_from_other_thread_queue(task_type& task, int index) {
378 for (int i = 0; i < m_worker_num; ++i) {
379 int pos = (index + i + 1) % m_worker_num;
380 if (pos != index && !m_interrupt_flags[pos] && m_queues[pos]->try_steal(task)) {
381 return true;
382 }
383 }
384 return false;
385 }
