50 explicit ThreadPool(size_t n, bool until_empty = true)
51 : m_done(false), m_worker_num(n), m_running_until_empty(until_empty) {
52 try {
53 // The threads are started after all the thread resources have been initialized
54 for (int i = 0; i < m_worker_num; i++) {
55 // Create the worker threads and their task queues
56 m_threads.emplace_back(&ThreadPool::worker_thread, this, i);
57 }
58 } catch (...) {
59 m_done = true;
60 // Wake up and join the already started worker threads before the members are
61 // destroyed, otherwise the workers would access the destroyed members
62 size_t started = m_threads.size();
63 for (size_t i = 0; i < started; i++) {
64 m_master_work_queue.push(FuncWrapper());
65 }
66 m_master_work_queue.notify_all();
67 for (size_t i = 0; i < started; i++) {
68 if (m_threads[i].joinable()) {
69 m_threads[i].join();
70 }
71 }
72 throw;
73 }
74 }

102 auto submit(FunctionType&& f) {

107 std::packaged_task<result_type()> task(std::forward<FunctionType>(f));

130 void stop() {
131 if (m_done.exchange(true, std::memory_order_relaxed)) {
132 return;
133 }
134
135 // At the same time the end task indication is added, so that it can also be terminated when
136 // the dll exits
137 for (size_t i = 0; i < m_worker_num; i++) {
138 m_master_work_queue.push(FuncWrapper());
139 }
140
141 m_master_work_queue.notify_all();
142
143 {
144 std::lock_guard<std::mutex> lock(m_mutex_join);
145 for (size_t i = 0; i < m_worker_num; i++) {
146 if (m_threads[i].joinable()) {
147 m_threads[i].join();
148 }
149 }
150 }
151
152 m_master_work_queue.clear();
153 }

159 void join() {
160 if (m_done) {
161 return;
162 }
163
164 // It instructs every worker thread to stop running when no work task is got
165 if (!m_running_until_empty) {
166 m_done = true;
167 }
168
169 // It is still possible that some thread does not get it and thus is not terminated
170 for (size_t i = 0; i < 2 * m_worker_num; i++) {
171 m_master_work_queue.push(FuncWrapper());
172 }
173
174 m_master_work_queue.notify_all();
175
176 {
177 std::lock_guard<std::mutex> lock(m_mutex_join);
178 for (size_t i = 0; i < m_worker_num; i++) {
179 if (m_threads[i].joinable()) {
180 m_threads[i].join();
181 }
182 }
183 }
184
185 m_done = true;
186 m_master_work_queue.clear();
187 }
