183 struct ResourceDeleter {
184 ResourceTlsVersionPool *pool;
185 std::thread::id owner_thread_id; // Records the thread ID when the resource is acquired
186
187 void operator()(ResourceType *resource) const {
188 if (!resource) {
189 return;
190 }
191
192 auto current_thread = std::this_thread::get_id();
193 if (current_thread != owner_thread_id) {
194 // A cross-thread return: the resource is deleted directly to avoid accessing the
195 // thread_local variables. The owner pool state (maintained by its owner thread
196 // only) is not touched, so the pool capacity consumed here is never recovered
197 // afterwards.
198 HKU_WARN(
199 "Resource returned from different thread, deleting directly to avoid "
200 "undefined behavior");
201 delete resource;
202 } else if (pool) {
203 // A same-thread return: it is returned to the pool normally
204 pool->returnResource(resource);
205 } else {
206 // The pool is empty (possibly during the destruction), delete it directly
207 delete resource;
208 }
209 }
210 };

260 stdx::expected<ResourcePtr, std::string> get() {
261 // 1. Try to get it from the idle list (dequeue from the head of the Ring Buffer)
262 if (m_freeCount > 0) {
263 ResourceType *p = m_resourceList[m_head];
264 m_resourceList[m_head] = nullptr;
265 m_head = (m_head + 1) % MAX_POOL_SIZE_LIMIT;
266 m_freeCount--;
267
268 // Check whether the resource version matches
269 if (p->getVersion() != m_version) {
270 // The version does not match, destroy the old resource
271 delete p;
272 m_count--;
273
274 // Try to create a new version resource
275 if (m_count < m_maxCount) {
276 try {
277 p = new ResourceType(m_param);
278 p->setVersion(m_version);
279 m_count++;
280 return ResourcePtr(p, ResourceDeleter{this, std::this_thread::get_id()});
281 } catch (const std::exception &e) {
282 return stdx::unexpected(
283 fmt::format("Failed to create resource: {}", e.what()));
284 } catch (...) {
285 return stdx::unexpected("Failed to create resource: Unknown error");
286 }
287 } else {
288 return stdx::unexpected("No available resources and maximum pool size reached");
289 }
290 }
291
292 // The version matches, return normally
293 return ResourcePtr(p, ResourceDeleter{this, std::this_thread::get_id()});
294 }
295
296 // 2. There is no idle resource, check whether a new one can be created
297 if (m_count >= m_maxCount) {
298 return stdx::unexpected("No available resources and maximum pool size reached");
299 }
300
301 // 3. Create a new resource
302 ResourceType *p = nullptr;
303 try {
304 p = new ResourceType(m_param);

306 } catch (const std::exception &e) {

383 void releaseIdleResource() {
384 for (size_t i = 0; i < m_freeCount; ++i) {
385 size_t idx = (m_head + i) % MAX_POOL_SIZE_LIMIT;
386 if (m_resourceList[idx]) {
387 delete m_resourceList[idx];
388 m_resourceList[idx] = nullptr;
389 }
390 }
391 m_count -= m_freeCount;
