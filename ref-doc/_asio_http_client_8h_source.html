45inline void validateHttpHeader(const std::string& name, const std::string& value) {
46 static const std::string token_special = "!#$%&'*+-.^_`|~";
47 HKU_CHECK(!name.empty(), "Empty http header name");
48 for (unsigned char c : name) {
49 bool is_token = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
50 (c >= '0' && c <= '9') || token_special.find(c) != std::string::npos;
51 HKU_CHECK(is_token, "Invalid http header name: {}", name);
52 }
53 for (unsigned char c : value) {
54 // HTAB is the only allowed control character; CR/LF would inject headers
55 HKU_CHECK(c == '\t' || c >= ' ', "Control character in http header '{}'", name);
56 HKU_CHECK(c != 0x7f, "Control character in http header '{}'", name);
57 }
58}
59
61inline void validateHttpHeaders(const HttpHeaders& headers) {
62 for (const auto& [name, value] : headers) {
63 validateHttpHeader(name, value);
64 }
65}
