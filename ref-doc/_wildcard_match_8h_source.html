18inline void utf8_to_unicode(const std::string& utf8_str, std::vector<int32_t>& out) {
19 out.clear();
20 if (utf8_str.empty()) {
21 return;
22 }
23
24 const uint8_t* ptr = reinterpret_cast<const uint8_t*>(utf8_str.data());
25 size_t len = utf8_str.size();
26
27 while (len > 0) {
28 int32_t codepoint;
29 size_t bytes = utf8proc_iterate(ptr, len, &codepoint);
30 if (bytes == 0) {
31 break;
32 }
33 out.push_back(codepoint);
34 ptr += bytes;
35 len -= bytes;
36 }
37}

39inline bool wildcardMatchImpl(const std::vector<int32_t>& text,
40 const std::vector<int32_t>& pattern) {
41 size_t m = text.size();
42 size_t n = pattern.size();
43
44 size_t i = 0;
45 size_t j = 0;
46 size_t starIdx = std::string::npos;
47 size_t matchIdx = 0;
48
49 while (i < m) {
50 if (j < n && (pattern[j] == '?' || pattern[j] == text[i])) {
51 ++i;
52 ++j;
53 } else if (j < n && pattern[j] == '*') {
54 starIdx = j;
55 matchIdx = i;
56 ++j;
57 } else if (starIdx != std::string::npos) {
58 ++matchIdx;
59 i = matchIdx;
60 j = starIdx + 1;
61 } else {
62 return false;
63 }
64 }
65
66 while (j < n && pattern[j] == '*') {
67 ++j;
68 }
69
70 return j == n;
71}

73inline bool wildcardMatch(const std::string& text, const std::string& pattern) {
74 bool hasWildcard =
75 (pattern.find('*') != std::string::npos) || (pattern.find('?') != std::string::npos);
76
77 if (hasWildcard) {
78 std::vector<int32_t> text_unicode;
79 std::vector<int32_t> pattern_unicode;
80 utf8_to_unicode(text, text_unicode);
81 utf8_to_unicode(pattern, pattern_unicode);
82 return wildcardMatchImpl(text_unicode, pattern_unicode);
83 } else {
84 std::vector<int32_t> text_unicode;
85 std::vector<int32_t> pattern_unicode;
86 utf8_to_unicode(text, text_unicode);
87 utf8_to_unicode(pattern, pattern_unicode);
88
89 size_t patternLen = pattern_unicode.size();
90 size_t textLen = text_unicode.size();
91
92 if (patternLen > textLen) {
93 return false;
94 }
95
96 for (size_t i = 0; i < patternLen; ++i) {
97 if (text_unicode[i] != pattern_unicode[i]) {
98 return false;
99 }
100 }
101 return true;
102 }
103}
