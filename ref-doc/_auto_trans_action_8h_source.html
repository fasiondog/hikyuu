70class TransAction {
71public:
76 explicit TransAction(const DBConnectPtr& driver)
77 : m_driver(driver), m_committed(false), m_started(true) {
78 HKU_CHECK(driver, "Null DBConnectPtr!");
79 m_driver->transaction();
80 }
81
83 ~TransAction() {
84 // If the transaction has not been committed actively it is regarded as needing a rollback
85 if (m_started && !m_committed) {
86 HKU_WARN("The transaction is rolled back!");
87 m_driver->rollback();
88 } else if (!m_committed) {
89 HKU_WARN("Not manul begin transaction!");
90 }
91 }
92
94 const DBConnectPtr& connect() const {
95 return m_driver;
96 }
97
99 void begin() {
100 if (!m_started) {
101 m_driver->transaction();
102 m_committed = false;
103 m_started = true;
104 }
105 }
106
108 void end() {
109 HKU_CHECK(m_started, "No transaction has started!");
110 if (!m_committed) {
111 m_driver->commit();
112 m_committed = true;
113 m_started = false;
114 }
115 }
116
117private:
118 TransAction() = delete;
119 TransAction(const AutoTransAction&) = delete;
120 TransAction& operator=(const AutoTransAction&) = delete;
121
122private:
123 DBConnectPtr m_driver;
124 bool m_committed;
125 bool m_started;
126};
