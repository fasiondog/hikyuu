219template <class EventT>
220class AsyncSerialEventProcessor {
221public:
226 explicit AsyncSerialEventProcessor(bool quit_wait = true) : m_quit_wait(quit_wait) {
227 m_tg = std::unique_ptr<ThreadPool>(new ThreadPool(1));
228 }
229
231 virtual ~AsyncSerialEventProcessor() {
232 if (m_quit_wait) {
233 m_tg->join();
234 } else {
235 m_tg->stop();
236 }
237 }

246 FilterNodePtr addAction(const EventT& event, const FilterNodePtr& action) {
247 HKU_CHECK(action, "Input action is null!");
248 std::lock_guard<std::shared_mutex> lock(m_mutex);
249 auto iter = m_trees.find(event);
250 if (iter != m_trees.end()) {
251 iter->second->addChild(action);
252 } else {
253 m_trees[event] = action;
254 }
255 return action;
256 }
257
264 void dispatch(const EventT& event, const any_t& data) {
265 m_tg->submit([this, event, data] {
266 try {
267 std::shared_lock<std::shared_mutex> lock(m_mutex);
268 auto iter = m_trees.find(event);
269 HKU_WARN_IF_RETURN(iter == m_trees.end(), void(),
270 "There is no matching handling method for the event({})!",
271 event);
272 iter->second->run(data);
273 } catch (const std::exception& e) {
274 HKU_WARN("Exception in event dispatch: {}", e.what());
275 } catch (...) {
276 HKU_WARN("Unknown exception in event dispatch!");
277 }
278 });
279 }
280
281private:
282 mutable std::shared_mutex m_mutex;
283 std::unordered_map<EventT, FilterNodePtr> m_trees;
284 std::unique_ptr<ThreadPool> m_tg;
285 bool m_quit_wait = true;
286};
