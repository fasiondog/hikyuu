57 ~AsyncAutoTransAction() {
58 if (!m_started || !m_driver || !m_io_context) {
59 return;
60 }
61
62 if (m_io_context->stopped()) {
63 HKU_WARN(
64 "AsyncAutoTransAction: io_context stopped, the transaction cannot be committed "
65 "and will be rolled back when the connection is closed!");
66 return;
67 }
68
69 // Start a detached coroutine to commit (fire and forget); it rolls back automatically
70 // when the commit fails
71 net::asio::co_spawn(
72 *m_io_context,
73 [driver = m_driver]() -> net::awaitable<void> {
74 try {
75 co_await driver->commit();
76 co_return;
77 } catch (const std::exception& e) {
78 HKU_ERROR("Failed to commit transaction! {}", e.what());
79 } catch (...) {
80 HKU_ERROR("Failed to commit! Unknown exception!");
81 }
82 try {
83 co_await driver->rollback();
84 HKU_INFO("Transaction rolled back successfully!");
85 } catch (...) {
86 HKU_ERROR("Failed to rollback transaction! Unknown exception!");
87 }
88 },
89 net::asio::detached);
90 }
91
92private:
93 /* Private constructor */
94 explicit AsyncAutoTransAction(const AsyncDBConnectPtr& driver)
95 : m_driver(driver), m_io_context(nullptr), m_started(false) {
96 HKU_CHECK(m_driver, "Null AsyncDBConnectPtr!");
97 }
98
99 /* Internal method: start the transaction */
100 net::awaitable<void> startTransaction() {
101 // Get the io_context of the current coroutine environment
102 auto exec = co_await net::this_coro::executor;
103 m_io_context = &static_cast<boost::asio::io_context&>(exec.context());
104
105 // Start the transaction
106 co_await m_driver->transaction();
107 m_started = true;
108 }
109
110private:
111 AsyncDBConnectPtr m_driver;
112 boost::asio::io_context* m_io_context = nullptr;
113 bool m_started = false;
114};

139 static net::awaitable<std::shared_ptr<AsyncTransAction>> create(

142 co_await action->begin();
143 co_return action;
144 }
145
147 const AsyncDBConnectPtr& connect() const {
148 return m_driver;
149 }

155 net::awaitable<void> begin() {
156 if (!m_started) {
157 // Get the io_context of the current coroutine environment
158 auto exec = co_await net::this_coro::executor;
159 m_io_context = &static_cast<boost::asio::io_context&>(exec.context());
160
161 // Start the transaction
162 co_await m_driver->transaction();
163 m_started = true;
164 m_committed = false;
165 }
166 co_return;
167 }
168
173 net::awaitable<void> end() {
174 HKU_CHECK(m_started, "No transaction has started!");
175 if (!m_committed) {
176 co_await m_driver->commit();
177 m_committed = true;
178 m_started = false;
179 }
180 co_return;
181 }
182
184 net::awaitable<void> rollback() {
185 if (m_started && !m_committed) {
186 co_await m_driver->rollback();
187 m_started = false;
188 m_committed = false;
189 }
190 co_return;
191 }
192
194 ~AsyncTransAction() {
195 // If the transaction has not been committed actively it is regarded as needing a rollback
196 if (m_started && !m_committed && m_driver && m_io_context) {
197 HKU_WARN("AsyncTransAction: The transaction is rolled back in destructor!");
198
199 // Start a detached coroutine to roll back (fire and forget)
200 boost::asio::co_spawn(
201 *m_io_context,
202 [driver = m_driver]() -> net::awaitable<void> {
203 try {
204 co_await driver->rollback();
205 } catch (const std::exception& e) {
206 HKU_WARN("Failed to rollback transaction! {}", e.what());
207 } catch (...) {
208 HKU_WARN("Failed to rollback transaction! Unknown exception!");
209 }
210 },
211 boost::asio::detached);
212 }
213 }
214
215private:
217 explicit AsyncTransAction(const AsyncDBConnectPtr& driver)
218 : m_driver(driver), m_io_context(nullptr), m_committed(false), m_started(false) {
219 HKU_CHECK(m_driver, "Null AsyncDBConnectPtr!");
220 }
221
222private:
223 AsyncDBConnectPtr m_driver;
224 net::asio::io_context* m_io_context = nullptr;
225 bool m_committed = false;
226 bool m_started = false;
227};
