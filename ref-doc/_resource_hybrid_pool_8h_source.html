112 explicit ResourceHybridPool(const Parameter &param,
113 size_t max_tls_pool_size = MAX_TLS_POOL_SIZE_LIMIT,
114 size_t max_global_pool_size = 64)
115 : m_max_global_pool_size(max_global_pool_size), m_max_tls_pool_size(max_tls_pool_size) {
116 // Check and truncate the TLS pool size
117 if (m_max_tls_pool_size > MAX_TLS_POOL_SIZE_LIMIT) {
118 HKU_WARN("max_tls_pool_size({}) exceeds physical limit ({}), truncated to {}",

123 // Initialize the default parameters of the TLS Pool

126 // Set the actually used size of the TLS Pool (maxCount is called after getInstance())

128
129 // Create the global shared pool (with the size given at runtime, supporting the
130 // multi-thread concurrency)
131 m_global_pool = std::make_unique<GlobalPoolType>(param, m_max_global_pool_size);

161 std::shared_ptr<ResourceType> get() {
162 // If the TLS Pool is disabled, acquire it from the global pool directly
163 if (m_max_tls_pool_size == 0) {
164 return m_global_pool->get();
165 }
166
167 // Acquire it from the TLS Pool preferentially (the fast path)
168 auto tls_result = TlsPoolType::getInstance().get();
169 if (tls_result) {
170 return tls_result.value(); // The TLS Pool succeeded, extract the shared_ptr
171 }
172
173 // The TLS Pool failed, try to acquire it from the global shared pool
174 return m_global_pool->get(); // It may return nullptr or throw CreateResourceException
175 }

192 std::shared_ptr<ResourceType> getWaitFor(uint64_t ms_timeout) {
193 // If the TLS Pool is disabled (max_tls_pool_size == 0), wait and acquire it from the global
194 // pool directly
195 if (m_max_tls_pool_size == 0) {
196 return m_global_pool->getWaitFor(ms_timeout); // It may throw an exception
197 }
198
199 // Acquire it from the TLS Pool preferentially (the fast path)
200 auto tls_result = TlsPoolType::getInstance().get();
201 if (tls_result) {
202 return tls_result.value(); // The TLS Pool succeeded, extract the shared_ptr
203 }
204
205 // The TLS Pool failed, try to wait and acquire it from the global shared pool (with a
206 // timeout)
207 return m_global_pool->getWaitFor(ms_timeout); // It may throw GetResourceTimeoutException
208 }
