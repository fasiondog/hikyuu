47 bool load(const std::string& pluginname, bool print = true) noexcept {
48 std::string filename = getFileName(pluginname);
49 // HKU_WARN_IF_RETURN(!existFile(filename), false, "file({}) not exist!", filename);
50 HKU_DEBUG_IF_RETURN(!existFile(filename), false, "file({}) not exist!", filename);
51
52#if HKU_OS_WINDOWS
53 m_handle = LoadLibrary(HKU_PATH(filename).c_str());
54 if (!m_handle) {
55 HKU_WARN_IF(print, "load plugin({}) failed! errcode: {}", filename, GetLastError());
56 return false;
57 }
58
59#else
60 m_handle = dlopen(filename.c_str(), RTLD_LAZY);
61 if (!m_handle) {
62 HKU_WARN_IF(print, "load plugin({}) failed! {}", filename, dlerror());
63 return false;
64 }
65#endif
66
67 typedef PluginBase* (*CreateFunction)();
68 CreateFunction createFunction =
69 reinterpret_cast<CreateFunction>(getFunciton("createPlugin"));
70 if (!createFunction) {
71 HKU_WARN_IF(print, "Failed to get plugin({}) handle!", filename);
72 return false;
73 }
74
75 m_plugin.reset(createFunction());
76 if (!m_plugin) {
77 HKU_ERROR_IF(print, "Failed to create plugin ({})!", filename);
78 unload();
79 m_plugin.reset();
80 return false;
81 }
82
83 return true;
84 }
85
86 std::string getFileName(const std::string& pluginname) const noexcept {
87#if HKU_OS_WINDOWS
88 return fmt::format("{}/{}.dll", m_path, pluginname);
89#elif HKU_OS_LINUX
90 return fmt::format("{}/lib{}.so", m_path, pluginname);
91#elif HKU_OS_OSX
92 return fmt::format("{}/lib{}.dylib", m_path, pluginname);
93#endif
94 }
95
96private:
97 void unload() noexcept {
98 m_plugin.reset();
99 if (m_handle) {
100#if HKU_OS_WINDOWS
101 FreeLibrary(m_handle);
102#else
103 dlclose(m_handle);
104#endif
105 m_handle = nullptr;
106 }
107 }
108
109 void* getFunciton(const char* symbol) noexcept {
110#if HKU_OS_WINDOWS
111 void* func = (void*)GetProcAddress(m_handle, symbol);
112#else
113 void* func = dlsym(m_handle, symbol);
114#endif
115 return func;
116 }
117
118private:
119#if HKU_OS_WINDOWS
120 HMODULE m_handle{nullptr};
121#else
122 void* m_handle{nullptr};
123#endif
124 std::string m_path;
125 std::unique_ptr<PluginBase> m_plugin{nullptr};
126};
