291template <typename T>
292typename std::enable_if<std::numeric_limits<T>::is_integer>::type AsyncSQLStatementBase::getColumn(
293 int idx, T &item) {
294 if constexpr (std::is_same_v<T, uint64_t> || std::is_same_v<T, unsigned long long> ||
295 (std::is_unsigned_v<T> && sizeof(T) == 8)) {
296 uint64_t temp;
297 sub_getColumnAsUInt64(idx, temp);
298 item = static_cast<T>(temp);
299 } else if constexpr (std::is_signed_v<T> && sizeof(T) < 8) {
300 // Reject silently narrowing an int64 value into a smaller signed type
301 int64_t temp;
302 sub_getColumnAsInt64(idx, temp);
303 if (temp == Null<int64_t>()) {
304 // Map the NULL sentinel onto the target type's own Null sentinel
305 item = (std::numeric_limits<T>::max)();
306 return;
307 }
308 if (temp < static_cast<int64_t>(std::numeric_limits<T>::min()) ||
309 temp > static_cast<int64_t>(std::numeric_limits<T>::max())) {
310 throw exception("Column value overflows the target integer type");
311 }
312 item = static_cast<T>(temp);
313 } else if constexpr (std::is_unsigned_v<T>) {
314 // Smaller unsigned types: reject negative values and values above the target range
315 int64_t temp;
316 sub_getColumnAsInt64(idx, temp);
317 if (temp == Null<int64_t>()) {
318 item = (std::numeric_limits<T>::max)();
319 return;
320 }
321 if (temp < 0 ||
322 static_cast<uint64_t>(temp) > static_cast<uint64_t>(std::numeric_limits<T>::max())) {
323 throw exception("Column value overflows the target unsigned integer type");
324 }
325 item = static_cast<T>(temp);
326 } else {
327 sub_getColumnAsInt64(idx, reinterpret_cast<int64_t &>(item));
328 }
329}

332typename std::enable_if<!std::numeric_limits<T>::is_integer>::type AsyncSQLStatementBase::getColumn(
333 int idx, T &item) {
334 std::string tmp;
335 try {
336 sub_getColumnAsBlob(idx, tmp);
337 } catch (null_blob_exception &) {
338 return;
339 }
340 std::istringstream sin(tmp);
341 boost::archive::binary_iarchive ia(sin);
342 ia >> BOOST_SERIALIZATION_NVP(item);
343}

417inline void AsyncSQLStatementBase::bind_params(const BoundValues &params) {
418 for (size_t i = 0, len = params.size(); i < len; ++i) {
419 std::visit(
420 [&](const auto &value) {
421 using U = std::decay_t<decltype(value)>;
422 if constexpr (std::is_same_v<U, std::nullptr_t>) {
423 bind(static_cast<int>(i));
424 } else {
425 bind(static_cast<int>(i), value);
426 }
427 },
428 params[i]);
429 }
