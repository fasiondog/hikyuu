13// Extended position information (it is suitable for the one-buy-one-sell case only; for the
14// one-buy-multiple-sell case some statistics are inaccurate and are for reference only)
15struct HKU_API PositionExtInfo final {
16 PositionRecord position;
17 price_t maxHighPrice{0.}; // Maximum of the high prices in the period
18 price_t minLowPrice{0.0}; // Minimum of the low prices in the period
19 price_t maxClosePrice{0.}; // Maximum of the close prices in the period
20 price_t minClosePrice{0.0}; // Minimum of the close prices in the period
21 price_t currentClosePrice{0.}; // Current close price
22 price_t maxPullBack1{0.}; // Maximum drawdown ratio 1 (calculated from the maximum and
23 // the minimum close prices only) (negative)
24 price_t maxPullBack2{0.}; // Maximum drawdown ratio 2 (calculated from the maximum high
25 // price and the minimum low price in the period) (negative)
26 price_t currentProfit{0.}; // Current floating profit and loss (excluding the estimated
27 // sell cost)
28
31 price_t currentPullBack1() const {
32 HKU_IF_RETURN(maxClosePrice <= 0., 0.);
33 price_t ret = (maxClosePrice - currentClosePrice) / maxClosePrice;
34 return ret > 0. ? 0. : ret;
35 }
36
39 price_t currentPullBack2() const {
40 HKU_IF_RETURN(maxHighPrice <= 0., 0.);
41 price_t ret = (maxHighPrice - currentClosePrice) / maxHighPrice;
42 return ret > 0. ? 0. : ret;
43 }
44
47 price_t maxFloatingProfit1() const {
48 price_t ret = maxClosePrice * position.number + position.sellMoney - position.buyMoney;
49 return ret < 0. ? 0. : ret;
50 }
51
54 price_t maxFloatingProfit2() const {
55 price_t ret = maxHighPrice * position.number + position.sellMoney - position.buyMoney;
56 return ret < 0. ? 0. : ret;
57 }
58
61 price_t minLossProfit1() const {
62 price_t ret = minClosePrice * position.number + position.sellMoney - position.buyMoney;
63 return ret > 0. ? 0. : ret;
64 }
65
69 price_t minLossProfit2() const {
70 price_t ret = minLowPrice * position.number + position.sellMoney - position.buyMoney;
71 return ret > 0. ? 0. : ret;
72 }
73
74 PositionExtInfo() = default;
75 PositionExtInfo(const PositionExtInfo&) = default;
76 PositionExtInfo& operator=(const PositionExtInfo&) = default;
77
78 PositionExtInfo(PositionExtInfo&& rhs)
79 : position(std::move(rhs.position)),
80 maxHighPrice(rhs.maxHighPrice),
81 minLowPrice(rhs.minLowPrice),
82 maxClosePrice(rhs.maxClosePrice),
83 minClosePrice(rhs.minClosePrice),
84 currentClosePrice(rhs.currentClosePrice),
85 maxPullBack1(rhs.maxPullBack1),
86 maxPullBack2(rhs.maxPullBack2),
87 currentProfit(rhs.currentProfit) {}
88
89 PositionExtInfo& operator=(PositionExtInfo&& rhs) {
90 if (this != &rhs) {
91 position = std::move(rhs.position);
92 maxHighPrice = rhs.maxHighPrice;
93 minLowPrice = rhs.minLowPrice;
94 maxClosePrice = rhs.maxClosePrice;
95 minClosePrice = rhs.minClosePrice;
96 currentClosePrice = rhs.currentClosePrice;
97 maxPullBack1 = rhs.maxPullBack1;
98 maxPullBack2 = rhs.maxPullBack2;
99 currentProfit = rhs.currentProfit;
100 }
101 return *this;
102 }
