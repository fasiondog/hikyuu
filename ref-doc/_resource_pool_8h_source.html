82 explicit ResourcePool(const Parameter &param, size_t maxPoolSize = 0, size_t maxIdleNum = 100)
83 : m_maxPoolSize(maxPoolSize), m_maxIdelSize(maxIdleNum), m_count(0), m_param(param) {}

88 virtual ~ResourcePool() {
89 std::unique_lock<std::mutex> lock(m_mutex);
90
91 // Mark closed and wake all waiters, then wait until every blocked getter has left so it
92 // never touches the pool after destruction
93 m_closed = true;
94 m_cond.notify_all();
95 m_exit_cond.wait(lock, [this] { return m_waiting == 0; });
96
97 // Unbind the closer of all the allocated resources from the pool
98 for (auto iter = m_closer_set.begin(); iter != m_closer_set.end(); ++iter) {
99 (*iter)->unbind();
100 }
101
102 // Delete all the idle resources
103 while (!m_resourceList.empty()) {
104 ResourceType *p = m_resourceList.front();
105 m_resourceList.pop();
106 if (p) {
107 delete p;
108 }
109 }
110 }

144 ResourcePtr get() {
145 std::lock_guard<std::mutex> lock(m_mutex);
146 if (m_closed) {
147 HKU_THROW_EXCEPTION(ResourcePoolClosedException, "Failed get resource!");
148 }
149 ResourcePtr result;
150 ResourceType *p = nullptr;
151 if (m_resourceList.empty()) {
152 if (m_maxPoolSize > 0 && m_count >= m_maxPoolSize) {
153 return result;
154 }
155 try {
156 p = new ResourceType(m_param);
157 } catch (const std::exception &e) {
158 HKU_THROW_EXCEPTION(CreateResourceException, "Failed create a new Resource! {}",
159 e.what());
160 } catch (...) {
161 HKU_THROW_EXCEPTION(CreateResourceException,
162 "Failed create a new Resource! Unknown error!");
163 }
164 m_count++;
165 result = ResourcePtr(p, ResourceCloser(this));
166 m_closer_set.insert(std::get_deleter<ResourceCloser>(result));
167 return result;
168 }
169 p = m_resourceList.front();
170 m_resourceList.pop();
171 result = ResourcePtr(p, ResourceCloser(this));
172 m_closer_set.insert(std::get_deleter<ResourceCloser>(result));
173 return result;

181 ResourcePtr getWaitFor(uint64_t ms_timeout) { // NOSONAR
182 std::unique_lock<std::mutex> lock(m_mutex);
183 if (m_closed) {
184 HKU_THROW_EXCEPTION(ResourcePoolClosedException, "Failed get resource!");
185 }
186 ResourcePtr result;
187 ResourceType *p = nullptr;
188 if (m_resourceList.empty()) {
189 if (m_maxPoolSize > 0 && m_count >= m_maxPoolSize) {
190 m_waiting++;
191 auto pred = [this] { return m_closed || !m_resourceList.empty(); };
192 if (ms_timeout > 0) {
193 m_cond.wait_for(lock, std::chrono::duration<uint64_t, std::milli>(ms_timeout),
194 pred);
195 } else {
196 m_cond.wait(lock, pred);
197 }
198 m_waiting--;
199 m_exit_cond.notify_all();
200 if (m_closed) {
201 HKU_THROW_EXCEPTION(ResourcePoolClosedException, "Failed get resource!");
202 }
203 if (ms_timeout > 0 && m_resourceList.empty()) {
204 HKU_THROW_EXCEPTION(GetResourceTimeoutException, "Failed get resource!");
205 }
206 } else {
207 try {
208 p = new ResourceType(m_param);
209 } catch (const std::exception &e) {
210 HKU_THROW_EXCEPTION(CreateResourceException, "Failed create a new Resource! {}",
211 e.what());
212 } catch (...) {
213 HKU_THROW_EXCEPTION(CreateResourceException,
214 "Failed create a new Resource! Unknown error!");
215 }
216 m_count++;
217 result = ResourcePtr(p, ResourceCloser(this));
218 m_closer_set.insert(std::get_deleter<ResourceCloser>(result));
219 return result;
220 }
221 }
222 p = m_resourceList.front();
223 m_resourceList.pop();
224 result = ResourcePtr(p, ResourceCloser(this));

281 class ResourceCloser {
282 public:
283 explicit ResourceCloser(ResourcePool *pool) : m_pool(pool) { // NOSONAR

285
286 void operator()(ResourceType *conn) { // NOSONAR
287 if (conn) {
288 // If the pool is bound, the resource is returned; otherwise it is deleted
289 if (m_pool) {
290 // HKU_DEBUG("retuan to pool");
291 m_pool->returnResource(conn, this);
292 } else {
293 // HKU_DEBUG("delete resource not in pool");
294 delete conn;
295 }
296 }
297 }
298
299 // Unbind the resource pool
300 void unbind() {
301 m_pool = nullptr;
302 }
303
304 private:
305 ResourcePool *m_pool;
306 };
307
309 void returnResource(ResourceType *p, ResourceCloser *closer) {
310 std::unique_lock<std::mutex> lock(m_mutex);
311 if (p) {
312 if (!m_closed && m_resourceList.size() < m_maxIdelSize) {
313 m_resourceList.push(p);
314 m_cond.notify_all();
315 } else {
316 delete p;
317 m_count--;
318 }
319 } else {
320 m_count--;
321 // HKU_WARN("Trying to return an empty pointer!");
322 }
323 if (closer) {
324 m_closer_set.erase(closer); // Remove this closer
325 }
326 }
327
328 std::unordered_set<ResourceCloser *> m_closer_set; // The closers occupying the resources

506 ResourcePtr get() {
507 std::lock_guard<std::mutex> lock(m_mutex);
508 if (m_closed) {
509 HKU_THROW_EXCEPTION(ResourcePoolClosedException, "Failed get resource!");
510 }
511 ResourcePtr result;
512 // Take the first idle resource of the current version; stale entries left by an
513 // incVersion call are destroyed, freeing capacity for a fresh one
514 ResourceType *p = _popCurrentIdle();
515 if (!p) {
516 if (m_maxPoolSize > 0 && m_count >= m_maxPoolSize) {
517 return result;
518 }
519 try {
520 p = new ResourceType(m_param);
521 p->setVersion(m_version);
522 } catch (const std::exception &e) {
523 HKU_THROW_EXCEPTION(CreateResourceException, "Failed create a new Resource! {}",
524 e.what());
525 } catch (...) {
526 HKU_THROW_EXCEPTION(CreateResourceException,
527 "Failed create a new Resource! Unknown error!");
528 }
529 m_count++;
530 }
531 result = ResourcePtr(p, ResourceCloser(this));
532 m_closer_set.insert(std::get_deleter<ResourceCloser>(result));

543 if (m_closed) {
544 HKU_THROW_EXCEPTION(ResourcePoolClosedException, "Failed get resource!");
545 }
546 ResourcePtr result;
547 ResourceType *p = _popCurrentIdle();
548 if (!p) {
549 if (m_maxPoolSize > 0 && m_count >= m_maxPoolSize) {
550 m_waiting++;
551 auto pred = [this] { return m_closed || !m_resourceList.empty(); };
552 if (ms_timeout > 0) {
553 m_cond.wait_for(lock, std::chrono::duration<uint64_t, std::milli>(ms_timeout),
554 pred);
555 } else {
556 m_cond.wait(lock, pred);
557 }
558 m_waiting--;
559 m_exit_cond.notify_all();
560 if (m_closed) {
561 HKU_THROW_EXCEPTION(ResourcePoolClosedException, "Failed get resource!");
562 }
563 p = _popCurrentIdle();
564 if (!p) {
565 // the wait ended with nothing of the current version to take: keep the
566 // original timeout contract for getWaitFor; for the indefinite getAndWait
567 // discarding stale entries always frees capacity, so hitting the limit here
568 // means no resource can be served and reporting is safer than an empty pop
569 if (ms_timeout > 0 || m_count >= m_maxPoolSize) {
570 HKU_THROW_EXCEPTION(GetResourceTimeoutException, "Failed get resource!");
571 }
572 }
573 }
574 if (!p) {
575 try {
576 p = new ResourceType(m_param);
577 p->setVersion(m_version);
578 } catch (const std::exception &e) {
579 HKU_THROW_EXCEPTION(CreateResourceException, "Failed create a new Resource! {}",
580 e.what());
581 } catch (...) {
582 HKU_THROW_EXCEPTION(CreateResourceException,
583 "Failed create a new Resource! Unknown error!");
584 }
585 m_count++;
586 }

633 // Pop the first idle resource carrying the current version, destroying older entries on the
634 // way: the version can also be raised through incVersion, and an old-version resource must
635 // never be handed out (the sibling version pools validate the version at take time the same
636 // way). Must be called with m_mutex held
637 ResourceType *_popCurrentIdle() {
638 while (!m_resourceList.empty()) {
639 ResourceType *p = m_resourceList.front();
640 m_resourceList.pop();
641 if (p && p->getVersion() == m_version) {
642 return p;
643 }
644 delete p;
645 m_count--;
646 }
647 return nullptr;
648 }

691 void returnResource(ResourceType *p, ResourceCloser *closer) {
692 std::unique_lock<std::mutex> lock(m_mutex);
693 if (p) {
694 // When the version of the currently returned resource equals the resource pool version
695 // and the idle resource list is less than the maximum number of the idle resources, the
696 // returned resource is accepted
697 if (!m_closed && p->getVersion() == m_version &&
698 m_resourceList.size() < m_maxIdelSize) {
699 m_resourceList.push(p);
700 m_cond.notify_all();
701 } else {
702 delete p;
703 m_count--;
704 }
