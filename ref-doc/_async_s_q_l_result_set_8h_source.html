58 AsyncSQLResultSet(const AsyncDBConnectPtr& connect, const std::string& sql)
59 : m_connect(connect), m_sql_template("SELECT * FROM {} WHERE {} {} LIMIT {} OFFSET {}") {

121 net::awaitable<size_t> size() const {
122 if (!m_connect) {
123 co_return 0;
124 }
125 std::string sql =
126 fmt::format("select count(1) from {} where {}", TableT::getTableName(), m_where);
127 size_t total = co_await m_connect->queryNumber<size_t>(sql, 0, m_params);
128 // the row limit of the condition caps the reported size, the pages are cut the same way
129 co_return m_limit >= 0 ? std::min(total, static_cast<size_t>(m_limit)) : total;
130 }

186private:
188 void _setup(const std::string& where, const std::string& orderBy, int limit,
189 BoundValues params) {
190 m_params = std::move(params);
191 m_limit = limit;
192
193 std::string text = where;
194 trim(text);
195 m_where = text.empty() ? "1=1" : text;
196
197 if (orderBy.empty()) {
198 m_orderby_inner = "ORDER BY id";
199 } else {
200 m_orderby_inner = fmt::format("{}, id ASC", orderBy);
201 m_orderby_outer = m_orderby_inner;
202 }
203 }
204
209 size_t _pageLimit(size_t page) const {
210 if (m_limit < 0) {
211 return page_size;
212 }
213
214 const size_t offset = page * page_size;
215 const size_t left = static_cast<size_t>(m_limit);
216 return offset >= left ? 0 : std::min(page_size, left - offset);
217 }
218
220 bool _connectOrPage(size_t page) const {
221 return m_connect && _pageLimit(page) > 0;
222 }
223
225 std::string _selectSQL(size_t page) const {
226 return fmt::format(fmt::runtime(m_sql_template), TableT::getTableName(), m_where,
227 m_orderby_inner, _pageLimit(page), page * page_size);
228 }
229
231 net::awaitable<std::vector<TableT>> _loadPage(size_t page) {
232 std::vector<TableT> result;
233 auto st = co_await m_connect->getStatementWithParams(_selectSQL(page), m_params);
234 co_await st->exec();
235
236 while (co_await st->moveNext()) {
237 TableT tmp;
238 tmp.load(st);
239 result.push_back(tmp);
240 }
241 co_return result;
242 }
243
249 net::awaitable<TableT> get(size_t index) {
250 TableT result{Null<TableT>()};
251 if (index == Null<size_t>()) {
252 co_return result;
253 }
254
255 size_t page = index / page_size;
256 if (m_connect && page != m_current_page) {
257 m_buffer.clear();
258 if (_connectOrPage(page)) {
259 m_buffer = co_await _loadPage(page);
260 }
261 m_current_page = page;
262 }
263
264 if (m_buffer.empty()) {
265 co_return result;
266 }
267
268 size_t pos = index - page * page_size;
269 if (pos >= m_buffer.size()) {
270 co_return result;
271 }
272
273 result = m_buffer[index - page * page_size];
274 co_return result;
275 }
276
277private:
278 AsyncDBConnectPtr m_connect;
279 std::vector<TableT> m_buffer;
280 std::string m_where;
281 std::string m_sql_template;
282 std::string m_orderby_inner;
283 std::string m_orderby_outer;
284 BoundValues m_params;
285 int m_limit = -1;
286 size_t m_current_page = Null<size_t>();
287};

289template <class TableT, size_t page_size>
290class AsyncSQLResultSetIterator {
291public:
292 using ResultSet = AsyncSQLResultSet<TableT, page_size>;
293
294 AsyncSQLResultSetIterator() = default;
295 ~AsyncSQLResultSetIterator() = default;
296
297 explicit AsyncSQLResultSetIterator(ResultSet* result_set, size_t index)
298 : m_set(result_set), m_index(index) {
299 // Note: co_await cannot be used directly in the constructor, it needs to be initialized
300 // outside
301 }
302
307 net::awaitable<void> init() {
308 if (m_index != Null<size_t>()) {
309 m_value = co_await m_set->get(m_index);
310 if (!m_value.valid()) {
311 m_index = Null<size_t>();
312 }
313 }
314 co_return;
315 }
316
317 AsyncSQLResultSetIterator(const AsyncSQLResultSetIterator& other)
318 : m_set(other.m_set), m_index(other.m_index), m_value(other.m_value) {}
319
320 AsyncSQLResultSetIterator& operator=(const AsyncSQLResultSetIterator& other) {
321 if (this == &other)
322 return *this;
323 m_index = other.m_index;
324 m_set = other.m_set;
325 m_value = other.m_value;
326 return *this;
327 }
328
329 const TableT& operator*() const {
330 return m_value;
331 }
332
333 TableT& operator*() {
334 return m_value;
335 }
336
337 const TableT* const operator->() const {
338 return &m_value;
339 }
340
341 TableT* operator->() {
342 return &m_value;
343 }
344
349 net::awaitable<AsyncSQLResultSetIterator> operator_pre_increment() {
350 HKU_CHECK_THROW(m_index != Null<size_t>(), std::logic_error,
351 "Cannot increment an end iterator.");
352 m_index++;
353 m_value = co_await m_set->get(m_index);
354 if (!m_value.valid()) {
355 m_index = Null<size_t>();
356 }
357 co_return *this;
358 }
359
360 bool operator!=(const AsyncSQLResultSetIterator& iter) const {
361 return m_index != iter.m_index;
362 }
363
364private:
365 ResultSet* m_set = nullptr;
366 size_t m_index = Null<size_t>();
367 TableT m_value;
368};
