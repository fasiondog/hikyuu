180 public:

188 const_iterator() = default; // Default placeholder; imp_ is a null shared_ptr
189
190 reference operator*() const {
191 return imp_->getKRecord(index_);
192 }
193 pointer operator->() const {

195 }
196
197 const_iterator& operator++() {
198 ++index_;
199 return *this;
200 }
201
202 const_iterator operator++(int) {
203 const_iterator temp = *this;
204 ++index_;
205 return temp;
206 }
207
208 bool operator==(const const_iterator& other) const {
209 return imp_ == other.imp_ && index_ == other.index_;
210 }
211
212 bool operator!=(const const_iterator& other) const {
213 return !(*this == other);
214 }
