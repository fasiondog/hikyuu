65 if (m_driverList.empty()) {

107 void returnDriver(DriverConnectT *p) {
108 std::unique_lock<std::mutex> lock(m_mutex);
109 if (p) {
110 if (m_driverList.size() < m_maxIdelSize) {
111 m_driverList.push(p);
112 m_cond.notify_all();
113 } else {
114 delete p;
115 m_count--;
116 }
