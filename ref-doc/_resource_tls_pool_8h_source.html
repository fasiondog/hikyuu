160 struct ResourceDeleter {
161 ResourceTlsPool *pool;
162 std::thread::id owner_thread_id; // Records the thread ID when the resource is acquired

164 void operator()(ResourceType *resource) const {
165 if (!resource) {
166 return;
167 }
168
169 auto current_thread = std::this_thread::get_id();
170 if (current_thread != owner_thread_id) {
171 // A cross-thread return: the resource is deleted directly to avoid accessing the
172 // thread_local variables. The owner pool state (maintained by its owner thread
173 // only) is not touched, so the pool capacity consumed here is never recovered
174 // afterwards.
175 HKU_WARN(
176 "Resource returned from different thread, deleting directly to avoid "
177 "undefined behavior");
178 delete resource;
179 } else if (pool) {
180 // A same-thread return: it is returned to the pool normally
181 pool->returnResource(resource);
182 } else {
183 // The pool is empty (possibly during the destruction), delete it directly
184 delete resource;
185 }
186 }
