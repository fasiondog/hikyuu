38inline std::string sqlStringLiteral(const std::string &val) {
39 std::string escaped;
40 escaped.reserve(val.size() + 2);
41 escaped += '"';
42 for (char c : val) {
43 if (c == '"') {
44 escaped += '"';
45 escaped += '"';
46 } else {
47 escaped += c;
48 }
49 }
50 escaped += '"';
51 return escaped;
52}

66inline std::string sqlIdentifier(const std::string &name) {
67 HKU_CHECK(!name.empty(), "Invalid table name: it is empty!");
68 HKU_CHECK(name.find('\0') == std::string::npos,
69 "Invalid table name: it must not contain the null character!");
70
71 if (name.size() >= 2 && name.front() == '`' && name.back() == '`') {
72 size_t i = 1;
73 const size_t end = name.size() - 1;
74 while (i < end) {
75 if (name[i] == '`') {
76 if (i + 1 < end && name[i + 1] == '`') {
77 i += 2;
78 continue;
79 }
80 break;
81 }
82 ++i;
83 }
84 if (i == end) {
85 return name;
86 }
87 }
88
89 std::string quoted;
90 quoted.reserve(name.size() + 2);
91 quoted += '`';
92 for (char c : name) {
93 if (c == '`') {
94 quoted += "``";
95 } else {
96 quoted += c;
97 }
98 }
99 quoted += '`';
100 return quoted;
101}

112inline void validateSqlName(const std::string &name) {
113 HKU_CHECK(!name.empty(), "Invalid column name: it is empty!");
114 HKU_CHECK(name.find('\0') == std::string::npos,
115 "Invalid column name: it must not contain the null character!");
116 for (char c : name) {
117 HKU_CHECK(c != '"' && c != '\'' && c != '`' && c != '?' && c != ';' && c != '#',
118 "Invalid column name: {}!", name);
119 }
120 HKU_CHECK(name.find("--") == std::string::npos && name.find("/*") == std::string::npos &&
121 name.find("*/") == std::string::npos,
122 "Invalid column name: {}!", name);
123}

341// Under linux the template specialization of a class member function must be implemented outside
342// the class
343// Otherwise the compilation reports: explicit specialization in non-namespace scope
344template <>
345inline DBCondition Field::in<std::string>(const std::vector<std::string> &vals) {
346 HKU_CHECK(!vals.empty(), "input vals can't be empty!");
347 HKU_CHECK(vals.size() <= MAX_SQL_BIND_PARAMS,
348 "The in condition has {} values, over the limit of {} placeholders!", vals.size(),
349 MAX_SQL_BIND_PARAMS);
350 std::vector<std::string> slots;
351 slots.reserve(vals.size());
352 BoundValues params;
353 params.reserve(vals.size());
354 for (size_t i = 0, len = vals.size(); i < len; ++i) {
355 slots.push_back(fmt::format("?{}", i + 1));
356 params.push_back(vals[i]);
357 }
358 return DBCondition::fromFragment(fmt::format("({} in ({}))", name, fmt::join(slots, ",")),
359 std::move(params));
360}

380// A pointer list carries text values as well: it must not fall back to the generic template, which
381// splices them into the statement unquoted
382template <>
383inline DBCondition Field::in<const char *>(const std::vector<const char *> &vals) {
384 HKU_CHECK(!vals.empty(), "input vals can't be empty!");
385 std::vector<std::string> values;
386 values.reserve(vals.size());
387 for (const auto val : vals) {
388 values.emplace_back(val);
389 }
390 return in(values);
391}
