70 TimeDelta _transTimeDelta(uint64_t time) const {
71 int64_t hours = time / 100;
72 HKU_CHECK(hours >= 0 && hours <= 23, "Invalid time: {}!", time);
73 int64_t mins = time - hours * 100;
74 HKU_CHECK(mins >= 0 && mins <= 59, "Invalid time: {}!", time);
75 return TimeDelta(0, hours, mins);
