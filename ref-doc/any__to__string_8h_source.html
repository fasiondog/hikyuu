36//------------------------------------------------------------------------------
37//
38// The mutual conversion functions between the any_t wrapping the common basic types and std::string
39// For a user-defined type, any_to_string requires it to have a constructor from std::string
40// string_to_any requires the user to implement the std::to_string specialization method
41//
42//------------------------------------------------------------------------------
43
44template <typename ValueT>
45inline std::string any_to_string(const any_t& data) {
46 return any_cast<ValueT>(data).str();
47}
48
49template <typename ValueT>
50inline any_t string_to_any(const std::string& data) {
51 return any_t(ValueT(data));
52}
