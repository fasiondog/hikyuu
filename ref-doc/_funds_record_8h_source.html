35 // Current total assets = cash + long market value + short quantity × (borrow price - current
36 // price)
37 // = cash + market_value + borrow_asset - short_market_value
38 // Current liabilities = borrowed funds + value of the borrowed assets
39 // = borrow_cash + borrow_asset
40 // Current net assets = total assets - liabilities
41 // = cash + market_value - short_market_value - borrow_cash
42 // Currently invested principal value assets = invested principal + invested asset value
43 // = base_cash + base_asset
44 // Current profit = current net assets - currently invested principal value assets
45 // = cash + market_value - short_market_value - borrow_cash - base_cash - base_asset
46
47 // Current total assets
48 price_t total_assets() const {
49 return cash + market_value + borrow_asset - short_market_value;
50 }
51
52 // Current net assets
53 price_t net_assets() const {
54 return cash + market_value - short_market_value - borrow_cash;
55 }
56
57 // Total liabilities
58 price_t total_borrow() const {
59 return borrow_cash + borrow_asset;
60 }
61
62 // Currently invested principal value assets
63 price_t total_base() const {
64 return base_cash + base_asset;
65 }
66
67 // Current profit
68 price_t profit() const {
69 return cash + market_value - short_market_value - borrow_cash - base_cash - base_asset;
70 }
