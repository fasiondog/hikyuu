56 ParamItemRecord() {}
57 ParamItemRecord(const string& name, const boost::any& arg) : name(name) {
58 if (arg.type() == typeid(bool)) {
59 type = "bool";
60 bool x = boost::any_cast<bool>(arg);
61 value = boost::lexical_cast<string>(x);
62 } else if (arg.type() == typeid(int)) {
63 type = "int64";
64 int x = boost::any_cast<int>(arg);
65 value = boost::lexical_cast<string>(x);
66 } else if (arg.type() == typeid(int64_t)) {
67 type = "int64";
68 int64_t x = boost::any_cast<int64_t>(arg);
69 value = boost::lexical_cast<string>(x);
70 } else if (arg.type() == typeid(double)) {
71 type = "double";
72 double x = boost::any_cast<double>(arg);
73 // 以 max_digits10 精度保存，避免 lexical_cast 默认 6 位有效数字的往返精度损失
74 std::ostringstream os;
75 os.precision(std::numeric_limits<double>::max_digits10);
76 os << x;
77 value = os.str();
78 } else if (strcmp(arg.type().name(), typeid(string).name()) == 0) {
79 type = "string";
80 value = boost::any_cast<string>(arg);
81 } else if (strcmp(arg.type().name(), typeid(Datetime).name()) == 0) {
82 type = "Datetime";
83 datetime = boost::any_cast<Datetime>(arg);
84 } else if (strcmp(arg.type().name(), typeid(Stock).name()) == 0) {
85 type = "stock";
86 value = "stock";
87 stock = boost::any_cast<Stock>(arg);
88 } else if (strcmp(arg.type().name(), typeid(Block).name()) == 0) {
89 type = "block";
90 value = "block";
91 block = boost::any_cast<Block>(arg);
92 } else if (strcmp(arg.type().name(), typeid(KQuery).name()) == 0) {
93 type = "query";
94 value = "query";
95 query = boost::any_cast<KQuery>(arg);
96 } else if (strcmp(arg.type().name(), typeid(KData).name()) == 0) {
97 type = "kdata";
98 value = "kdata";
99 kdata = boost::any_cast<KData>(arg);
100 } else if (strcmp(arg.type().name(), typeid(PriceList).name()) == 0) {
101 type = "PriceList";
102 value = "price_list";
103 price_list = boost::any_cast<PriceList>(arg);
104 } else if (strcmp(arg.type().name(), typeid(DatetimeList).name()) == 0) {
105 type = "DatetimeList";
106 value = "date_list";
107 date_list = boost::any_cast<DatetimeList>(arg);
108 } else {
109 type = "Unknown";
110 value = "Unknown";
111 }

304 namespace bs = boost::serialization;
305 size_t total = 0;

308 for (size_t i = 0; i < total; i++) {
309 ar& bs::make_nvp<ParamItemRecord>("Item", record);

488template <typename ValueType>
489void Parameter::set(const string& name, const ValueType& value) {
490 auto iter = m_params.find(name);
491 if constexpr (std::same_as<std::decay_t<ValueType>, boost::any>) {
492 if (iter == m_params.end()) {
493 if (!support(value)) {
494 throw std::logic_error("Unsupported Type! input value type: " +
495 string(value.type().name()));
496 }
497 if (value.type() == typeid(int)) {
498 m_params[name] = static_cast<int64_t>(boost::any_cast<int>(value));
499 } else {
500 m_params[name] = value;
501 }
502 return;
503 }
504
505 if (iter->second.type() != value.type()) {
506 throw std::logic_error("Mismatching type! need type " +
507 string(iter->second.type().name()) + " but value type is " +
508 string(value.type().name()));
509 }
510 iter->second = value;
511
512 } else if constexpr (std::same_as<std::decay_t<ValueType>, int>) {
513 if (iter == m_params.end()) {
514 m_params[name] = static_cast<int64_t>(value);
515 return;
516 }
517
518 if (iter->second.type() != typeid(int64_t) && iter->second.type() != typeid(int)) {
519 throw std::logic_error(
520 "Mismatching type! need type int or int64_t, but value type is " +
521 string(typeid(ValueType).name()));
522 }
523 iter->second = static_cast<int64_t>(value);
524
525 } else {
526 if (iter == m_params.end()) {
527 if (!support(value)) {
528 throw std::logic_error("Unsupported Type! input value type: " +
529 string(typeid(ValueType).name()));
530 }
531 m_params[name] = value;
532 return;
533 }
534
535 if (iter->second.type() != typeid(ValueType)) {
536 if ((iter->second.type() == typeid(int) || iter->second.type() == typeid(int64_t)) &&
537 (typeid(ValueType) == typeid(int) || typeid(ValueType) == typeid(int64_t))) {
538 // int 与 int64_t 视为同类型，允许互设
539 } else {
540 throw std::logic_error("Mismatching type! need type " +
541 string(iter->second.type().name()) + " but value type is " +
542 string(typeid(ValueType).name()));
543 }
544 }
545 iter->second = value;
546 }
547}

559template <>
560inline int Parameter::get(const string& name) const {
561 param_map_t::const_iterator iter;
562 iter = m_params.find(name);
563 if (iter == m_params.end()) {
564 throw std::out_of_range("out_of_range in Parameter::get : " + name);
565 }
566 int64_t x = 0;
567 try {
568 if (iter->second.type() == typeid(int)) {
569 x = boost::any_cast<int>(iter->second);
570 } else {
571 x = boost::any_cast<int64_t>(iter->second);
572 }
573 } catch (...) {
574 throw std::runtime_error("failed conversion param: " + name);
575 }
576 if (x > int64_t(std::numeric_limits<int>::max()) ||
577 x < int64_t(std::numeric_limits<int>::min())) {
578 throw std::out_of_range("out_of_range in Parameter::get<int> : " + name);
579 }
580 return static_cast<int>(x);
581}
