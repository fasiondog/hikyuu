25template <typename CountGetter, typename BarFiller>
26inline bool fill_by_weight_series(const KData& k, size_t start_pos, const CountGetter& get_count,
27 const BarFiller& fill) {
28 const KRecord* kdata = k.data();
29 size_t total = k.size();
30 Datetime lastdate = kdata[total - 1].datetime.startOfDay();
31 StockWeightList sw_list = k.getStock().getWeight(Datetime::min(), lastdate + Days(1));
32 HKU_IF_RETURN(sw_list.empty(), false);
33
34 // Find the first record whose share count is not 0
35 price_t pre_count = 0.0;
36 Datetime pre_sw_date;
37 auto sw_iter = sw_list.begin();
38 for (; sw_iter != sw_list.end(); ++sw_iter) {
39 if (get_count(*sw_iter) > 0) {
40 pre_count = get_count(*sw_iter);
41 pre_sw_date = sw_iter->datetime();
42 break;
43 }
44 }
45 // Return when there is no record with a positive share count, or when its date is later than
46 // the last K-line date
47 HKU_IF_RETURN(sw_iter == sw_list.end() || pre_sw_date > lastdate, false);
48
49 size_t pos = start_pos;
50 for (; sw_iter != sw_list.end(); ++sw_iter) {
51 price_t count = get_count(*sw_iter);
52 Datetime cur_sw_date = sw_iter->datetime();
53 if (count <= 0.0) {
54 continue; // Ignore the ex-rights/ex-dividend record whose outstanding shares are 0
55 }
56
57 while (pos < total && kdata[pos].datetime < cur_sw_date) {
58 if (kdata[pos].datetime >= pre_sw_date) {
59 fill(pos, kdata[pos], pre_count);
60 }
61 pos++;
62 }
63
64 pre_count = count;
65 pre_sw_date = cur_sw_date;
66 HKU_IF_RETURN(pos >= total, true);
67 }
68
69 for (; pos < total; pos++) {
70 fill(pos, kdata[pos], pre_count);
71 }
72 return true;
73}
