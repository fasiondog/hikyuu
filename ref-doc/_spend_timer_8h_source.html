279class SpendTimerGuad {
280public:
286 explicit SpendTimerGuad(bool open) : m_open(open), m_old_open(false) {
287 m_old_open = !SpendTimer::isClosed();
288 if (m_open == m_old_open) {
289 return;
290 }
291 if (m_open) {
292 open_spend_time();
293 } else {
294 close_spend_time();
295 }
296 }
297
299 ~SpendTimerGuad() {
300 if (m_open == m_old_open) {
301 return;
302 }
303 if (m_old_open) {
304 open_spend_time();
305 } else {
306 close_spend_time();
307 }
308 }
309
310private:
311 bool m_open;
312 bool m_old_open;
313};
