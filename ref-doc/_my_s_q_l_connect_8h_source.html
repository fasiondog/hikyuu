29inline bool isReadOnlySql(const std::string &sql) {
30 size_t pos = sql.find_first_not_of(" \t\r\n(");
31 if (pos == std::string::npos) {
32 return false;
33 }
34 std::string word;
35 while (pos < sql.size() && std::isalpha(static_cast<unsigned char>(sql[pos]))) {
36 word += static_cast<char>(std::toupper(static_cast<unsigned char>(sql[pos])));
37 ++pos;
38 }
39 return word == "SELECT" || word == "SHOW" || word == "DESC" || word == "DESCRIBE" ||
40 word == "EXPLAIN";
41}
