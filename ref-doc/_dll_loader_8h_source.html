35 explicit DllLoader(const std::vector<std::string>& path) {
36 if (!path.empty()) {
37 m_search_paths.resize(path.size());
38 std::copy(path.begin(), path.end(), m_search_paths.begin());
39 } else {
40 HKU_WARN("DllLoader: empty search path! Use default search path!");
41 initDefaultSearchPath();
42 }
43 }

48 DllLoader(DllLoader&& rhs)
49 : m_handle(rhs.m_handle), m_search_paths(std::move(rhs.m_search_paths)) {
50 rhs.m_handle = nullptr;

53 DllLoader& operator=(DllLoader&& rhs) {
54 if (this != &rhs) {
55 unload();
56 m_handle = rhs.m_handle;
57 m_search_paths = std::move(rhs.m_search_paths);
58 rhs.m_handle = nullptr;
59 }
60 return *this;

67 bool load(const std::string& dllname) noexcept {
68 std::string filename = search(dllname);
69 HKU_WARN_IF_RETURN(filename.empty(), false, "Not found dll: {}!", dllname);
70
71#if HKU_OS_WINDOWS
72 m_handle = LoadLibrary(HKU_PATH(filename).c_str());
73#else
74 m_handle = dlopen(filename.c_str(), RTLD_LAZY);
75#endif
76 HKU_WARN_IF_RETURN(!m_handle, false, "load dll({}) failed!", filename);
77 return true;
78 }
79
80 void unload() noexcept {
81 if (m_handle) {
82#if HKU_OS_WINDOWS
83 FreeLibrary(m_handle);
84#else
85 dlclose(m_handle);
86#endif
87 m_handle = nullptr;
88 }
89 }

91 std::string search(const std::string& dllname) const noexcept {
92#if HKU_OS_WINDOWS
93 std::string dll = fmt::format("{}.dll", dllname);
94#elif HKU_OS_OSX
95 std::string dll = fmt::format("lib{}.dylib", dllname);
96#else
97 std::string dll = fmt::format("lib{}.so", dllname);
98#endif
99 for (const auto& path : m_search_paths) {
100 auto filename = fmt::format("{}/{}", path, dll);
101 if (existFile(filename)) {
102 return filename;
103 }
104 }
105
106 return std::string();
107 }
108
109 void* getSymbol(const char* symbol) noexcept {
110#if HKU_OS_WINDOWS
111 void* func = GetProcAddress(m_handle, symbol);
112#else
113 void* func = dlsym(m_handle, symbol);
114#endif
115 return func;
116 }

118private:
119 void initDefaultSearchPath() noexcept {
120 m_search_paths.emplace_back(".");
121
122 std::string userdir = getUserDir();
123 if (!userdir.empty()) {
124 m_search_paths.emplace_back(userdir + "/lib");
125 }
126
127#if HKU_OS_WINDOWS
128 m_search_paths.emplace_back("C:/Windows/System32");
129 m_search_paths.emplace_back("C:/Windows/SysWOW64");
130 m_search_paths.emplace_back("C:/Windows");
131 const char* path = getenv("PATH");
132 if (path) {
133 std::string pathstr(path);
134 auto items = split(pathstr, ";");
135 for (auto& item : items) {
136 std::string nitem(item);
137 trim(nitem);
138 if (!item.empty()) {
139 m_search_paths.emplace_back(nitem);
140 }
141 }
142 }
143
144#elif HKU_OS_OSX
145 const char* path = getenv("DYLD_IMAGE_SUFFIX");
146 if (path) {
147 std::string pathstr(path);
148 auto items = split(pathstr, ";");
149 for (auto& item : items) {
150 std::string nitem(item);
151 trim(nitem);
152 if (!item.empty()) {
153 m_search_paths.emplace_back(nitem);
154 }
155 }
156 }
157 // DYLD_FRAMEWORK_PATH and DYLD_LIBRARY_PATH are colon-separated path lists
158 path = getenv("DYLD_FRAMEWORK_PATH");
159 if (path) {
160 std::string pathstr(path);
161 auto items = split(pathstr, ":");
162 for (auto& item : items) {
163 std::string nitem(item);
164 trim(nitem);
165 if (!item.empty()) {
166 m_search_paths.emplace_back(nitem);
167 }
168 }
169 }
170 path = getenv("DYLD_LIBRARY_PATH");
171 if (path) {
172 std::string pathstr(path);
173 auto items = split(pathstr, ":");
174 for (auto& item : items) {
175 std::string nitem(item);
176 trim(nitem);
177 if (!item.empty()) {
178 m_search_paths.emplace_back(nitem);
179 }
180 }
181 }
182 m_search_paths.emplace_back("/Library/Frameworks");
183 m_search_paths.emplace_back("/Network/Library/Frameworks");
184 m_search_paths.emplace_back("/System/Library/Frameworks");
185 m_search_paths.emplace_back("/usr/local/lib");
186 m_search_paths.emplace_back("/usr/lib");
187 m_search_paths.emplace_back("/lib");
188
189#else
190 // LD_LIBRARY_PATH is a colon-separated path list
191 const char* path = getenv("LD_LIBRARY_PATH");
192 if (path) {
193 std::string pathstr(path);
194 auto items = split(pathstr, ":");
195 for (auto& item : items) {
196 std::string nitem(item);
197 trim(nitem);
198 if (!item.empty()) {
199 m_search_paths.emplace_back(nitem);
200 }
201 }
202 }
203 m_search_paths.emplace_back("/usr/local/lib");
204 m_search_paths.emplace_back("/usr/lib");
205 m_search_paths.emplace_back("/lib");
206 if (HKU_ARCH_X64) {
207 m_search_paths.emplace_back("/usr/lib/x86_64-linux-gnu");
208 } else if (HKU_ARCH_ARM64) {
209 m_search_paths.emplace_back("/usr/lib/aarch64-linux-gnu");
210 } else if (HKU_ARCH_X86) {
211 m_search_paths.emplace_back("/usr/lib/i386-linux-gnu");
212 } else if (HKU_ARCH_ARM) {
213 m_search_paths.emplace_back("/usr/lib/arm-linux-gnueabihf");
214 }
215#endif
216 }
217

223#endif
