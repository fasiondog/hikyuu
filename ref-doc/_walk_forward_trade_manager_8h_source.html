468//============================================
469// Serialization support
470//============================================
471// A runtime-only view over the host account, rebuilt by WalkForwardSystem on every run; it is
472// never part of an archive. Reaching the throw means a view instance leaked into a serialized
473// object graph (a bug), and loading such an archive could silently break the run_start filter
474#if HKU_SUPPORT_SERIALIZATION
475private:
476 friend class boost::serialization::access;
477 template <class Archive>
478 void serialize(Archive& ar, const unsigned int version) {
479 (void)ar;
480 (void)version;
481 HKU_THROW("WalkForwardTradeManager is a runtime-only view and cannot be serialized");
482 }
483#endif /* HKU_SUPPORT_SERIALIZATION */
